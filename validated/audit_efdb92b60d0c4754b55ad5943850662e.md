The premise is incorrect: `POST /v2/keys/csa/export/:ID` is not merely gated by generic auth middleware — it is explicitly wrapped with `auth.RequiresAdminRole(csakc.Export)` in the route table. [1](#0-0) 

`RequiresAdminRole` strictly checks `user.Role != clsessions.UserRoleAdmin` and aborts with `403 Forbidden` otherwise, independent of which authentication method (session or API token) was used to populate the `SessionUserKey`. [2](#0-1) 

Chainlink API tokens are not scoped to a subset of endpoints/resource types ("run-only" tokens don't exist as a distinct scope) — an API token is simply bound to a `User` record, and that `User` has a single `Role` (`admin`/`edit`/`run`/`view`) enforced via `AuthenticateByToken`, which loads the user and sets `SessionUserKey` to that user's actual role. [3](#0-2) 

A token belonging to a `run`-role user hitting `Export` would fail the `RequiresAdminRole` check and receive `403 Forbidden` before `CSAKeysController.Export` (in `core/web/csa_keys_controller.go`) is ever invoked, so the described pivot is not exploitable as stated.

func AuthenticateByToken(c *gin.Context, authr Authenticator) error {
	ctx := c.Request.Context()
	token := &auth.Token{
		AccessKey: c.GetHeader(APIKey),
		Secret:    c.GetHeader(APISecret),
	}
	if token.AccessKey == "" {
		return auth.ErrorAuthFailed
	}

	if token.Secret == "" {
		return auth.ErrorAuthFailed
	}

	// We need to first load the user row so we can compare tokens using the stored salt
	user, err := authr.FindUserByAPIToken(ctx, token.AccessKey)
	if err != nil {
		if errors.Is(err, sql.ErrNoRows) || errors.Is(err, clsessions.ErrUserSessionExpired) {
			return auth.ErrorAuthFailed
		}
		return err
	}

	ok, err := clsessions.AuthenticateUserByToken(token, &user)
	if err != nil {
		return err
	}
	if !ok {
		return auth.ErrorAuthFailed
	}

	c.Set(SessionUserKey, &user)

	return nil
}

// RequiresAdminRole extracts the user object from the context, and asserts the user's role is 'admin'
func RequiresAdminRole(handler func(*gin.Context)) func(*gin.Context) {
	return func(c *gin.Context) {
		user, ok := GetAuthenticatedUser(c)
		if !ok {
			c.Abort()
			jsonAPIError(c, http.StatusUnauthorized, errors.New("not a valid session"))
			return
		}
		if user.Role != clsessions.UserRoleAdmin {
			c.Abort()
			addForbiddenErrorHeaders(c, "admin", string(user.Role), user.Email)
			jsonAPIError(c, http.StatusForbidden, errors.New("Forbidden"))
			return
		}
		handler(c)
	}
