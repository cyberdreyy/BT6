### Title
`PUT /v2/jobs/:ID` allows an edit-role user to silently change a job's `Type` while retaining its numeric ID, causing job-identity confusion - (File: `core/web/jobs_controller.go`)

### Summary
`JobsController.Update` never compares the `job.Type` of the newly-submitted TOML against the type of the job it is replacing before calling `App.DeleteJob` followed by `App.AddJobV2`. Because the ORM re-inserts the new job with the same numeric ID that was just deleted, an edit-role caller can convert an existing job (e.g. `cron`) into a completely different type (e.g. `gateway`) while preserving its `:ID`, breaking the invariant that a PUT request should only mutate the resource/type it claims to.

### Finding Description
The route `PUT /v2/jobs/:ID` is wired to `jc.Update` behind `auth.RequiresEditRole` [1](#0-0) . `Update` binds the request TOML, calls `validateJobSpec` (which derives `jb.Type` purely from the submitted TOML's `type` field, independent of what currently exists at `:ID`), sets `jb.ID` from the URL parameter, deletes the existing job by ID, and then calls `AddJobV2` with the newly parsed job — at no point comparing `jb.Type` to the type of the job being replaced [2](#0-1) .

`App.DeleteJob` deletes purely by `jobID`, agnostic of type [3](#0-2) , and the underlying `orm.InsertJob` explicitly re-uses the caller-supplied `job.ID` ("if job has id, emplace otherwise insert with a new id") [4](#0-3) . Since `jb.SetID(c.Param("ID"))` is called before `AddJobV2`, the new job of arbitrary type is inserted with the exact same numeric ID as the just-deleted job [5](#0-4) .

The existing integration test `TestJobsController_Update_HappyPath` only exercises same-type (OCR→OCR) updates and does not assert that type is preserved or rejected across mismatches [6](#0-5) , confirming there is no test or code path enforcing type consistency.

### Impact Explanation
An edit-role user who knows/enumerates a target job ID (via `GET /v2/jobs`) can replace that job's type entirely (e.g. cron → gateway, or any type → any other supported type) while keeping the same numeric `job.ID`. This corrupts job-identity/audit-trail assumptions: any external system, dashboard, or audit log correlating job ID to a specific type/purpose will now silently observe a different job under the same identity, with the same `AddJobV2` audit event structure but a materially different meaning. This maps to Chainlink's "cross-user response confusion / unauthorized action on another user's job" impact class rather than raw privilege escalation, since it still requires edit-role, but it lets a lower-trust edit user (who might otherwise be restricted to managing specific jobs by convention/ID) silently repurpose the identity of a job they didn't create.

### Likelihood Explanation
Requires only an edit-role session (no admin) and a valid job ID, which is enumerable via `GET /v2/jobs` (no elevated permissions needed for listing) [1](#0-0) . The sequence is a single HTTP PUT request with attacker-controlled TOML of any supported job type; it is deterministic and fully repeatable.

### Recommendation
In `JobsController.Update`, before calling `App.DeleteJob`, fetch the existing job by `:ID` (e.g. via `JobORM().FindJob`) and compare its `Type` to the newly-parsed `jb.Type`; reject the request with `422 Unprocessable Entity` if they differ (or require an explicit confirmation flag if type migration is an intended feature).

### Proof of Concept
Go handler-level integration test plan:
1. Start a test app with cron and gateway support enabled.
2. Create a cron job via `POST /v2/jobs` with a valid cron TOML; capture returned `job.ID`.
3. Send `PUT /v2/jobs/{ID}` with a valid gateway TOML.
4. Assert response is `200 OK` (current buggy behavior) rather than `422`.
5. Query `JobORM().FindJob(ctx, ID)` and assert `jb.Type == job.Gateway` (type changed) while `jb.ID` is unchanged from step 2 — demonstrating the identity/type mismatch.
6. Assert the audit log entry (`audit.JobCreated`) for the new job references the same numeric ID as the deleted cron job's audit trail, confirming loss of type/identity correlation.

func (jc *JobsController) Update(c *gin.Context) {
	request := UpdateJobRequest{}
	if err := c.ShouldBindJSON(&request); err != nil {
		jsonAPIError(c, http.StatusUnprocessableEntity, err)
		return
	}

	jb, status, err := jc.validateJobSpec(c.Request.Context(), request.TOML)
	if err != nil {
		jsonAPIError(c, status, err)
		return
	}

	err = jb.SetID(c.Param("ID"))
	if err != nil {
		jsonAPIError(c, http.StatusUnprocessableEntity, err)
		return
	}

	ctx, cancel := context.WithTimeout(c.Request.Context(), 5*time.Second)
	defer cancel()

	// If the provided job id is not matching any job, delete will fail with 404 leaving state unchanged.
	err = jc.App.DeleteJob(ctx, jb.ID)
	// Error can be either come from ORM or from the activeJobs map.
	if err != nil {
		if errors.Is(err, sql.ErrNoRows) || strings.Contains(err.Error(), "job not found") {
			jsonAPIError(c, http.StatusNotFound, errors.Wrap(err, "failed to update job"))
			return
		}
		jsonAPIError(c, http.StatusInternalServerError, err)
		return
	}

	err = jc.App.AddJobV2(ctx, &jb)
	if err != nil {
		if errors.Is(errors.Cause(err), job.ErrNoSuchKeyBundle) || errors.As(err, &keystore.KeyNotFoundError{}) || errors.Is(errors.Cause(err), job.ErrNoSuchTransmitterKey) || errors.Is(errors.Cause(err), job.ErrNoSuchSendingKey) {
			jsonAPIError(c, http.StatusBadRequest, err)
			return
		}
		jsonAPIError(c, http.StatusInternalServerError, err)
		return
	}

	jsonAPIResponse(c, presenters.NewJobResource(jb), jb.Type.String())
}

func (app *ChainlinkApplication) DeleteJob(ctx context.Context, jobID int32) error {
	// Do not allow the job to be deleted if it is managed by the Feeds Manager
	isManaged, err := app.FeedsService.IsJobManaged(ctx, int64(jobID))
	if err != nil {
		return err
	}

	if isManaged {
		return errors.New("job must be deleted in the feeds manager")
	}

	return app.jobSpawner.DeleteJob(ctx, nil, jobID)
}

func TestJobsController_Update_HappyPath(t *testing.T) {
	t.Parallel()
	ctx := t.Context()
	cfg := configtest.NewGeneralConfig(t, func(c *chainlink.Config, s *chainlink.Secrets) {
		c.OCR.Enabled = new(true)
		c.P2P.V2.Enabled = new(true)
		c.P2P.V2.ListenAddresses = &[]string{fmt.Sprintf("127.0.0.1:%d", freeport.GetOne(t))}
		c.P2P.PeerID = &cltest.DefaultP2PPeerID
	})
	app := cltest.NewApplicationWithConfigAndKey(t, cfg, cltest.DefaultP2PKey)

	require.NoError(t, app.KeyStore.OCR().Add(ctx, cltest.DefaultOCRKey))
	require.NoError(t, app.Start(ctx))

	_, bridge := cltest.MustCreateBridge(t, app.GetDB(), cltest.BridgeOpts{})
	_, bridge2 := cltest.MustCreateBridge(t, app.GetDB(), cltest.BridgeOpts{})

	client := app.NewHTTPClient(nil)

	var jb job.Job
	ocrspec := testspecs.GenerateOCRSpec(testspecs.OCRSpecParams{
		DS1BridgeName: bridge.Name.String(),
		DS2BridgeName: bridge2.Name.String(),
		Name:          "old OCR job",
		EVMChainID:    cltest.FixtureChainID.String(),
	})
	err := toml.Unmarshal([]byte(ocrspec.Toml()), &jb)
	require.NoError(t, err)

	// BCF-2095
	// disable fkey checks until the end of the test transaction
	require.NoError(t, utils.JustError(
		app.GetDB().ExecContext(ctx, `SET CONSTRAINTS job_spec_errors_v2_job_id_fkey DEFERRED`)))

	var ocrSpec job.OCROracleSpec
	err = toml.Unmarshal([]byte(ocrspec.Toml()), &ocrSpec)
	require.NoError(t, err)
	jb.OCROracleSpec = &ocrSpec
	jb.OCROracleSpec.TransmitterAddress = &app.Keys[0].EIP55Address
	err = app.AddJobV2(ctx, &jb)
	require.NoError(t, err)
	dbJb, err := app.JobORM().FindJob(ctx, jb.ID)
	require.NoError(t, err)
	require.Equal(t, dbJb.Name.String, ocrspec.Name)

	// test Calling update on the job id with changed values should succeed.
	updatedSpec := testspecs.GenerateOCRSpec(testspecs.OCRSpecParams{
		DS1BridgeName:      bridge2.Name.String(),
		DS2BridgeName:      bridge.Name.String(),
		Name:               "updated OCR job",
		TransmitterAddress: app.Keys[0].Address.Hex(),
		EVMChainID:         cltest.FixtureChainID.String(),
	})
	require.NoError(t, err)
	body, _ := json.Marshal(web.UpdateJobRequest{
		TOML: updatedSpec.Toml(),
	})
	response, cleanup := client.Put("/v2/jobs/"+strconv.Itoa(int(jb.ID)), bytes.NewReader(body))
	t.Cleanup(cleanup)

	dbJb, err = app.JobORM().FindJob(ctx, jb.ID)
	require.NoError(t, err)
	require.Equal(t, dbJb.Name.String, updatedSpec.Name)

	cltest.AssertServerResponse(t, response, http.StatusOK)
}
