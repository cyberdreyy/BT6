### Finding Description
`GatewayVaultRequestProcessor.authorizeAndStamp` calls `p.authorizer.AuthorizeRequest(ctx, *req)` [1](#0-0) , which internally calls `validateSecretOwnersMatchAuthorized(req, authResult.AuthorizedOwner())` after replay-guard checks and before ID prefixing/param stamping [2](#0-1) . For `MethodSecretsCreate`/`MethodSecretsUpdate`, the function unmarshals the request and calls `validateEncryptedSecretOwnerMismatch`, which loops over **all** entries (`for idx, encryptedSecret := range encryptedSecrets`) and returns an error on the first owner mismatch it finds, rejecting the whole request rather than filtering a subset [3](#0-2) . The same full-batch iteration applies to `MethodSecretsDelete` via `validateSecretIdentifierOwnerMismatch` [4](#0-3) . `MethodSecretsList` is separately validated by comparing `listReq.Owner` to `workflowOwner` [5](#0-4) . There is no method for which this owner check is skipped except `MethodPublicKeyGet`, which carries no owner-scoped secret data and returns `nil` immediately [6](#0-5) ; any other/unregistered method is explicitly rejected via the `default` branch [7](#0-6) .

This full-batch validation is also confirmed by existing tests: `TestValidateSecretOwnersMatchAuthorized_RejectsMalformedBatches` submits a mixed-owner batch (`owner` at index 0, `"0xother"` at index 1) for both create and delete, and asserts the whole call returns an error referencing the mismatched index [8](#0-7) , and `TestJWTBasedAuth_RejectsCreateRequestWithoutWorkflowOwnerWhenIdentifierOwnerDiffers` demonstrates the same rejection through the full `Authorizer.AuthorizeRequest` path [9](#0-8) .

func validateEncryptedSecretOwnerMismatch(encryptedSecrets []*vaultcommon.EncryptedSecret, workflowOwner string) error {
	if len(encryptedSecrets) == 0 {
		return errors.New("request batch must contain at least 1 item")
	}
	for idx, encryptedSecret := range encryptedSecrets {
		if encryptedSecret == nil {
			return fmt.Errorf("encrypted secret must not be nil at index %d", idx)
		}
		if encryptedSecret.Id == nil {
			return fmt.Errorf("secret ID must not be nil at index %d", idx)
		}
		if vaultutils.NormalizeOwner(encryptedSecret.Id.Owner) != vaultutils.NormalizeOwner(workflowOwner) {
			return fmt.Errorf("encrypted secret owner at index %d %q does not match authorized workflow owner %q", idx, encryptedSecret.Id.Owner, workflowOwner)
		}
	}
	return nil
}

func validateEncryptedSecretsUniformOwners(encryptedSecrets []*vaultcommon.EncryptedSecret) error {
	var owner string
	for idx, enc := range encryptedSecrets {
		if enc == nil || enc.Id == nil {
			continue
		}
		if owner == "" {
			owner = enc.Id.Owner
			continue
		}
		if vaultutils.NormalizeOwner(enc.Id.Owner) != vaultutils.NormalizeOwner(owner) {
			return fmt.Errorf("encrypted secret owner at index %d %q does not match batch owner %q", idx, enc.Id.Owner, owner)
		}
	}
	return nil
}
