The reported bug is specific to a reward-integral accounting pattern (Sushiswap-Masterchef-style `rewardIntegral` vs `rewardIntegralFor[account]`) where a global rate is refreshed against updated debt/interest before a user's individual stake is synced. This repository has no analog of that mechanism: fees here (`protocolFeeAmount`, `integratorFeeAmount`) are computed once via `FeeCalcLib.getFeeAmounts` at order-fill/escrow-creation time and baked directly into the immutable `parameters` bytes that get hashed into the escrow's deterministic salt [1](#0-0) [2](#0-1) . There is no ongoing interest accrual, no globally mutable per-token/per-account reward-rate state, and no separate "last synced" checkpoint per user that could drift out of sync with a shared denominator, as seen in `EscrowDst._withdraw` and `EscrowSrc`/`BaseEscrowFactory.createDstEscrow`, which simply read fixed values from the immutables struct rather than recomputing a rate against a live, mutable total [3](#0-2) . Since the protocol's fee/accounting values are fixed at escrow deployment and never recalculated against a fluctuating global denominator afterward, the described root cause (stale per-account state vs. updated global state) has no reachable analog in the escrow/factory/Merkle-invalidation code within this bounty's scope.

    function getFeeAmounts(
        uint256 amount,
        uint256 protocolFee,
        uint256 integratorFee,
        uint256 integratorShare
    ) internal pure returns (uint256 integratorFeeAmount, uint256 protocolFeeAmount) {
        uint256 denominator = _BASE_1E5 + integratorFee + protocolFee;
        uint256 integratorFeeTotal = amount.mulDiv(integratorFee, denominator);
        integratorFeeAmount = integratorFeeTotal.mulDiv(integratorShare, _BASE_1E2);
        protocolFeeAmount = amount.mulDiv(protocolFee, denominator) + integratorFeeTotal - integratorFeeAmount;
    }
}

    {
        uint256 integratorFeeAmount = immutables.integratorFeeAmountCd();
        uint256 protocolFeeAmount = immutables.protocolFeeAmountCd();
        if (integratorFeeAmount > 0) {
            _uniTransfer(immutables.token.get(), immutables.integratorFeeRecipientCd().get(), integratorFeeAmount);

            _uniTransfer(immutables.token.get(), immutables.protocolFeeRecipientCd().get(), protocolFeeAmount);
        }
        uint256 amount = immutables.amount - integratorFeeAmount - protocolFeeAmount;
        _uniTransfer(immutables.token.get(), immutables.maker.get(), amount);
        _ethTransfer(msg.sender, immutables.safetyDeposit);
        emit EscrowWithdrawal(secret);
    }
