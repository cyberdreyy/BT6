**Reasoning:** `HandleNodeMessage` is never directly reachable by an unauthenticated/unprivileged external caller. The `nodeAddr` parameter passed into `handler.HandleNodeMessage` in `core/services/gateway/handlers/vault/handler.go` (line 489) is not attacker-supplied data extracted from message content — it is supplied by `donConnectionManager.readLoop` in `core/services/gateway/connectionmanager.go`, which is a per-connection goroutine keyed to the node address that was cryptographically verified during the handshake. [1](#0-0) 

That handshake requires the connecting party to sign a server-issued random challenge with the private key corresponding to the claimed node address, verified in `FinalizeHandshake`: [2](#0-1) 

and the initial auth header itself must be signed and matched to a known node in the DON config in `StartHandshake`: [3](#0-2) 

Only after this signature-verified handshake does the gateway create a per-node connection (`nodeState.conn`) whose `readLoop` invokes `handler.HandleNodeMessage(ctx, &resp, nodeAddress)` with the `nodeAddress` string bound to that specific authenticated connection — not derived from the message payload itself. There is no HTTP route, JSON-RPC method, or gateway-user-facing endpoint that lets an external, unprivileged caller invoke `HandleNodeMessage` with an arbitrary `nodeAddr` of their choosing; it is only invoked internally per already-authenticated node connection.

The premise in the question — "an unprivileged external caller could exploit this only if the node-port transport itself doesn't enforce mutual auth for this specific call" — does not hold: the node-port websocket transport does enforce challenge/response signature-based mutual authentication before any node message is dispatched to a handler, and the `nodeAddr` is a property of the authenticated connection, not user-controllable input. Directly calling `h.HandleNodeMessage(ctx, forgedResp, "legit-node-addr")` as suggested in the PoC is not a reachable attacker action — it requires code-level access to the `handler` struct, which an unprivileged external attacker does not have; the only way to reach that code path is through a connection that has already passed the ECDSA-based handshake in `connectionmanager.go`. This falls into the excluded "malicious-node / network-layer" and "requires prior compromise" categories rather than an unauthenticated attacker path.

func (m *connectionManager) StartHandshake(authHeader []byte) (attemptId string, challenge []byte, err error) {
	m.lggr.Debug("StartHandshake")
	authHeaderElems, signer, err := network.UnpackSignedAuthHeader(authHeader)
	if err != nil {
		return "", nil, errors.Join(network.ErrAuthHeaderParse, err)
	}
	nodeAddress := "0x" + hex.EncodeToString(signer)
	donConnMgr, ok := m.dons[authHeaderElems.DonId]
	if !ok {
		return "", nil, network.ErrAuthInvalidDonId
	}
	nodeState, ok := donConnMgr.nodes[nodeAddress]
	if !ok {
		return "", nil, network.ErrAuthInvalidNode
	}
	if authHeaderElems.GatewayId != m.config.AuthGatewayId {
		return "", nil, network.ErrAuthInvalidGateway
	}
	nowTs := uint32(m.clock.Now().Unix())
	ts := authHeaderElems.Timestamp
	if ts < nowTs-m.config.AuthTimestampToleranceSec || nowTs+m.config.AuthTimestampToleranceSec < ts {
		return "", nil, network.ErrAuthInvalidTimestamp
	}
	attemptId, challenge, err = m.newAttempt(nodeState, nodeAddress, ts)
	if err != nil {
		return "", nil, err
	}
	return attemptId, challenge, nil
}

func (m *connectionManager) FinalizeHandshake(attemptId string, response []byte, conn *websocket.Conn) error {
	m.lggr.Debugw("FinalizeHandshake", "attemptId", attemptId)
	m.connAttemptsMu.Lock()
	attempt, ok := m.connAttempts[attemptId]
	delete(m.connAttempts, attemptId)
	m.connAttemptsMu.Unlock()
	if !ok {
		return network.ErrChallengeAttemptNotFound
	}
	signer, err := common.ExtractSigner(response, network.PackChallenge(&attempt.challenge))
	if err != nil || attempt.nodeAddress != "0x"+hex.EncodeToString(signer) {
		return network.ErrChallengeInvalidSignature
	}

func (m *donConnectionManager) readLoop(nodeAddress string, nodeState *nodeState) {
	ctx, cancel := m.shutdownCh.NewCtx()
	defer cancel()
	for {
		select {
		case <-m.shutdownCh:
			m.closeWait.Done()
			return
		case item := <-nodeState.conn.ReadChannel():
			var resp jsonrpc.Response[json.RawMessage]
			err := json.Unmarshal(item.Data, &resp)
			if err != nil {
				m.lggr.Errorw("parse error when reading from node", "nodeAddress", nodeAddress, "err", err)
				break
			}
			handler, err := m.getHandler(resp.Method)
			if err != nil {
				m.lggr.Errorw("no handler for node message", "nodeAddress", nodeAddress, "method", resp.Method, "err", err)
				break
			}
			startTime := time.Now()
			err = handler.HandleNodeMessage(ctx, &resp, nodeAddress)
			m.gMetrics.RecordNodeMsgHandlerInvocation(ctx, nodeAddress, nodeState.name, err == nil)
			m.gMetrics.RecordNodeMsgHandlerDuration(ctx, nodeAddress, nodeState.name, time.Since(startTime), err == nil)
			if err != nil {
				m.lggr.Errorw("error when calling HandleNodeMessage", "err", err, "nodeAddress", nodeAddress, "nodeState", nodeState.name, "responseID", resp.ID)
			}
