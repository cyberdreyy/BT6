### Title
Non-returning ERC20 tokens are permanently unrecoverable from cooldown contracts - (File: contracts/strategies/ethena/EthenaCooldownRequest.sol)

### Summary
`EthenaCooldownRequest.rescue` is the only path for removing arbitrary ERC20 tokens sent to a cooldown contract, but it calls `IERC20Detailed(_token).transfer` through an interface that expects a `bool` return value. [1](#0-0)  Tokens such as USDT return no value, so their transfer succeeds externally but the Solidity ABI decoder reverts on the empty return data. 

### Finding Description
A cooldown clone can receive any ERC20 token, while `rescue` is restricted to `TL_MULTISIG` and transfers the contract’s entire balance of the supplied token. [2](#0-1)  Although the function does not explicitly require the return value to be true, `IERC20Detailed.transfer` declares a boolean return value, causing calls to non-returning ERC20 implementations to revert after the external call completes.  `unstake` only exits sUSDe through Ethena and cannot recover unrelated tokens. [3](#0-2) 

### Impact Explanation
The entire balance of any ERC20 token that does not return a boolean is permanently locked in the cooldown contract. [4](#0-3)  An unprivileged direct token sender can fund the contract, and even the honest `TL_MULTISIG` cannot recover a token such as USDT. [4](#0-3) 

### Likelihood Explanation
No privileged attacker or protocol failure is required; an ordinary accidental ERC20 transfer is sufficient. [4](#0-3)  Likelihood is constrained by the need for a non-standard-return token to be sent to the clone, but every such deposited token is irrecoverable once sent. 

### Recommendation
Use OpenZeppelin `SafeERC20.safeTransfer` for the arbitrary-token rescue path, because it accepts both boolean-returning and non-returning ERC20 implementations. [4](#0-3)

```solidity
contract EthenaCooldownRescueTest is Test {

        0xdAC17F958D2ee523a2206206994597C13D831ec7;
    address internal constant TL_MULTISIG =
        0xFb3bD022D5DAcF95eE28a6B07825D4Ff9C5b3814;

    function testCannotRescueNonReturningToken() public {
        vm.createSelectFork(vm.envString("MAINNET_RPC_URL"));

        EthenaCooldownRequest request = new EthenaCooldownRequest();
        uint256 amount = 100_000e6;

        deal(USDT, address(request), amount);

        vm.prank(TL_MULTISIG);
        vm.expectRevert();
        request.rescue(USDT);
    }
}
```

```text
contract EthenaCooldownRequest is Clone {
  address internal constant SUSDE = 0x9D39A5DE30e57443BfF2A8307A4256c8797A3497;
  address internal constant TL_MULTISIG = 0xFb3bD022D5DAcF95eE28a6B07825D4Ff9C5b3814;
  
  /// @notice only the cdo can call this otherwise one could send 
  /// small amount of funds to this contract and request another
  /// cooldown which will restart the cooldown period
  function startCooldown() external {
    require(msg.sender == _getCDO(), '6');
    IStakedUSDeV2(SUSDE).cooldownShares(IERC20Detailed(SUSDE).balanceOf(address(this)));
  }

  /// @notice anyone can call this as the receiver is the user who
  /// created the contract in the first place
  function unstake() external {
    IStakedUSDeV2(SUSDE).unstake(_getUser());
  }

  /// @notice rescue any token from the contract
  function rescue(address _token) external {
    require(msg.sender == TL_MULTISIG, '6');
    IERC20Detailed(_token).transfer(msg.sender, IERC20Detailed(_token).balanceOf(address(this)));
  }
```
