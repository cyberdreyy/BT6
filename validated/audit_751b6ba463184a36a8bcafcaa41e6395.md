Based on the analysis, the described bug does not hold. `filepath.Abs`/`filepath.Rel`/`filepath.Clean` in Go's standard library already lexically normalize `.` segments, trailing slashes, and duplicate slashes before the `..` prefix check is performed in `fileArchiver.findRelativePathInProject` and `fileArchiver.process`, so the exact quirks hypothesized (trailing slashes, `.` segments) cannot produce a resolved-outside-wd path whose `rel` string fails to start with `".."`. [1](#0-0) 

Critically, this is a **double-gated** design: even if a crafted glob pattern's non-glob "base" prefix (checked by `findRelativePathInProject`) resolves inside `wd`, but the glob "pattern" suffix (`patt`, from `doublestar.SplitPattern`) contains smuggled `../` segments that are only joined in afterward without a fresh Abs/Rel check, every actual matched file that comes out of `doublestar.FilepathGlob` is walked and re-validated independently in `fileArchiver.process`, which redoes `filepath.Abs` + `filepath.Rel` and rejects anything whose relative path is prefixed with `".."`. [2](#0-1) [3](#0-2) 

As for the `shell.Expand` step in `normalizeArgs`, expansion of `$CI_PROJECT_DIR` resolves through the OS environment set by the runner itself (the actual working directory), not an attacker-supplied value — a job author can only append literal `../..` segments to that expanded base, which is exactly the scenario the two-layer check (and the existing test suite) is built to catch. [4](#0-3) 

The existing test suite in `file_archiver_test.go` already directly covers the scenarios named in the question — absolute paths outside the project, relative `../../../relative.txt` paths, and (in `TestFileArchiver_pathIsInProject`) explicit cases for trailing slashes on both absolute and relative working-directory paths, and `../*.*` patterns — all asserting rejection. [5](#0-4) [6](#0-5) 

No plausible normalization quirk in the standard `path/filepath` package (which the hypothesis is banking on) exists that would cause `filepath.Rel` to return a path lacking the `".."` prefix while the resolved absolute path is actually outside `c.wd`, because `Rel` computes its result from `Clean`-ed absolute paths on both sides, making such an escape a Go standard-library bug rather than a Runner-specific one.

func (c *fileArchiver) processPath(path string) {
	if path == "" {
		logrus.Warningf("No matching files. Path is empty.")
		return
	}

	rel, err := c.findRelativePathInProject(path)
	if err != nil {
		// Do not fail job when a file is invalid or not found.
		logrus.Warningf("processPath: %v", err.Error())
		return
	}

	// Use WithNoFollow option to prevent symlink cycles during the initial glob
	matches, err := doublestar.FilepathGlob(rel, doublestar.WithNoFollow())
	if err != nil {
		logrus.Warningf("%s: %v", path, err)
		return
	}

	found := 0

	for _, match := range matches {
		err := filepath.Walk(match, func(path string, info os.FileInfo, err error) error {
			if c.process(path) {
				found++
			}
			return nil
		})
		if err != nil {
			logrus.Warningln("Walking", match, err)
		}
	}

	if found == 0 {
		logrus.Warningf(
			"%s: no matching files. Ensure that the artifact path is relative to the working directory (%s)",
			path,
			c.wd,
		)
	} else {
		logrus.Infof("%s: found %d matching artifact files and directories", path, found)
	}
}

func (c *fileArchiver) findRelativePathInProject(path string) (string, error) {
	slashPath := filepath.ToSlash(path)
	if filepath.Clean(slashPath) == filepath.Clean(c.wd) {
		return ".", nil
	}

	base, patt := slashPath, ""
	// check if path contains a glob pattern
	if strings.ContainsAny(slashPath, "*?[{") {
		base, patt = doublestar.SplitPattern(slashPath)
	}

	abs, err := filepath.Abs(base)
	if err != nil {
		return "", fmt.Errorf("could not resolve artifact absolute path %s: %w", path, err)
	}

	rel, err := filepath.Rel(c.wd, abs)
	if err != nil {
		return "", fmt.Errorf("could not resolve artifact relative path %s: %w", path, err)
	}

	// If fully resolved relative path begins with ".." it is not a subpath of our working directory
	if strings.HasPrefix(rel, ".."+string(filepath.Separator)) || rel == ".." {
		return "", fmt.Errorf("artifact path is not a subpath of project directory: %s", path)
	}

	// Relative path is needed now that our fsys "root" is at the working directory
	rel = filepath.Join(rel, patt)
	rel = filepath.FromSlash(rel)
	return rel, nil
