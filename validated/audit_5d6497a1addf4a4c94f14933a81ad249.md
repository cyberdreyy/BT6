### Title
Pending-withdrawal interest is excluded from programmable-borrower reserves - (`contracts/strategies/idle/ProgrammableBorrower.sol`)

### Summary
`ProgrammableBorrower` reserves only the withdrawal principal present at epoch start, but `IdleCreditVault.prepareStopEpochWithApr0` can later increase `pendingWithdraws` by the APR0 withdrawal bucket’s share of epoch interest. [1](#0-0) [2](#0-1) 

An honest borrower can therefore draw all funds reported as available while leaving the facility short of the eventual withdrawal liability, causing a solvent epoch to enter the default path. [3](#0-2) [4](#0-3) 

### Finding Description
`IdleCDOEpochVariant.requestWithdraw` removes the receipt principal from active NAV, while `IdleCreditVault.requestWithdraw` adds the current underlying claim to `pendingWithdraws` and tracks APR0 principal separately. [5](#0-4) [6](#0-5) 

When an epoch starts, `onStartEpoch` snapshots only that current `pendingWithdraws` value into `epochPendingWithdraws`. [1](#0-0) 

`availableToBorrow` subtracts only this stale principal reservation from cash plus vault assets, and `_borrow` allows the borrower to draw the full resulting amount. [7](#0-6) [3](#0-2) 

At stop time, `prepareStopEpochWithApr0` assigns a pro-rata share of realized epoch interest to the pending withdrawal bucket and increases `pendingWithdraws`. [8](#0-7) 

`stopEpoch` then requests the enlarged pending amount from the programmable borrower, but the facility may only hold the originally reserved principal after the permitted draw. [9](#0-8) 

### Impact Explanation
With a 10,000 USDC pool and a 5,000 USDC APR0 withdrawal request, `borrow(0)` can withdraw the other 5,000 USDC while only 5,000 USDC remains reserved. [7](#0-6) 

If the borrowed principal accrues 50 USDC of borrower interest, the pending bucket receives approximately 25 USDC when active principal and pending principal are equal, so `stopEpoch` requests 5,025 USDC while the facility holds only 5,000 USDC. [10](#0-9) [11](#0-10) 

The failed transfer marks the CDO as defaulted, pauses deposits, stops the epoch, and disables new withdrawal requests, temporarily freezing both the 5,025 USDC pending claim and active LP funds until recovery/finalization. [12](#0-11) 

### Likelihood Explanation
This requires the programmable-borrower minted-interest configuration, a positive pending APR0 withdrawal, and a borrower drawing the liquidity made available by `availableToBorrow`. [13](#0-12) [3](#0-2) 

The borrower is not acting maliciously; `borrow(0)` is explicitly designed to draw all liquidity reported as available. [14](#0-13) 

The issue is not prevented by `onStopEpoch`, because that hook only compares the enlarged shortfall with currently held vault assets and cannot reserve interest that was omitted earlier. [15](#0-14) 

### Recommendation
Calculate borrowable liquidity against the projected stop-time withdrawal liability, not only the request-time principal. [7](#0-6) [2](#0-1) 

A safe implementation would reserve `pendingWithdraws` plus the maximum/projected APR0 interest attributable to those receipts, or update `epochPendingWithdraws` as epoch interest accrues so `borrow` cannot consume funds needed by `prepareStopEpochWithApr0`. [1](#0-0) [8](#0-7) 

### Proof of Concept

```solidity
function testApr0PendingWithdrawInterestIsNotReserved() external {
    MockStopEpochLiquidityVault noYieldVault =
        new MockStopEpochLiquidityVault(USDC);

    vm.prank(manager);
    programmableBorrower.setVault(address(noYieldVault));
    vm.prank(manager);
    programmableBorrower.setBorrowerApr(10e18);
    vm.prank(manager);
    strategy.setAprs(0, 0);
    vm.prank(owner);
    cdoEpoch.setIsInterestMinted(true);

    uint256 amount = 10_000 * oneScale;
    idleCDO.depositAA(amount);

    uint256 pendingPrincipal = cdoEpoch.requestWithdraw(
        aaTranche.balanceOf(address(this)) / 2,
        address(aaTranche)
    );

    _startEpochAndCheckPrices(0);
    assertEq(
        programmableBorrower.epochPendingWithdraws(),
        pendingPrincipal,
        "only request principal is reserved"
    );

    vm.prank(revolvingBorrower);
    programmableBorrower.borrow(0);

    assertEq(
        programmableBorrower.totalUnderlying(),
        pendingPrincipal,
        "facility retains only pending principal"
    );

    vm.warp(cdoEpoch.epochEndDate() + 1);

    uint256 interestDue = programmableBorrower.totalInterestDueNow();
    uint256 apr0Principal = strategy.apr0TotalPrincipal();
    uint256 apr0Interest =
        interestDue * apr0Principal /
        (cdoEpoch.getContractValue() + apr0Principal);
    uint256 required = pendingPrincipal + apr0Interest;

    assertGt(apr0Interest, 0, "test requires pending-withdraw interest");
    assertLt(
        programmableBorrower.totalUnderlying(),
        required,
        "facility lacks the enlarged pending withdrawal"
    );

    vm.prank(manager);
    cdoEpoch.stopEpoch(0, 0);

    assertTrue(cdoEpoch.defaulted(), "under-reserved epoch defaulted");
    assertEq(strategy.pendingWithdraws(), required);
    assertFalse(cdoEpoch.allowAAWithdrawRequest());
    assertFalse(cdoEpoch.allowBBWithdrawRequest());
}
```

```text
contract MockStopEpochLiquidityVault is ERC20 {
  IERC20Detailed public immutable assetToken;
  uint256 public withdrawLimit = type(uint256).max;

  constructor(address _asset) ERC20("Stop Epoch Liquidity Vault", "SELV") {
    assetToken = IERC20Detailed(_asset);
  }

  function asset() external view returns (address) {
    return address(assetToken);
  }

  function convertToAssets(uint256 shares) public view returns (uint256) {
    uint256 supply = totalSupply();
    if (supply == 0) return shares;
    return shares * assetToken.balanceOf(address(this)) / supply;
  }

  function maxWithdraw(address owner) external view returns (uint256) {
    uint256 assets = convertToAssets(balanceOf(owner));
    return assets < withdrawLimit ? assets : withdrawLimit;
  }

  function setWithdrawLimit(uint256 assets) external {
    withdrawLimit = assets;
  }

  function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
    uint256 assetsBefore = assetToken.balanceOf(address(this));
    uint256 supply = totalSupply();
    shares = supply == 0 || assetsBefore == 0 ? assets : assets * supply / assetsBefore;
    assetToken.transferFrom(msg.sender, address(this), assets);
    _mint(receiver, shares);
  }

  function withdraw(uint256 assets, address receiver, address owner) external returns (uint256 shares) {
    require(assets <= withdrawLimit, "insufficient-liquidity");
    shares = _toSharesRoundUp(assets);
    if (msg.sender != owner) _spendAllowance(owner, msg.sender, shares);
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }

  function redeem(uint256 shares, address receiver, address owner) external returns (uint256 assets) {
    assets = convertToAssets(shares);
    require(assets <= withdrawLimit, "insufficient-liquidity");
    if (msg.sender != owner) _spendAllowance(owner, msg.sender, shares);
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }

  function _toSharesRoundUp(uint256 assets) internal view returns (uint256 shares) {
    uint256 supply = totalSupply();
    uint256 managedAssets = assetToken.balanceOf(address(this));
    if (supply == 0 || managedAssets == 0) return assets;
    shares = assets * supply / managedAssets;
    if (shares * managedAssets < assets * supply) shares += 1;
  }
```
