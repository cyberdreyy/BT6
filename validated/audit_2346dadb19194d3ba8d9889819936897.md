### Title
Programmable borrower APR below 1% truncates to zero - ([File: contracts/strategies/idle/ProgrammableBorrower.sol])

### Summary
`ProgrammableBorrower._calcInterest` divides `borrowerApr` by `100` before multiplying it by principal and elapsed time, so any configured APR below `1e18` truncates to zero and produces no borrower interest. [1](#0-0) 

### Finding Description
`borrowerApr` is documented as a rate scaled by `1e18`, where `100e18` represents 100%. [2](#0-1) 

The current expression evaluates `(borrowerApr / 100)` using integer division, so a valid `0.99e18` APR becomes zero before the principal and duration multiplications occur. [1](#0-0) 

`borrowerInterestAccruedNow` adds this incorrectly calculated pending interest to the accrued debt used by `totalInterestDueNow`. [3](#0-2) 

At epoch settlement, `totalInterestDueNow` contributes borrower contractual interest to the net interest distributed to the pool. [4](#0-3) 

`IdleCDOEpochVariant.stopEpoch` then resolves programmable-borrower interest, updates accounting, and mints strategy tokens for the minted-interest epoch result. [5](#0-4) 

### Impact Explanation
A programmable facility configured with any positive APR below 1% reports zero borrower interest, permanently under-crediting tranche holders for that accrual period. [1](#0-0) 

For example, a 10 million USDC draw at 0.99% APR for 30 days should accrue approximately 8,136.99 USDC, but the contract reports zero interest. [6](#0-5) 

The loss directly reduces the interest value used by epoch settlement and therefore reduces lender NAV and tranche price appreciation. [7](#0-6) 

### Likelihood Explanation
The issue does not require a malicious privileged actor because low sub-1% APR values are accepted by the normal APR setters without a minimum-rate guard. [8](#0-7) 

The loss occurs during ordinary borrow, accrual, and `stopEpoch` execution once such a rate is configured and principal is outstanding. [9](#0-8) 

Likelihood is limited by the requirement that deployments use programmable-borrower accounting and configure a positive APR below `1e18`. [10](#0-9) 

### Recommendation
Perform all multiplications before division and preserve the intended percentage denominator:

```solidity
// contracts/strategies/idle/ProgrammableBorrower.sol
function _calcInterest(uint256 principal, uint256 elapsed) internal view returns (uint256) {
    return principal * borrowerApr * elapsed / (100 * YEAR * ONE_TRANCHE_TOKEN);
}
```

Alternatively, store `borrowerApr` directly as a `1e18` fraction and calculate `principal * borrowerApr * elapsed / (YEAR * ONE_TRANCHE_TOKEN)`, updating initializer and setter documentation accordingly. [11](#0-10) 

Add a regression test asserting that a sub-1% positive APR accrues nonzero proportional interest. [12](#0-11)

pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";
import {IERC4626} from "../contracts/interfaces/IERC4626.sol";
import {ProgrammableBorrower} from "../contracts/strategies/idle/ProgrammableBorrower.sol";

contract MockCDO {
    address public immutable token;

    constructor(address token_) {
        token = token_;
    }
}

contract ProgrammableBorrowerLowAprTest is Test {
    // Fork a network containing the selected underlying and an ERC4626 vault
    // whose asset is that underlying.
    IERC20Detailed constant UNDERLYING = IERC20Detailed(UNDERLYING_ADDRESS);
    IERC4626 constant VAULT = IERC4626(ERC4626_VAULT_ADDRESS);

    ProgrammableBorrower borrowerAdapter;
    MockCDO cdo;

    address owner = address(this);
    address manager = makeAddr("manager");
    address borrower = makeAddr("borrower");

    function setUp() public {
        cdo = new MockCDO(address(UNDERLYING));
        borrowerAdapter = new ProgrammableBorrower();

        borrowerAdapter.initialize(
            address(VAULT),
            address(cdo),
            owner,
            manager,
            borrower,
            0.99e18 // 0.99% APR: positive, but less than 1e18
        );
    }

    function testSubOnePercentAprAccruesZero() public {
        uint256 principal = 10_000_000e6;
        uint256 duration = 30 days;

        deal(address(UNDERLYING), address(borrowerAdapter), principal);

        vm.prank(address(cdo));
        borrowerAdapter.onStartEpoch(0);

        vm.prank(borrower);
        borrowerAdapter.borrow(principal);

        vm.warp(block.timestamp + duration);

        uint256 expected =
            principal * 0.99e18 * duration / (100 * 365 days * 1e18);

        assertGt(expected, 0);
        assertEq(borrowerAdapter.borrowerInterestAccruedNow(), 0);
        assertEq(borrowerAdapter.totalInterestDueNow(), 0);
    }
}

  function _pendingBorrowerInterest(uint256 principal, uint256 last) internal view returns (uint256) {
    if (principal == 0 || last == 0 || borrowerApr == 0) return 0;
    uint256 elapsed = block.timestamp - last;
    return elapsed == 0 ? 0 : _calcInterest(principal, elapsed);
  }

  function _accrueBorrowerInterest() internal {
    uint256 last = lastBorrowerAccrual;
    uint256 principal = borrowerPrincipal;
    if (last == 0 || principal == 0 || borrowerApr == 0) {
      lastBorrowerAccrual = block.timestamp;
      return;
    }
    uint256 elapsed = block.timestamp - last;
    if (elapsed == 0) return;
    borrowerInterestAccrued += _calcInterest(principal, elapsed);
    lastBorrowerAccrual = block.timestamp;
  }
