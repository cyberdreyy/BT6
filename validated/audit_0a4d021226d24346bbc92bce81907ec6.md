### Analysis

The premise of the question does not hold. The rate limit owner key used in `checkRateLimit` is not derived from unvalidated request content — it is derived from the registry-backed `workflowReference` associated with the **already-resolved and already-authorized** `workflowID`, not from the raw `WorkflowSelector.WorkflowOwner` field submitted by the caller.

Tracing the flow in `HandleUserTriggerRequest`:
1. `resolveWorkflowID` runs first. If the caller supplies `WorkflowID` directly, it is looked up in `workflowMetadataHandler.GetWorkflowReference` and must already exist in the synced registry — the raw `WorkflowOwner` selector field is not even consulted in this path. If instead `WorkflowOwner`/`WorkflowName`/`WorkflowTag` are supplied, `GetWorkflowID` requires an *exact* match against a `workflowReference` that was populated from DON-consensus-aggregated metadata (`syncMetadata`), so an attacker cannot invent an arbitrary owner string that doesn't correspond to a real, registered workflow. [1](#0-0) 

2. `authorizeRequest` is called next with that resolved `workflowID`, and `WorkflowMetadataHandler.Authorize` verifies the JWT signer against the `authorizedKeys` map keyed by that specific `workflowID`. If the attacker's key is not registered as authorized for the victim's `workflowID`, authorization fails and processing stops before `checkRateLimit` is ever reached. [2](#0-1) [3](#0-2) 

3. Only after both checks pass does `checkRateLimit` execute, and it fetches `workflowRef` again from `GetWorkflowReference(workflowID)` — the same registry-backed source of truth for the same authorized `workflowID`, not the raw request fields. `contexts.WithCRE` is populated with `workflowRef.workflowOwner` (registry data), so the bucket key is tied to the authenticated/resolved identity. [4](#0-3) 

For an attacker to make the rate-limit key resolve to a victim's owner, they would need `resolveWorkflowID` to resolve to the victim's actual registered `workflowID` (which requires exact owner/name/tag match to real registry data — not attacker-invented), and then pass `authorizeRequest` for that same `workflowID` (which requires possessing a signer key that is genuinely registered as authorized for the victim's workflow). Both conditions together mean the "attacker" would already be a legitimately authorized signer for that workflow — this is not a cross-owner isolation break, it's the intended access-control design.

	workflowRef, found := h.workflowMetadataHandler.GetWorkflowReference(workflowID)
	if !found {
		h.handleUserError(ctx, requestID, jsonrpc.ErrInvalidRequest, "workflow reference not found", callback)
		return errors.New("workflow reference not found")
	}

	ctx = contexts.WithCRE(ctx, contexts.CRE{Owner: workflowRef.workflowOwner, Workflow: workflowID})
	if err := h.userRateLimiter.AllowErr(ctx); err != nil {
		lggr := logger.With(h.lggr, platform.KeyWorkflowID, workflowID, platform.KeyWorkflowOwner, workflowRef.workflowOwner, "requestID", requestID, "err", err)
		if errLimited, ok := errors.AsType[limits.ErrorRateLimited](err); ok {
			switch errLimited.Scope {
			case settings.ScopeWorkflow:
				lggr.Errorf("failed to start execution: per workflow rate limit exceeded")
				h.metrics.IncrementWorkflowThrottled(ctx, h.lggr)
			default:
				lggr.Errorf("failed to start execution: unexpected rate limit for scope %s", errLimited.Scope)
			}
			h.handleUserError(ctx, requestID, jsonrpc.ErrLimitExceeded, "rate limit exceeded", callback)
			return err
		}
		return fmt.Errorf("failed to check rate limit: %w", err)
	}
	return nil
}

func (h *WorkflowMetadataHandler) Authorize(workflowID string, token string, req *jsonrpc.Request[json.RawMessage]) (*gateway.AuthorizedKey, error) {
	claims, signer, err := utils.VerifyRequestJWT(token, *req)
	if err != nil {
		h.lggr.Errorw("Failed to verify JWT", "error", err)
		return nil, err
	}

	if h.jwtCache.isReplay(claims.ID) {
		h.lggr.Warnw("JWT token has already been used", "workflowID", workflowID, "signer", signer.Hex(), "jti", claims.ID)
		return nil, errors.New("JWT token has already been used. Please generate a new one with new id (jti)")
	}

	keys, exists := h.authorizedKeys[workflowID]
	if !exists {
		h.lggr.Errorw("Workflow ID not found in authorized keys", "workflowID", workflowID)
		return nil, fmt.Errorf("workflow ID %s not found", workflowID)
	}
	key := gateway.AuthorizedKey{
		KeyType:   gateway.KeyTypeECDSAEVM,
		PublicKey: strings.ToLower(signer.Hex()),
	}
	if _, exists = keys[key]; !exists {
		h.lggr.Errorw("Signer not found in authorized keys", "signer", signer.Hex())
		return nil, fmt.Errorf("signer '%s' is not authorized for workflow '%s'. Ensure that the signer is registered in the workflow definition", signer.Hex(), workflowID)
	}
	h.jwtCache.recordUsage(claims.ID)

	return &key, nil
}
