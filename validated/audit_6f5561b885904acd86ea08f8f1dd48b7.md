The claimed file location is incorrect to begin with — `lockup/src/internal.rs` contains no `attached_deposit` or `MIN_ATTACHED_BALANCE` logic at all. The relevant refund logic lives in `staking-pool-factory/src/lib.rs`.

Tracing the actual binding: in `create_staking_pool`, `env::attached_deposit()` is read once at line 145 for the assertion, and read again at line 174 (`transfer(env::attached_deposit())`) and line 189 (`env::attached_deposit().into()` passed as the callback's `attached_deposit: U128` param) — all within the same synchronous function execution, so all three reads necessarily return the identical value captured for that one receipt/call. [1](#0-0) 

That `U128` value is then passed unchanged as a parameter through the promise chain into `on_staking_pool_create`; the callback itself is invoked with `NO_DEPOSIT`, so `env::attached_deposit()` inside the callback is `0` and is never used for the refund — only the passed-through parameter `attached_deposit.0` is used in `Promise::new(predecessor_account_id).transfer(attached_deposit.0)`. [2](#0-1) 

Since the value is captured exactly once and threaded through as an immutable parameter rather than re-derived from `env::attached_deposit()` at refund time, there is no path for divergence between the amount debited from the attacker at call time and the amount refunded. This is confirmed by the existing unit test `test_create_staking_pool_rollback` and the sim test `create_staking_pool_bad_staking_key`, both of which assert the refunded amount equals exactly the originally attached deposit.     pub fn on_staking_pool_create(
        &mut self,
        staking_pool_account_id: AccountId,
        attached_deposit: U128,
        predecessor_account_id: AccountId,
    ) -> PromiseOrValue<bool> {
        assert_self();

        let staking_pool_created = is_promise_success();

        if staking_pool_created {
            env::log(
                format!(
                    "The staking pool @{} was successfully created. Whitelisting...",
                    staking_pool_account_id
                )
                .as_bytes(),
            );
            ext_whitelist::add_staking_pool(
                staking_pool_account_id,
                &self.staking_pool_whitelist_account_id,
                NO_DEPOSIT,
                gas::WHITELIST_STAKING_POOL,
            )
            .into()
        } else {
            self.staking_pool_account_ids
                .remove(&staking_pool_account_id);
            env::log(
                format!(
                    "The staking pool @{} creation has failed. Returning attached deposit of {} to @{}",
                    staking_pool_account_id,
                    attached_deposit.0,
                    predecessor_account_id
                ).as_bytes()
            );
            Promise::new(predecessor_account_id).transfer(attached_deposit.0);
            PromiseOrValue::Value(false)
        }

#[test]
fn create_staking_pool_bad_staking_key() {
    let (mut r, foundation, owner) = setup_factory();

    let res: U128 = view_factory(&r, "get_min_attached_balance", "");
    assert_eq!(res.0, ntoy(30));

    let res: u64 = view_factory(&r, "get_number_of_staking_pools_created", "");
    assert_eq!(res, 0);

    let _owner_staking_account = foundation
        .create_external(&mut r, OWNER_STAKING_ACCOUNT_ID.to_string(), ntoy(30))
        .unwrap();
    let bad_staking_key: Base58PublicKey = vec![0; 33].try_into().unwrap();

    let owner_balance = owner.account(&r).amount;

    let res = owner
        .function_call(
            &mut r,
            FACTORY_ACCOUNT_ID,
            "create_staking_pool",
            &serde_json::to_vec(&json!({
                "staking_pool_id": STAKING_POOL_ID.to_string(),
                "owner_id": OWNER_STAKING_ACCOUNT_ID.to_string(),
                "stake_public_key": bad_staking_key.clone(),
                "reward_fee_fraction": {
                    "numerator": 10,
                    "denominator": 100,
                }
            }))
            .unwrap(),
            ntoy(31),
        )
        .unwrap();
    assert_eq!(res.status, ExecutionStatus::SuccessValue(b"false".to_vec()));

    // Check the factory didn't store the pool.
    let res: u64 = view_factory(&r, "get_number_of_staking_pools_created", "");
    assert_eq!(res, 0);

    // Check the pool was not whitelisted.
    let is_whitelisted: bool = call_view(
        &r,
        &STAKING_POOL_WHITELIST_ACCOUNT_ID,
        "is_whitelisted",
        &serde_json::to_string(
            &json!({ "staking_pool_account_id": STAKING_POOL_ACCOUNT_ID.to_string() }),
        )
        .unwrap(),
    );
    assert!(!is_whitelisted);

    // Check the amount was refunded
    let new_owner_balance = owner.account(&r).amount;
    assert_eq!(new_owner_balance, owner_balance);

    // Pool account was not created
    let pool_account = r.view_account(&STAKING_POOL_ACCOUNT_ID.to_string());
    assert_eq!(pool_account, None);
}
